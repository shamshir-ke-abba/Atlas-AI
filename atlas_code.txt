const chat = document.getElementById("chat");
const input = document.getElementById("message");
const sendButton = document.getElementById("send");


function addMessage(
    sender,
    text,
    type
) {

    const div =
        document.createElement("div");

    div.className =
        `message ${type}`;

    const senderElement =
        document.createElement("b");

    senderElement.textContent =
        `${sender}: `;

    const textElement =
        document.createElement("span");

    textElement.textContent =
        text;

    div.appendChild(
        senderElement
    );

    div.appendChild(
        textElement
    );

    chat.appendChild(
        div
    );

    chat.scrollTop =
        chat.scrollHeight;

    return textElement;
}


async function sendMessage() {

    const text =
        input.value.trim();

    if (!text)
        return;


    addMessage(
        "You",
        text,
        "user"
    );


    input.value = "";

    sendButton.disabled = true;
    input.disabled = true;


    const responseElement =
        addMessage(
            "Atlas",
            "",
            "atlas"
        );


    try {

        const response =
            await fetch(
                "/stream",
                {
                    method: "POST",

                    headers: {
                        "Content-Type":
                            "application/json",

                        "Accept":
                            "application/x-ndjson"
                    },

                    body: JSON.stringify({
                        message: text
                    })
                }
            );


        if (!response.ok) {

            throw new Error(
                `HTTP ${response.status}`
            );
        }


        if (!response.body) {

            throw new Error(
                "Streaming is unavailable."
            );
        }


        const reader =
            response.body.getReader();

        const decoder =
            new TextDecoder("utf-8");


        let buffer = "";


        while (true) {

            const {
                value,
                done
            } = await reader.read();


            if (done)
                break;


            buffer +=
                decoder.decode(
                    value,
                    {
                        stream: true
                    }
                );


            const lines =
                buffer.split("\n");


            buffer =
                lines.pop() || "";


            for (
                const line of lines
            ) {

                const trimmed =
                    line.trim();


                if (!trimmed)
                    continue;


                let data;


                try {

                    data =
                        JSON.parse(
                            trimmed
                        );

                } catch (error) {

                    console.error(
                        "Invalid stream data:",
                        trimmed
                    );

                    continue;
                }


                // ----------------------------------
                // Backend error
                // ----------------------------------

                if (data.error) {

                    throw new Error(
                        data.error
                    );
                }


                // ----------------------------------
                // Normal chunk
                // ----------------------------------

                if (
                    typeof data.chunk
                    === "string"
                ) {

                    responseElement
                        .textContent +=
                            data.chunk;

                    chat.scrollTop =
                        chat.scrollHeight;
                }
            }
        }


        // Process any final buffered line.

        const finalLine =
            buffer.trim();


        if (finalLine) {

            try {

                const data =
                    JSON.parse(
                        finalLine
                    );


                if (data.error) {

                    throw new Error(
                        data.error
                    );
                }


                if (
                    typeof data.chunk
                    === "string"
                ) {

                    responseElement
                        .textContent +=
                            data.chunk;
                }

            } catch (error) {

                if (
                    error instanceof Error
                    &&
                    error.message !==
                    "Unexpected end of JSON input"
                ) {

                    throw error;
                }
            }
        }


    } catch (error) {

        console.error(
            "Atlas request failed:",
            error
        );


        responseElement.textContent =
            `Error: ${error.message}`;

    } finally {

        sendButton.disabled = false;
        input.disabled = false;

        input.focus();
    }
}


sendButton.addEventListener(
    "click",
    sendMessage
);


input.addEventListener(
    "keydown",
    (event) => {

        if (
            event.key === "Enter"
            && !event.shiftKey
        ) {

            event.preventDefault();

            sendMessage();
        }
    }
);

class ChatRequest(BaseModel):
    message: str


@app.post("/chat")
def chat(request: ChatRequest):

    logger.info(
        "Chat request received."
    )

    try:

        response = controller.chat(
            request.message
        )

        logger.info(
            "Chat request completed."
        )

        return {
            "response": response
        }

    except Exception:

        logger.exception(
            "Chat request failed."
        )

        raise HTTPException(
            status_code=500,
            detail="Atlas encountered an internal error.",
        )


@app.post("/stream")
def stream(request: ChatRequest):

    logger.info(
        "Streaming request received."
    )

    def generate():

        try:

            for chunk in controller.stream_chat(
                request.message
            ):

                yield (
                    json.dumps(
                        {
                            "chunk": chunk
                        }
                    )
                    + "\n"
                )

            logger.info(
                "Streaming request completed."
            )

        except Exception:

            logger.exception(
                "Streaming request failed."
            )

            # Send an error event to the client
            yield (
                json.dumps(
                    {
                        "error":
                        "Atlas encountered an internal error."
                    }
                )
                + "\n"
            )

    return StreamingResponse(
        generate(),
        media_type="application/x-ndjson",
    )

"""
Time Plugin
"""

from datetime import datetime

from plugins.base import AtlasPlugin


class TimePlugin(AtlasPlugin):

    name = "time"

    description = (
        "Returns the current time."
    )

    triggers = [
        "what time is it",
        "what's the time",
        "what is the time",
        "tell me the time",
        "current time",
        "current time please",
        "time now",
        "what time",
        "time please",
    ]

    def can_handle(
        self,
        message,
    ):

        text = (
            message
            .lower()
            .strip()
            .rstrip("?!.")
        )

        for trigger in self.triggers:

            if text == trigger:
                return True

            if text.startswith(
                trigger + " "
            ):
                return True

        return False

    def run(
        self,
        **kwargs,
    ):

        return datetime.now().strftime(
            "%I:%M %p"
        )

"""

Main Atlas Window.

from PySide6.QtWidgets import (
    QHBoxLayout,
    QMainWindow,
    QWidget,
)

from gui.widgets.sidebar import AtlasSidebar
from gui.widgets.chat import AtlasChat
from gui.widgets.statusbar import AtlasStatusBar


class AtlasWindow(QMainWindow):

    def __init__(
        self,
        bus,
        state,
        controller,
    ):

        super().__init__()

        self.bus = bus
        self.state = state
        self.controller = controller

        self.setup_window()
        self.build_ui()

    # =====================================================
    # Window
    # =====================================================

    def setup_window(self):

        self.setWindowTitle(
            "Atlas AI"
        )

        self.resize(
            1400,
            900,
        )

        self.setMinimumSize(
            1100,
            700,
        )

    # =====================================================
    # UI
    # =====================================================

    def build_ui(self):

        root = QWidget()

        root.setObjectName(
            "AtlasWindow"
        )

        self.setCentralWidget(
            root
        )

        layout = QHBoxLayout(
            root
        )

        layout.setContentsMargins(
            0,
            0,
            0,
            0,
        )

        layout.setSpacing(
            0
        )

        # ---------------------------------
        # Sidebar
        # ---------------------------------

        self.sidebar = AtlasSidebar(
            bus=self.bus,
            controller=self.controller,
        )

        self.sidebar.setMinimumWidth(
            240
        )

        # ---------------------------------
        # Chat
        # ---------------------------------

        self.chat = AtlasChat(
            self.bus
        )

        self.chat.add_message(
            "Atlas",
            "Hello Saad, systems online."
        )

        # ---------------------------------
        # Conversation Synchronization
        # ---------------------------------

        self.sidebar.conversation_loaded.connect(
            self.handle_conversation_loaded
        )

        # ---------------------------------
        # Status
        # ---------------------------------

        self.status = AtlasStatusBar(
            self.bus
        )

        self.status.setMinimumWidth(
            260
        )

        # ---------------------------------
        # Layout
        # ---------------------------------

        layout.addWidget(
            self.sidebar
        )

        layout.addWidget(
            self.chat,
            1,
        )

    # =====================================================
    # Conversation UI
    # =====================================================

    def handle_conversation_loaded(
        self,
        conversation,
    ):

        # ---------------------------------
        # New Chat / Empty Conversation
        # ---------------------------------

        if conversation is None:

            self.chat.clear_messages()

            self.chat.add_message(
                "Atlas",
                "Hello Saad, systems online."
            )

            return

        # ---------------------------------
        # Existing Conversation
        # ---------------------------------

        self.chat.load_conversation(
            conversation
        )

    # =====================================================
    # History
    # =====================================================

    def refresh_history(self):

        self.sidebar.refresh_history()

                style_path,

        )

        )

        layout.addWidget(
            self.input_bar
        )

    # =====================================================
    # Messages
    # =====================================================

    def add_message(
        self,
        sender,
        text,
    ):

        message = AtlasMessage(
            sender,
            text,
        )

        self.messages_layout.insertWidget(
            self.messages_layout.count() - 1,
            message,
        )

        return message

    # =====================================================
    # Clear Messages
    # =====================================================

    def clear_messages(self):

        while self.messages_layout.count() > 1:

            item = self.messages_layout.takeAt(
                0
            )

            widget = item.widget()

            if widget is not None:

                widget.deleteLater()

        self.current_ai_message = None

    # =====================================================
    # Load Conversation
    # =====================================================

        self,
        conversation,
    ):

        self.clear_messages()

        for message in conversation.history():

            role = message.get(
                "role"
            )

            text = message.get(
                "content",
                "",
            )

            if role == "user":

                sender = "User"

                sender = "Atlas"

                "Atlas",
                event.text,

        else:

            self.current_ai_message.set_text(

                + event.text
            )

        self.scroll_to_bottom()

    # =====================================================

    def scroll_to_bottom(self):

        scrollbar = (
            self.scroll.verticalScrollBar()
        )

        scrollbar.setValue(
            scrollbar.maximum()
        )

"""
Atlas Brain

Coordinates memory, context,
conversation and the model.
"""

from ai.model import AtlasModel
from ai.context_builder import ContextBuilder
from ai.memory_patterns import PATTERNS
from ai.planner import Planner

from core.memory import MemoryManager
from core.tool_manager import ToolManager
from core.conversation import ConversationManager
from core.conversation_store import ConversationStore
from core.intent import IntentRouter


class AtlasBrain:

    def __init__(self):

        self.model = AtlasModel()

        self.memory = MemoryManager()

        self.tools = ToolManager()

        self.context = ContextBuilder()

        self.conversation = ConversationManager()

        self.conversation_store = ConversationStore()

        self.router = IntentRouter(
            self.tools.plugins.plugins
        )

        self.planner = Planner(
            self.router
        )

    # ==================================================
    # Conversation
    # ==================================================

    def clear_conversation(self):

        self.conversation = ConversationManager()

    def new_conversation(self):

        self._save_current_conversation()

        self.conversation = ConversationManager()

        return self.conversation.get_id()

    def load_conversation(
        self,
        conversation_id,
    ):

        # Save the currently active conversation first.
        self._save_current_conversation()

        data = self.conversation_store.load(
            conversation_id
        )

        if data is None:
            return False

        # --------------------------------------------------
        # ConversationStore may return either:
        #
        # 1. A dictionary from JSON
        # 2. A ConversationManager
        #
        # Handle both safely.
        # --------------------------------------------------

        if isinstance(
            data,
            ConversationManager,
        ):

            self.conversation = data

        elif isinstance(
            data,
            dict,
        ):

            conversation = ConversationManager()

            conversation.load(
                data
            )

            self.conversation = conversation

        else:

            return False

        return True

    def list_conversations(self):

        return (
            self.conversation_store.list_conversations()
        )

    def delete_conversation(
        self,
        conversation_id,
    ):

        deleted = (
            self.conversation_store.delete(
                conversation_id
            )
        )

        if not deleted:
            return False

        # If the deleted conversation was active,
        # immediately switch to a fresh conversation.

        if (
            conversation_id
            == self.conversation.get_id()
        ):

            self.conversation = ConversationManager()

        return True

    def rename_conversation(
        self,
        conversation_id,
        title,
    ):

        title = title.strip()

        if not title:
            return False

        data = self.conversation_store.load(
            conversation_id
        )

        if data is None:
            return False

        # --------------------------------------------------
        # Convert stored data into ConversationManager
        # if necessary.
        # --------------------------------------------------

        if isinstance(
            data,
            ConversationManager,
        ):

            conversation = data

        elif isinstance(
            data,
            dict,
        ):

            conversation = ConversationManager()

            conversation.load(
                data
            )

        else:

            return False

        conversation.set_title(
            title
        )

        self.conversation_store.save(
            conversation
        )

        # Keep the active conversation synchronized.

        if (
            conversation_id
            == self.conversation.get_id()
        ):

            self.conversation.set_title(
                title
            )

        return True

    def get_active_conversation_id(self):

        return self.conversation.get_id()

    # ==================================================
    # Processing
    # ==================================================

    def process(
        self,
        message,
    ):

        text = message.lower()

        # ----------------------------------
        # Conversation
        # ----------------------------------

        self.conversation.add_user(
            message
        )

        # ----------------------------------
        # Automatic Conversation Title
        # ----------------------------------

        self._auto_title(
            message
        )

        # ----------------------------------
        # Memory Extraction
        # ----------------------------------

        for phrase, category, key in PATTERNS:

            if phrase in text:

                index = text.find(
                    phrase
                )

                value = message[
                    index + len(phrase):
                ].strip()

                self.memory.remember(
                    category,
                    key,
                    value,
                )

                break

        # ----------------------------------
        # Memory Recall
        # ----------------------------------

        recall = {

            "what's my name": (
                "identity",
                "name",
            ),

            "what is my name": (
                "identity",
                "name",
            ),

            "who am i": (
                "identity",
                "name",
            ),

            "where do i live": (
                "identity",
                "city",
            ),

            "what city do i live in": (
                "identity",
                "city",
            ),
        }

        for trigger, memory in recall.items():

            if trigger in text:

                value = self.memory.recall(
                    memory[0],
                    memory[1],
                )

                if value:

                    self.conversation.add_assistant(
                        value
                    )

                    self._save_current_conversation()

                    yield value

                else:

                    response = "I don't know."

                    self.conversation.add_assistant(
                        response
                    )

                    self._save_current_conversation()

                    yield response

                return

        # ----------------------------------
        # Planning
        # ----------------------------------

        plan = self.planner.plan(
            message
        )

        # ----------------------------------
        # Tool Execution
        # ----------------------------------

        if plan.action == Planner.TOOL:

            plugin = plan.plugin

            result = self.tools.execute(
                plugin.name,
                expression=plan.arguments,
            )

            self.conversation.add_assistant(
                result
            )

            self._save_current_conversation()

            yield result

            return

        # ----------------------------------
        # Build Context
        # ----------------------------------

        context = self.context.build(
            self.conversation.last(),
            message,
        )

        # ----------------------------------
        # Model
        # ----------------------------------

        reply = ""

        for chunk in self.model.generate(
            context,
        ):

            reply += chunk

            yield chunk

        # ----------------------------------
        # Save Conversation
        # ----------------------------------

        self.conversation.add_assistant(
            reply
        )

        self._save_current_conversation()

    # ==================================================
    # Automatic Title
    # ==================================================

    def _auto_title(
        self,
        message,
    ):

        # Only automatically name fresh chats.

        if self.conversation.get_title() != "New Chat":
            return

        text = " ".join(
            message.strip().split()
        )

        if not text:
            return

        words = text.split()

        if len(words) > 7:

            title = " ".join(
                words[:7]
            ) + "..."

        else:

            title = text

        self.conversation.set_title(
            title
        )

    # ==================================================
    # Internal Persistence
    # ==================================================

    def _save_current_conversation(self):

        if not self.conversation.history():
            return

        self.conversation_store.save(
            self.conversation
        )

        '५': '5', '६': '6', '७': '7', '८': '8', '९': '9',
    }

    @staticmethod
    def contains_devanagari(text: str) -> bool:
        """Check if text contains Devanagari script."""
        return any(
            ord(char) in LanguageTranslator.DEVANAGARI_RANGE
            for char in text
        )

    @staticmethod
    def contains_hinglish_words(text: str) -> bool:
        """Check if text contains common Hinglish words."""
        text_lower = text.lower()
        # Split into words and check if any are Hinglish indicators
        words = re.findall(r'\b\w+\b', text_lower)
        return any(word in LanguageTranslator.HINGLISH_INDICATORS for word in words)

    @staticmethod
    def is_hinglish(text: str) -> bool:
        """
        Detect if text is Hinglish (mix of English and Hindi).
        Checks for:
        1. Devanagari script
        2. Common Hinglish words
        """
        has_devanagari = LanguageTranslator.contains_devanagari(text)
        has_hinglish_words = LanguageTranslator.contains_hinglish_words(text)
        
        return has_devanagari or has_hinglish_words

    @staticmethod
    def transliterate_devanagari(text: str) -> str:
        """
        Basic Devanagari to Latin transliteration.
        Handles pure Hindi text conversion.
        """
        result = text
        for devanagari, latin in LanguageTranslator.DEVANAGARI_TO_LATIN.items():
            result = result.replace(devanagari, latin)
        return result

    @staticmethod
    def apply_hinglish_rules(text: str) -> str:
        """
        Apply common Hinglish-to-English translation rules.
        Case-insensitive pattern matching.
        """
        result = text
        
        # Apply transliteration patterns (case-insensitive)
        for pattern, replacement in LanguageTranslator.HINGLISH_TO_ENGLISH.items():
            compiled = re.compile(pattern, re.IGNORECASE)
            result = compiled.sub(replacement, result)
        
        return result

    @staticmethod
    def normalize_hinglish(text: str) -> str:
        """
        Normalize Hinglish text by:
        1. Transliterating Devanagari to Latin
        2. Applying Hinglish translation rules
        3. Cleaning up extra spaces
        """
        # Step 1: Transliterate Devanagari if present
        if LanguageTranslator.contains_devanagari(text):
            text = LanguageTranslator.transliterate_devanagari(text)
        
        # Step 2: Apply Hinglish rules
        text = LanguageTranslator.apply_hinglish_rules(text)
        
        # Step 3: Clean up extra spaces
        text = re.sub(r'\s+', ' ', text).strip()
        
        return text

    @staticmethod
    def detect_and_convert(text: str) -> Tuple[str, bool, str]:
        """
        Detect language and convert if needed.
        
        Returns:
            Tuple of (converted_text, was_hinglish, original_language)
        """
        is_hinglish = LanguageTranslator.is_hinglish(text)
        
        if is_hinglish:
            converted = LanguageTranslator.normalize_hinglish(text)
            return converted, True, "hinglish"
        
        return text, False, "english"

    @staticmethod
    def add_hinglish_context(message: str) -> str:
        """
        Add Hinglish to the original message if detected.
        Helps preserve intent in the context.
        """
        if LanguageTranslator.is_hinglish(message):
            return f"[User is speaking in Hinglish/Hindi]\n{message}"
        return message


# Convenience functions
def is_hinglish(text: str) -> bool:
    """Quick check if text is Hinglish."""
    return LanguageTranslator.is_hinglish(text)


def normalize_hinglish(text: str) -> str:
    """Normalize Hinglish to English."""
    return LanguageTranslator.normalize_hinglish(text)


def detect_language(text: str) -> str:
    """Detect if text is Hinglish or English."""
    _, is_hin, lang = LanguageTranslator.detect_and_convert(text)
    return lang

"""
Atlas V3 Context Builder

from core.memory import MemoryManager


class ContextBuilder:

    def __init__(self):
        self.memory = MemoryManager()

    def build(
        self,
        history,
        message,
        is_hinglish=False,
    ):
        parts = []

        # --------------------------------------------------
        # Hinglish Context
        # --------------------------------------------------

        if is_hinglish:
            parts.append(
                "NOTE: User may be speaking in Hinglish/Hindi. "
                "Respond clearly and helpfully."
            )

        # --------------------------------------------------
        # System Context
        # --------------------------------------------------

        memory = self.memory.all()

        if memory:

            parts.append(
                "=== USER MEMORY ==="
            )

            for category, values in memory.items():

                parts.append(
                    f"[{category}]"
                )

                if isinstance(values, dict):

                    for key, value in values.items():

                        parts.append(
                            f"{key}: {value}"
                        )

                else:

                    parts.append(
                        str(values)
                    )

        # --------------------------------------------------
        # Conversation
        # --------------------------------------------------

        if history:

            parts.append(
                "\n=== RECENT CHAT ==="
            )

            for item in history:

                role = item["role"].capitalize()

                parts.append(
                    f"{role}: {item['content']}"
                )

        # --------------------------------------------------
        # Current Message
        # --------------------------------------------------

        parts.append(
            "\n=== CURRENT USER MESSAGE ==="
        )

        parts.append(message)

        return "\n".join(parts)

                .get("content", "")
            )

            if content:
                yield content

    def clear_history(self):

    ):

                self.router.extract_arguments(

                plugin=plugin,
                arguments=arguments,
            )

        return Plan(
            action=self.DIRECT

"""
Calculator Plugin

import ast
import operator

from plugins.base import AtlasPlugin


_OPERATORS = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.Mod: operator.mod,
    ast.Pow: operator.pow,
    ast.FloorDiv: operator.floordiv,

    ast.UAdd: operator.pos,
    ast.USub: operator.neg,
}


class CalculatorPlugin(AtlasPlugin):

    name = "calculator"

    description = (
        "Safely evaluates mathematical expressions."
    )

    triggers = [
        "calculate",
        "calc",
        "solve",
    ]

    def can_handle(
        self,
        message,
    ):

        text = message.lower().strip()

        for trigger in self.triggers:

            if text == trigger:
                return True

            if text.startswith(
                trigger + " "
            ):
                return True

        return False

    def _evaluate(
        self,
        node,
    ):

        if isinstance(
            node,
            ast.Constant,
        ):

            if isinstance(
                node.value,
                (int, float),
            ):

                return node.value

            raise ValueError(
                "Invalid constant."
            )

        if isinstance(
            node,
            ast.BinOp,
        ):

            operation = _OPERATORS.get(
                type(node.op)
            )

            if operation is None:
                raise ValueError(
                    "Unsupported operator."
                )

            return operation(
                self._evaluate(
                    node.left
                ),
                self._evaluate(
                    node.right
                ),
            )

        if isinstance(
            node,
            ast.UnaryOp,
        ):

            operation = _OPERATORS.get(
                type(node.op)
            )

            if operation is None:
                raise ValueError(
                    "Unsupported operator."
                )

            return operation(
                self._evaluate(
                    node.operand
                )
            )

        raise ValueError(
            "Unsupported expression."
        )

    def run(
        self,
        expression="",
        **kwargs,
    ):

        try:

            tree = ast.parse(
                expression,
                mode="eval",
            )

            result = self._evaluate(
                tree.body
            )

            return str(result)

        except Exception:

            return "Invalid expression."

    parents=True,

    "%(asctime)s | "
    "%(levelname)s | "
    "%(name)s | "
    "%(message)s"
)


def get_logger(name="atlas"):

    logger = logging.getLogger(name)

    # Prevent duplicate handlers if this module
    # is imported multiple times.
    if logger.handlers:
        return logger

    logger.setLevel(logging.INFO)

    logger.propagate = False

    formatter = logging.Formatter(
        LOG_FORMAT
    )

    # --------------------------------------------------
    # File Handler
    # --------------------------------------------------

    file_handler = RotatingFileHandler(
        LOG_FILE,
        maxBytes=5 * 1024 * 1024,
        backupCount=3,
        encoding="utf-8",
    )

    file_handler.setFormatter(
        formatter
    )

    logger.addHandler(
        file_handler
    )

    # --------------------------------------------------
    # Console Handler
    # --------------------------------------------------

    console_handler = logging.StreamHandler()

    console_handler.setFormatter(
        formatter
    )

    logger.addHandler(
        console_handler
    )

    return logger

"""

Handles persistent storage of conversations.
The GUI and AI should not need to know
how conversations are stored on disk.
"""

import json
from pathlib import Path

from core.conversation import ConversationManager


class ConversationStore:

    def __init__(self):

        self.directory = (
            Path(__file__).resolve().parent.parent
            / "data"
            / "conversations"
        )

        self.directory.mkdir(
            parents=True,
            exist_ok=True,
        )

    # ==================================================
    # Save
    # ==================================================

    def save(
        self,
        conversation,
    ):

        data = conversation.to_dict()

        path = self.directory / (
            f"{conversation.get_id()}.json"
        )

        temporary = path.with_suffix(
            ".tmp"
        )

        with open(
            temporary,
            "w",
            encoding="utf-8",
        ) as file:

            json.dump(
                data,
                file,
                indent=4,
                ensure_ascii=False,
            )

        temporary.replace(
            path
        )

    # ==================================================
    # Load
    # ==================================================

    def load(
        self,
        conversation_id,
    ):

        path = self.directory / (
            f"{conversation_id}.json"
        )

        if not path.exists():
            return None

        try:

            with open(
                path,
                "r",
                encoding="utf-8",
            ) as file:

                data = json.load(
                    file
                )

        except (
            json.JSONDecodeError,
            OSError,
        ):

            return None

        conversation = ConversationManager()

        conversation.load(
            data
        )

        return conversation

    # ==================================================
    # List Conversations
    # ==================================================

    def list_conversations(self):

        conversations = []

        for path in self.directory.glob(
            "*.json"
        ):

            try:

                with open(
                    path,
                    "r",
                    encoding="utf-8",
                ) as file:

                    data = json.load(
                        file
                    )

                conversations.append(
                    {
                        "id": data.get(
                            "id"
                        ),
                        "title": data.get(
                            "title",
                            "New Chat",
                        ),
                        "created_at": data.get(
                            "created_at",
                            "",
                        ),
                        "updated_at": data.get(
                            "updated_at",
                            "",
                        ),
                    }
                )

            except (
                json.JSONDecodeError,
                OSError,
            ):

                continue

        conversations.sort(
            key=lambda item: item.get(
                "updated_at",
                "",
            ),
            reverse=True,
        )

        return conversations

    # ==================================================
    # Delete
    # ==================================================

    def delete(
        self,
        conversation_id,
    ):

        path = self.directory / (
            f"{conversation_id}.json"
        )

        if path.exists():

            try:

                path.unlink()

                return True

            except OSError:

                return False

        return False

    # ==================================================
    # Exists
    # ==================================================

    def exists(
        self,
        conversation_id,
    ):

        path = self.directory / (
            f"{conversation_id}.json"
        )

        return path.exists()

    # ==================================================
    # Clear All
    # ==================================================

    def clear_all(self):

        for path in self.directory.glob(
            "*.json"
        ):

            try:

                path.unlink()

            except OSError:

                continue

"""
Atlas V3 Event Bus

The Event Bus allows independent components
to communicate without directly referencing
each other.
"""

from collections import defaultdict
from typing import Callable, Type

from core.events import Event


class EventBus:
    """
    Publish/Subscribe event bus.
    """

    def __init__(self):

        self._listeners = defaultdict(list)

    # ------------------------------------------------------
    # Subscribe
    # ------------------------------------------------------

    def subscribe(
        self,
        event_type: Type[Event],
        callback: Callable,
    ):

        if callback not in self._listeners[event_type]:

            self._listeners[event_type].append(
                callback
            )

    # ------------------------------------------------------
    # Unsubscribe
    # ------------------------------------------------------

    def unsubscribe(
        self,
        event_type: Type[Event],
        callback: Callable,
    ):

        if callback in self._listeners[event_type]:

            self._listeners[event_type].remove(
                callback
            )

    # ------------------------------------------------------
    # Publish
    # ------------------------------------------------------

    def publish(self, event: Event):

        for callback in self._listeners[type(event)]:

            callback(event)

"""
Atlas V3 Intent Router
"""


class IntentRouter:

    def __init__(self, plugins):
        self.plugins = plugins

    def match(self, message):

        text = message.lower().strip()

        best_plugin = None
        best_score = 0

        for plugin in self.plugins.values():

            if not plugin.enabled:
                continue

            if not plugin.can_handle(message):
                continue

            trigger_score = 0

            for trigger in plugin.triggers:

                trigger = trigger.lower().strip()

                if not trigger:
                    continue

                # Exact command
                if text == trigger:
                    trigger_score = max(
                        trigger_score,
                        1000,
                    )

                # Command followed by arguments
                elif text.startswith(
                    trigger + " "
                ):
                    trigger_score = max(
                        trigger_score,
                        100,
                    )

                # Trigger appears as a complete word/phrase
                elif (
                    f" {trigger} "
                    in f" {text} "
                ):
                    trigger_score = max(
                        trigger_score,
                        len(trigger),
                    )

            # IMPORTANT:
            # No trigger match = plugin does not match.
            if trigger_score == 0:
                continue

            score = (
                trigger_score
                + plugin.priority
            )

            if score > best_score:

                best_score = score
                best_plugin = plugin

        return best_plugin

    def extract_arguments(
        self,
        plugin,
        message,
    ):

        expression = message.strip()

        for trigger in sorted(
            plugin.triggers,
            key=len,
            reverse=True,
        ):

            trigger_lower = trigger.lower()

            if expression.lower().startswith(
                trigger_lower
            ):

                expression = expression[
                    len(trigger):
                ].strip()

                break

        return expression.rstrip(
            "?!.,;:"
        ).strip()

import importlib
import inspect
import pkgutil

from plugins.base import AtlasPlugin


class PluginManager:

    def __init__(self):

        self.plugins = {}

        self.load_plugins()

    # ==================================================
    # Loading
    # ==================================================

    def load_plugins(self):

        import plugins.system

        for _, module_name, _ in pkgutil.iter_modules(
            plugins.system.__path__
        ):

            module = importlib.import_module(
                f"plugins.system.{module_name}"
            )

            for _, obj in inspect.getmembers(
                module,
                inspect.isclass,
            ):

                if not issubclass(
                    obj,
                    AtlasPlugin,
                ):
                    continue

                if obj is AtlasPlugin:
                    continue

                plugin = obj()

                if not plugin.enabled:
                    continue

                if not plugin.name:
                    continue

                self.plugins[
                    plugin.name
                ] = plugin

    # ==================================================
    # Execution
    # ==================================================

    def execute(
        self,
        name,
        **kwargs,
    ):

        plugin = self.plugins.get(name)

        if plugin is None:
            raise ValueError(
                f"Unknown plugin: {name}"
            )

        return plugin.run(
            **kwargs
        )

    # ==================================================
    # Information
    # ==================================================

    def available(self):

        return list(
            self.plugins.keys()
        )

"""
Atlas V3 Memory Manager

Persistent user memory stored in the configured data directory.
"""

import json

from config import MEMORY_FILE


class MemoryManager:

    def __init__(self):
        self.file = MEMORY_FILE

        self.file.parent.mkdir(
            parents=True,
            exist_ok=True,
        )

        if not self.file.exists():
            self._save({})

        else:
            try:
                data = self._load()

                if not isinstance(data, dict):
                    self._save({})

            except (json.JSONDecodeError, OSError):
                self._save({})

    # --------------------------------------------------
    # Internal
    # --------------------------------------------------

    def _load(self):
        try:
            with self.file.open(
                "r",
                encoding="utf-8",
            ) as f:
                data = json.load(f)

            return data if isinstance(data, dict) else {}

        except (
            FileNotFoundError,
            json.JSONDecodeError,
            OSError,
        ):
            return {}

    def _save(self, data):
        temporary = self.file.with_suffix(".tmp")

        with temporary.open(
            "w",
            encoding="utf-8",
        ) as f:
            json.dump(
                data,
                f,
                indent=4,
                ensure_ascii=False,
            )

        temporary.replace(self.file)

    # --------------------------------------------------
    # Public API
    # --------------------------------------------------

    def remember(
        self,
        category,
        key,
        value,
    ):
        data = self._load()

        if category not in data:
            data[category] = {}

        data[category][key] = value

        self._save(data)

    def recall(
        self,
        category,
        key,
    ):
        data = self._load()

        return data.get(
            category,
            {},
        ).get(key)

    def forget(
        self,
        category,
        key,
    ):
        data = self._load()

        if category not in data:
            return

        data[category].pop(
            key,
            None,
        )

        if not data[category]:
            del data[category]

        self._save(data)

    def all(self):
        return self._load()

    def clear(self):
        self._save({})
