"""Atlas V3 Brain

Central AI orchestration layer.

Responsibilities:
- Build conversation-aware prompts
- Maintain conversation history
- Handle deterministic conversation recall
- Route messages through ToolManager
- Execute deterministic tools
- Generate normal responses through AtlasModel
- Maintain compatibility with Controller and AIWorker
"""

from typing import Any, Dict, Optional

from core.logger import logger
from ai.model import AtlasModel
from ai.context_builder import ContextBuilder


class AtlasBrain:
    """Central AI orchestration."""

    def __init__(
        self,
        model=None,
        memory=None,
        tools=None,
    ):

        self.model = (
            model
            if model is not None
            else AtlasModel()
        )

        self.memory = memory
        self.tools = tools

        self.active_conversation = None
        self.conversation_history = []

        self.context_builder = ContextBuilder()

        # Web-search state exposed to the browser runtime.
        self.last_web_query = ""
        self.last_web_sources = []

        logger.info(
            "AtlasBrain initialized"
        )

        if self.tools is not None:

            try:

                logger.info(
                    "Brain tool system loaded: %s",
                    self.tools.available(),
                )

            except Exception as exc:

                logger.warning(
                    "Could not inspect tool system: %s",
                    exc,
                )

    # ==================================================
    # CONVERSATION
    # ==================================================

    def set_active_conversation(
        self,
        conversation,
    ):

        self.active_conversation = conversation

        self.conversation_history = (
            self._extract_history(
                conversation
            )
        )

        logger.debug(
            "Active conversation updated: %d messages",
            len(
                self.conversation_history
            ),
        )

    # --------------------------------------------------

    def set_conversation_history(
        self,
        history,
    ):

        if history is None:

            self.conversation_history = []

            return

        if isinstance(
            history,
            list,
        ):

            self.conversation_history = list(
                history
            )

            logger.debug(
                "Conversation history set: %d messages",
                len(
                    self.conversation_history
                ),
            )

            return

        self.conversation_history = (
            self._extract_history(
                history
            )
        )

    # --------------------------------------------------

    def get_active_conversation_id(
        self,
    ) -> Optional[str]:

        conversation = (
            self.active_conversation
        )

        if conversation is None:
            return None

        if hasattr(
            conversation,
            "get_id",
        ):

            try:

                result = conversation.get_id()

                if result:
                    return result

            except Exception:
                pass

        conversation_id = getattr(
            conversation,
            "id",
            None,
        )

        if conversation_id:
            return conversation_id

        if isinstance(
            conversation,
            dict,
        ):

            return conversation.get(
                "id"
            )

        return None

    # ==================================================
    # MAIN PROCESSING
    # ==================================================

    def process_message(
        self,
        message: str,
        conversation_history=None,
    ) -> str:

        message = str(
            message
        ).strip()

        if not message:

            return (
                "Error: Empty message"
            )

        logger.info(
            "Brain processing message: %s",
            message,
        )

        try:

            if conversation_history is not None:

                self.set_conversation_history(
                    conversation_history
                )

            # ------------------------------------------
            # Deterministic recall
            # ------------------------------------------

            recall_response = (
                self._handle_conversation_recall(
                    message
                )
            )

            if recall_response is not None:

                self._update_memory(
                    message,
                    recall_response,
                )

                return recall_response

            # ------------------------------------------
            # Context
            # ------------------------------------------

            context = self._build_context(
                message
            )

            # ------------------------------------------
            # Routing
            # ------------------------------------------

            handler = self._route_message(
                message
            )

            # ------------------------------------------
            # Execute
            # ------------------------------------------

            response = handler(
                message,
                context,
            )

            if response is None:

                response = (
                    "No response generated."
                )

            response = str(
                response
            ).strip()

            if not response:

                response = (
                    "No response generated."
                )

            self._update_memory(
                message,
                response,
            )

            return response

        except Exception as exc:

            logger.exception(
                "Brain processing failed"
            )

            return (
                f"Error: {exc}"
            )

    # ==================================================
    # WEB-FRIENDLY STREAMING
    # ==================================================

    def stream_message(
        self,
        message: str,
        conversation_history=None,
    ):
        """Yield response chunks for headless/browser clients.

        The desktop controller uses process_message().  The browser needs
        an iterator, so this method keeps the same routing logic while
        exposing model/tool output incrementally.
        """
        message = str(message or "").strip()
        if not message:
            yield "Error: Empty message"
            return

        if conversation_history is not None:
            self.set_conversation_history(conversation_history)

        try:
            recall_response = self._handle_conversation_recall(message)
            if recall_response is not None:
                self._update_memory(message, recall_response)
                yield recall_response
                return

            context = self._build_context(message)
            tool = None
            if self.tools is not None and hasattr(self.tools, "match"):
                try:
                    tool = self.tools.match(message)
                except Exception as exc:
                    logger.warning("Web stream tool routing failed: %s", exc)

            if tool is not None:
                tool_name = str(getattr(tool, "name", "")).lower()
                if tool_name == "web_search":
                    response = self._run_web_search_tool(tool, message)
                else:
                    response = self._run_tool(tool, message)
                self._update_memory(message, response)
                yield response
                return

            prompt = context.get("prompt", "") or self._fallback_prompt(
                context.get("conversation_history", []),
                message,
            )

            parts = []
            for chunk in self.model.generate(prompt, stream=True):
                if chunk is None:
                    continue
                text = str(chunk)
                if text:
                    parts.append(text)
                    yield text

            response = "".join(parts).strip()
            if not response:
                response = "No response generated."
                yield response

            self._update_memory(message, response)

        except Exception as exc:
            logger.exception("Brain streaming failed")
            error = f"Error: {exc}"
            yield error

    # ==================================================
    # WEB SEARCH ANSWERING
    # ==================================================

    def _run_web_search_tool(self, tool, message: str) -> str:
        """Search the web, then ask Atlas to answer from the evidence."""
        try:
            query = message
            if hasattr(tool, "_clean_query"):
                query = tool._clean_query(message)

            self.last_web_query = str(query).strip()
            results = []
            if hasattr(tool, "search_results"):
                results = tool.search_results(query)

            if not results:
                raw = tool.run(expression=message)
                self.last_web_sources = []
                return str(raw).strip()

            self.last_web_sources = [
                {
                    "title": item.get("title", "Source"),
                    "url": item.get("url", ""),
                    "snippet": item.get("snippet", ""),
                }
                for item in results
                if item.get("url")
            ]

            evidence_lines = []
            for index, item in enumerate(results, 1):
                evidence_lines.append(
                    f"[Source {index}] {item.get('title', 'Untitled')}\n"
                    f"URL: {item.get('url', '')}\n"
                    f"Snippet: {item.get('snippet', '')}"
                )

            prompt = (
                "You are Atlas, a helpful local AI assistant.\n\n"
                "Answer the user's CURRENT question using the web evidence below.\n"
                "Give the direct answer FIRST. Do not output a search-result list.\n"
                "Do not say you cannot browse. Do not invent facts that are not supported\n"
                "by the evidence. If the evidence is insufficient, say so clearly.\n"
                "Keep the answer concise unless the question needs explanation.\n\n"
                "CURRENT USER QUESTION:\n"
                f"{message}\n\n"
                "WEB EVIDENCE:\n"
                + "\n\n".join(evidence_lines)
                + "\n\nANSWER:"
            )

            answer_parts = []
            for chunk in self.model.generate(prompt, stream=True):
                if chunk is not None:
                    answer_parts.append(str(chunk))

            answer = "".join(answer_parts).strip()
            if not answer:
                answer = "I found web results, but could not generate an answer from them."

            source_lines = ["", "Sources"]
            for item in self.last_web_sources[:6]:
                source_lines.append(
                    f"• {item['title']} — {item['url']}"
                )

            return answer + "\n" + "\n".join(source_lines)

        except Exception as exc:
            logger.exception("Web search answering failed")
            return f"Web search failed: {exc}"

    # ==================================================
    # RECALL
    # ==================================================

    def _handle_conversation_recall(
        self,
        message: str,
    ):

        if not self.conversation_history:
            return None

        normalized = " ".join(
            str(message)
            .lower()
            .strip()
            .split()
        )

        recall_phrases = (
            "what did i just tell you",
            "what did i tell you",
            "what did i just say",
            "what did i say",
            "what was my last message",
            "what was the last thing i said",
            "what was the last thing i told you",
            "do you remember what i said",
            "do you remember what i told you",
            "do you remember my last message",
        )

        if not any(
            phrase in normalized
            for phrase in recall_phrases
        ):
            return None

        for msg in reversed(
            self.conversation_history
        ):

            if not isinstance(
                msg,
                dict,
            ):
                continue

            is_user = msg.get(
                "is_user"
            )

            sender = str(
                msg.get(
                    "sender",
                    "",
                )
            ).strip().lower()

            if (
                is_user is True
                or sender in (
                    "you",
                    "user",
                )
            ):

                text = str(
                    msg.get(
                        "text",
                        "",
                    )
                ).strip()

                if text:

                    return (
                        f'You just told me: "{text}"'
                    )

        return (
            "I don't have a previous user message "
            "available in this conversation."
        )

    # ==================================================
    # CONTEXT
    # ==================================================

    def _build_context(
        self,
        current_message: str = "",
    ) -> Dict[str, Any]:

        history = list(
            self.conversation_history
        )

        context = {
            "conversation": (
                self.active_conversation
            ),
            "conversation_history": history,
            "memory": self._get_memory(),
            "prompt": "",
        }

        try:

            context["prompt"] = (
                self.context_builder.build(
                    history,
                    current_message,
                )
            )

        except Exception as exc:

            logger.warning(
                "ContextBuilder failed: %s",
                exc,
            )

            context["prompt"] = (
                self._fallback_prompt(
                    history,
                    current_message,
                )
            )

        return context

    # --------------------------------------------------

    def _fallback_prompt(
        self,
        history,
        current_message,
    ) -> str:

        prompt = (
            "You are Atlas, a local AI assistant.\n\n"
            "Use the conversation history to maintain "
            "context and answer naturally.\n\n"
        )

        if history:

            prompt += (
                "===== CONVERSATION HISTORY =====\n"
            )

            for msg in history:

                if not isinstance(
                    msg,
                    dict,
                ):
                    continue

                sender = str(
                    msg.get(
                        "sender",
                        "Unknown",
                    )
                )

                text = str(
                    msg.get(
                        "text",
                        "",
                    )
                )

                if sender.lower() in (
                    "you",
                    "user",
                ):

                    prompt += (
                        f"User: {text}\n"
                    )

                else:

                    prompt += (
                        f"Atlas: {text}\n"
                    )

            prompt += (
                "===== END CONVERSATION HISTORY =====\n\n"
            )

        prompt += (
            "===== CURRENT USER MESSAGE =====\n"
            f"User: {current_message}\n"
            "===== END CURRENT USER MESSAGE =====\n\n"
            "Atlas:"
        )

        return prompt

    # ==================================================
    # ROUTING
    # ==================================================

    def _route_message(
        self,
        message: str,
    ):

        if self.tools is not None:

            try:

                # --------------------------------------
                # Preferred ToolManager API
                # --------------------------------------

                if hasattr(
                    self.tools,
                    "match",
                ):

                    tool = self.tools.match(
                        message
                    )

                    if tool is not None:

                        logger.info(
                            "ROUTE: TOOL -> %s",
                            getattr(
                                tool,
                                "name",
                                type(tool).__name__,
                            ),
                        )

                        return (
                            lambda msg, ctx:
                            self._run_tool(
                                tool,
                                msg,
                            )
                        )

            except Exception as exc:

                logger.warning(
                    "Tool routing failed: %s",
                    exc,
                )

        logger.info(
            "ROUTE: MODEL"
        )

        return self._generate_response

    # ==================================================
    # TOOL EXECUTION
    # ==================================================

    def _run_tool(
        self,
        tool,
        message: str,
    ) -> str:

        try:

            if not hasattr(
                tool,
                "run",
            ):

                return (
                    "Error: Invalid tool"
                )

            tool_name = getattr(
                tool,
                "name",
                type(tool).__name__,
            )

            logger.info(
                "Executing tool: %s",
                tool_name,
            )

            # ==================================================
            # WEB SEARCH
            # ==================================================
            # Search results are evidence, not the final response.
            # Let Atlas answer from the retrieved sources.
            # ==================================================

            if tool_name == "web_search":
                return self._run_web_search_tool(
                    tool,
                    message,
                )

            # ==================================================
            # MATH TOOL
            # ==================================================
            #
            # MathPlugin now understands the complete
            # natural-language request.
            #
            # Examples:
            #
            #   differentiate x^2
            #   what is the derivative of x^2?
            #   integrate 2x
            #   solve x^2 - 4 = 0
            #
            # We deliberately pass the original message
            # instead of trying to parse mathematics here.
            #
            # The Brain orchestrates.
            # The MathPlugin interprets.
            # The MathEngine calculates.
            #
            # ==================================================

            if tool_name == "math":

                return self._run_math_tool(
                    tool,
                    message,
                )

            # ==================================================
            # STANDARD TOOL API
            # ==================================================

            try:

                result = tool.run(
                    message
                )

            except TypeError:

                try:

                    result = tool.run(
                        message=message
                    )

                except TypeError:

                    result = tool.run(
                        query=message
                    )

            if result is None:

                return (
                    "No result returned by tool."
                )

            return str(
                result
            ).strip()

        except Exception as exc:

            logger.exception(
                "Tool execution failed"
            )

            return (
                f"Error executing tool: {exc}"
            )

    # ==================================================
    # MATH TOOL
    # ==================================================

    def _run_math_tool(
        self,
        tool,
        message: str,
    ) -> str:

        try:

            # --------------------------------------------------
            # Preferred new MathPlugin API
            # --------------------------------------------------

            operation = ""

            variable = "x"

            # Detect operation if supported
            if hasattr(
                tool,
                "detect_operation",
            ):

                try:

                    operation = (
                        tool.detect_operation(
                            message
                        )
                    )

                except Exception as exc:

                    logger.warning(
                        "Math operation detection failed: %s",
                        exc,
                    )

            # Detect variable if supported
            if hasattr(
                tool,
                "detect_variable",
            ):

                try:

                    variable = (
                        tool.detect_variable(
                            message,
                            message,
                        )
                    )

                except Exception as exc:

                    logger.warning(
                        "Math variable detection failed: %s",
                        exc,
                    )

            logger.info(
                "MATH REQUEST: operation=%s variable=%s",
                operation,
                variable,
            )

            # --------------------------------------------------
            # Execute new structured MathPlugin
            # --------------------------------------------------

            try:

                result = tool.run(
                    expression=message,
                    operation=operation,
                    variable=variable,
                )

                if result is not None:

                    return str(
                        result
                    ).strip()

            except TypeError as exc:

                logger.debug(
                    "Structured math execution unavailable: %s",
                    exc,
                )

            # --------------------------------------------------
            # Backward compatibility
            # --------------------------------------------------

            logger.debug(
                "Falling back to legacy math execution"
            )

            try:

                result = tool.run(
                    message
                )

            except TypeError:

                result = tool.run(
                    message=message
                )

            if result is None:

                return (
                    "No result returned by math tool."
                )

            return str(
                result
            ).strip()

        except Exception as exc:

            logger.exception(
                "Math tool execution failed"
            )

            return (
                f"Error executing math tool: {exc}"
            )

    # ==================================================
    # MODEL
    # ==================================================

    def _generate_response(
        self,
        message: str,
        context: Dict[str, Any],
    ) -> str:

        if self.model is None:

            return (
                "Error: Model not available"
            )

        prompt = context.get(
            "prompt",
            "",
        )

        if not prompt:

            prompt = self._fallback_prompt(
                context.get(
                    "conversation_history",
                    [],
                ),
                message,
            )

        try:

            response_parts = []

            for chunk in self.model.generate(
                prompt,
                stream=True,
            ):

                if chunk is None:
                    continue

                response_parts.append(
                    str(chunk)
                )

            response = "".join(
                response_parts
            ).strip()

            if not response:

                return (
                    "No response generated."
                )

            return response

        except Exception as exc:

            logger.exception(
                "Model generation failed"
            )

            return (
                f"Error generating response: {exc}"
            )

    # ==================================================
    # MEMORY
    # ==================================================

    def _get_memory(
        self,
    ) -> Dict[str, Any]:

        if self.memory is None:
            return {}

        try:

            if hasattr(
                self.memory,
                "all",
            ):

                result = self.memory.all()

                if result is not None:
                    return result

        except Exception as exc:

            logger.warning(
                "Memory retrieval failed: %s",
                exc,
            )

        return {}

    # --------------------------------------------------

    def _update_memory(
        self,
        message: str,
        response: str,
    ):

        if self.memory is None:
            return

        try:

            if hasattr(
                self.memory,
                "remember",
            ):

                try:

                    self.memory.remember(
                        message,
                        response,
                    )

                    return

                except TypeError:

                    try:

                        self.memory.remember(
                            message
                        )

                        return

                    except Exception:
                        pass

            if hasattr(
                self.memory,
                "add",
            ):

                try:

                    self.memory.add(
                        message,
                        response,
                    )

                    return

                except TypeError:
                    pass

        except Exception as exc:

            logger.warning(
                "Memory update failed: %s",
                exc,
            )

    # ==================================================
    # HISTORY
    # ==================================================

    def _extract_history(
        self,
        conversation,
    ):

        if conversation is None:
            return []

        if isinstance(
            conversation,
            list,
        ):

            return list(
                conversation
            )

        if isinstance(
            conversation,
            dict,
        ):

            messages = conversation.get(
                "messages",
                [],
            )

            if isinstance(
                messages,
                list,
            ):

                return list(
                    messages
                )

            return []

        messages = getattr(
            conversation,
            "current_messages",
            None,
        )

        if messages is not None:

            try:
                return list(
                    messages
                )
            except Exception:
                pass

        messages = getattr(
            conversation,
            "messages",
            None,
        )

        if messages is not None:

            try:
                return list(
                    messages
                )
            except Exception:
                pass

        if hasattr(
            conversation,
            "last",
        ):

            try:

                result = conversation.last()

                if result is not None:
                    return list(result)

            except Exception:
                pass

        return []

    # ==================================================
    # STATUS
    # ==================================================

    def status(
        self,
    ) -> Dict[str, Any]:

        model_name = (
            type(
                self.model
            ).__name__
            if self.model is not None
            else None
        )

        tools = []

        if self.tools is not None:

            try:

                if hasattr(
                    self.tools,
                    "available",
                ):

                    tools = list(
                        self.tools.available()
                    )

            except Exception:

                tools = []

        return {
            "model": model_name,
            "memory": (
                type(
                    self.memory
                ).__name__
                if self.memory is not None
                else None
            ),
            "tools": tools,
            "active_conversation": (
                self.get_active_conversation_id()
            ),
            "history_messages": len(
                self.conversation_history
            ),
        }

"""Atlas V3 context builder optimized for a small local model."""

from typing import Any, Dict, List
from config import MAX_HISTORY


class ContextBuilder:
    def __init__(self):
        self.max_history = min(int(MAX_HISTORY or 20), 8)

    def build(self, conversation_history: List[Dict[str, Any]], current_message: str) -> str:
        current_message = str(current_message or "").strip()
        history = conversation_history if isinstance(conversation_history, list) else []

        lines = [
            "You are Atlas, a local AI assistant.",
            "Answer the CURRENT USER MESSAGE directly and naturally.",
            "The current message always has priority over older conversation context.",
            "For a simple greeting, reply briefly and conversationally.",
            "Do not continue an old task unless the current message asks you to.",
            "Do not invent a different question or task.",
            "Do not mention prompts, hidden instructions, token limits, or internal routing.",
            "Use plain text or Markdown only. Never output HTML tags or HTML entities.",
            "Keep simple answers concise; give detail only when useful.",
            "",
            "===== RECENT CONVERSATION =====",
        ]

        recent = history[-self.max_history:]
        if recent:
            for msg in recent:
                if not isinstance(msg, dict):
                    continue
                text = str(msg.get("text", msg.get("content", "")) or "").strip()
                if not text:
                    continue
                sender = str(msg.get("sender", msg.get("role", "User")) or "User").lower()
                label = "User" if sender in ("user", "you") or msg.get("is_user") is True else "Atlas"
                lines.append(f"{label}: {text}")
        else:
            lines.append("(none)")

        lines.extend([
            "===== END RECENT CONVERSATION =====",
            "",
            "===== CURRENT USER MESSAGE =====",
            f"User: {current_message}",
            "===== END CURRENT USER MESSAGE =====",
            "",
            "Atlas:",
        ])
        return "\n".join(lines)

"""
Atlas V3 Knowledge Decider

Decides whether a question should use:

LOCAL -> Local model
WEB   -> Web search
BOTH  -> Local model + web search

Lightweight rule-based routing layer.
"""

import re

from dataclasses import dataclass
from enum import Enum


# ============================================================
# Decision Types
# ============================================================


class Decision(Enum):

    LOCAL = "local"

    WEB = "web"

    BOTH = "both"


@dataclass
class DecisionResult:

    decision: Decision
    reason: str
    confidence: float


# ============================================================
# Knowledge Decider
# ============================================================


class KnowledgeDecider:

    def __init__(self):

        # ----------------------------------------------------
        # Explicit capability questions
        # ----------------------------------------------------

        self.capability_patterns = [

            r"\bwhat tools do you have\b",
            r"\bwhat tools are available\b",
            r"\bwhat capabilities do you have\b",
            r"\bwhat can you do\b",
            r"\bwhat are your capabilities\b",

            r"\bcan you search the web\b",
            r"\bcan you search online\b",
            r"\bcan you browse the web\b",
            r"\bcan you browse online\b",
            r"\bcan you access the internet\b",

            r"\bdo you have web search\b",
            r"\bdo you have web access\b",
            r"\bdo you have internet access\b",
            r"\bdo you have access to the internet\b",

            r"\bcan atlas search the web\b",
            r"\bcan atlas browse the web\b",
            r"\bdoes atlas have web search\b",
            r"\bdoes atlas have web access\b",

    # ============================================================
    # Public API
    # ============================================================

    def decide(
        self,
        question,
    ):

        question = question.strip()

        if not question:

            return DecisionResult(
                Decision.LOCAL,
                "Empty question.",
                1.0,
            )

        text = question.lower()

        # ====================================================
        # 1. Capability questions
        # ====================================================
        #
        # These must happen BEFORE web detection.
        #
        # "Can you search the web?"
        # should NOT actually perform a search.
        #
        # It is asking about Atlas's capability.
        #

        if self._matches(
            text,
            self.capability_patterns,
        ):

            return DecisionResult(
                Decision.LOCAL,
                "User is asking about Atlas capabilities.",
                0.99,
            )

        # ====================================================
        # 2. Explicit web request
        # ====================================================

        if self._matches(
            text,
            self.explicit_web_patterns,
        ):

            return DecisionResult(
                Decision.WEB,
                "User explicitly requested external information.",
                0.99,
            )

        # ====================================================
        # 3. Current information
        # ====================================================

        is_current = self._matches(
            text,
            self.web_patterns,
        )

        has_reasoning = self._matches(
            text,
            self.reasoning_patterns,
        )

        # Current + reasoning = BOTH

        if is_current and has_reasoning:

            return DecisionResult(
                Decision.BOTH,
                "Question requires current information plus reasoning.",
                0.95,
            )

        # Pure current fact = WEB

        if is_current:

            return DecisionResult(
                Decision.WEB,
                "Question depends on current or changing information.",
                0.98,
            )

        # ====================================================
        # 4. New / recent technical knowledge
        # ====================================================

        if self._matches(
            text,
            self.both_patterns,
        ):

            return DecisionResult(
                Decision.BOTH,
                "Question may require current external knowledge plus local reasoning.",
                0.90,
            )

        # ====================================================
        # 5. Current-event entities
        # ====================================================

        if self._matches(
            text,
            self.current_entity_patterns,
        ):

            if has_reasoning:

                return DecisionResult(
                    Decision.BOTH,
                    "Question involves a potentially changing topic and requires reasoning.",
                    0.88,
                )

            return DecisionResult(
                Decision.WEB,
                "Question involves a topic whose information may change over time.",
                0.90,
            )

        # ====================================================
        # 6. Mathematical / computational questions
        # ====================================================

        if self._matches(
            text,
            self.calculation_patterns,
        ):

            return DecisionResult(
                Decision.LOCAL,
                "Question is computational and does not require current information.",
                0.98,
            )

        # ====================================================
        # 7. Established knowledge
        # ====================================================

        if self._matches(
            text,
            self.local_patterns,
        ):

            return DecisionResult(
                Decision.LOCAL,
                "Question concerns established knowledge suitable for local reasoning.",
                0.90,
            )

        # ====================================================
        # 8. Default
        # ====================================================

        return DecisionResult(
            Decision.LOCAL,
            "No indication that current external information is required.",
            0.70,
        )

    # ============================================================
    # Helper
    # ============================================================

    @staticmethod
    def _matches(
        text,
        patterns,
    ):

        return any(
            re.search(
                pattern,
                text,
            )
            for pattern in patterns
        )

"""AtlasAI Model - Local LLM via Ollama
"""

import logging
import requests
from typing import Generator, Optional
from config import MODEL_NAME, OLLAMA_HOST, TEMPERATURE, MAX_TOKENS

logger = logging.getLogger(__name__)


class AtlasModel:
    """Local model interface via Ollama."""
    
    def __init__(self):
        self.model_name = MODEL_NAME
        self.host = OLLAMA_HOST
        self.temperature = TEMPERATURE
        self.max_tokens = MAX_TOKENS

    def generate(self, prompt: str, stream: bool = True) -> Generator[str, None, None]:
        """Generate response from LLM."""
        
        try:
            url = f"{self.host}/api/generate"
            
            payload = {
                "model": self.model_name,
                "prompt": prompt,
                "temperature": self.temperature,
                "stream": stream,
                "keep_alive": "10m",
                "options": {
                    "num_predict": self.max_tokens,
                },
            }
            
            response = requests.post(
                url,
                json=payload,
                stream=stream,
                timeout=(10, 300),
            )
            response.raise_for_status()
            
            if stream:
                for line in response.iter_lines():
                    if line:
                        try:
                            import json
                            data = json.loads(line)
                            if "response" in data:
                                yield data["response"]
                        except:
                            pass
            else:
                import json
                data = response.json()
                if "response" in data:
                    yield data["response"]
                    
        except requests.exceptions.ConnectionError:
            logger.error(f"Cannot connect to Ollama at {self.host}")
            yield f"Error: Ollama not running at {self.host}"
        except Exception as e:
            logger.error(f"Model error: {e}")
            yield f"Error: {str(e)}"

    def is_available(self) -> bool:
        """Check if model is available."""
        try:
            response = requests.get(
                f"{self.host}/api/tags",
                timeout=5,
            )
            return response.status_code == 200
        except:
            return False

"""
Atlas V3 Planner

Determines whether a request should be handled
by a deterministic plugin or the local model.

Responsibilities:
- Ask IntentRouter for the best plugin
- Extract plugin arguments
- Detect math operations
- Detect math variables
- Produce a deterministic Plan

The Planner NEVER executes plugins.
"""


from dataclasses import dataclass


@dataclass
class Plan:
    """Represents Atlas's deterministic execution plan."""

    action: str

    plugin: object = None

    arguments: str = ""

    operation: str = ""

    variable: str = "x"


class Planner:

    DIRECT = "direct"

    TOOL = "tool"

    def __init__(self, router):

        self.router = router

    # =========================================================
    # PLAN
    # =========================================================

    def plan(
        self,
        message,
    ):
        """
        Build a deterministic execution plan.

        Flow:

            User message
                  ↓
            IntentRouter.match()
                  ↓
               Plugin
                  ↓
          extract_arguments()
                  ↓
                Plan
        """

        message = str(
            message or ""
        ).strip()

        if not message:

            return Plan(
                action=self.DIRECT
            )

        # -----------------------------------------------------
        # Find plugin
        # -----------------------------------------------------

        plugin = self.router.match(
            message
        )

        if plugin is None:

            return Plan(
                action=self.DIRECT
            )

        # -----------------------------------------------------
        # Extract arguments
        # -----------------------------------------------------

        arguments = (
            self.router.extract_arguments(
                plugin,
                message,
            )
        )

        # -----------------------------------------------------
        # Specialized plugin arguments
        # -----------------------------------------------------

        if isinstance(
            arguments,
            dict,
        ):

            raw_arguments = (
                arguments.get(
                    "expression",
                    arguments.get(
                        "message",
                        "",
                    ),
                )
            )

            arguments = str(
                raw_arguments or ""
            ).strip()

        else:

            arguments = str(
                arguments or ""
            ).strip()

        # -----------------------------------------------------
        # Operation
        # -----------------------------------------------------

        operation = ""

        # -----------------------------------------------------
        # Variable
        # -----------------------------------------------------

        variable = "x"

        # =====================================================
        # MATH PLUGIN
        # =====================================================

        if getattr(
            plugin,
            "name",
            "",
        ) == "math":

            # -------------------------------------------------
            # Detect mathematical operation
            # -------------------------------------------------

            if hasattr(
                plugin,
                "detect_operation",
            ):

                try:

                    detected_operation = (
                        plugin.detect_operation(
                            message
                        )
                    )

                    if detected_operation:

                        operation = str(
                            detected_operation
                        ).strip()

                except Exception:
                    operation = ""

            # -------------------------------------------------
            # Detect variable
            # -------------------------------------------------

            if hasattr(
                plugin,
                "detect_variable",
            ):

                try:

                    detected_variable = (
                        plugin.detect_variable(
                            message,
                            arguments,
                        )
                    )

                    if detected_variable:

                        variable = str(
                            detected_variable
                        ).strip()

                except Exception:
                    variable = "x"

        # -----------------------------------------------------
        # Return plan
        # -----------------------------------------------------

        return Plan(
            action=self.TOOL,
            plugin=plugin,
            arguments=arguments,
            operation=operation,
            variable=variable,
        )

"""
Atlas V3 Conversation Service

Persistent conversation management.
"""

import json
import logging
import uuid

from datetime import datetime
from pathlib import Path
from typing import Any, Dict, List, Optional

from config import CONVERSATIONS_DIR


logger = logging.getLogger(__name__)


class ConversationService:
    """
    Manages Atlas conversations.

    The service represents the currently loaded
    conversation.

    Sidebar operations use list(), which returns
    dictionaries.

    Chat/Brain operations use history().
    """

    def __init__(self):

        self.conversations_dir = Path(
            CONVERSATIONS_DIR
        )

        self.conversations_dir.mkdir(
            parents=True,
            exist_ok=True,
        )

        self.current_id: Optional[str] = None

        self.current_messages: List[
            Dict[str, Any]
        ] = []

        self.current_title = "New Chat"

    # ==================================================
    # CREATE
    # ==================================================

    def new(self) -> str:

        conversation_id = str(
            uuid.uuid4()
        )

        self.current_id = conversation_id

        self.current_messages = []

        self.current_title = "New Chat"

        self._save_current()

        logger.info(
            "Created conversation: %s",
            conversation_id,
        )

        return conversation_id

    # ==================================================
    # LOAD
    # ==================================================

    def load(
        self,
        conversation_id: str,
    ) -> bool:

        try:

            file_path = self._get_path(
                conversation_id
            )

            if not file_path.exists():

                logger.warning(
                    "Conversation not found: %s",
                    conversation_id,
                )

                return False

            with file_path.open(
                "r",
                encoding="utf-8",
            ) as file:

                data = json.load(file)

            if not isinstance(
                data,
                dict,
            ):

                return False

            self.current_id = (
                conversation_id
            )

            messages = data.get(
                "messages",
                [],
            )

            if not isinstance(
                messages,
                list,
            ):

                messages = []

            self.current_messages = list(
                messages
            )

            self.current_title = str(
                data.get(
                    "title",
                    self._get_title(),
                )
            )

            logger.info(
                "Loaded conversation: %s",
                conversation_id,
            )

            return True

        except Exception as exc:

            logger.exception(
                "Error loading conversation: %s",
                exc,
            )

            return False

    # ==================================================
    # CURRENT CONVERSATION
    # ==================================================

    def get_id(self) -> Optional[str]:

        return self.current_id

    # --------------------------------------------------

    def get(self):

        return self

    # --------------------------------------------------

    def history(self) -> List[Dict[str, Any]]:

        return list(
            self.current_messages
        )

    # --------------------------------------------------

    def last(self) -> List[Dict[str, Any]]:

        return self.history()

    # ==================================================
    # ADD USER MESSAGE
    # ==================================================

    def add_user(
        self,
        message: str,
    ) -> None:

        message = str(
            message
        ).strip()

        if not message:
            return

        self.current_messages.append(
            {
                "sender": "You",
                "role": "user",
                "text": message,
                "content": message,
                "is_user": True,
                "timestamp": datetime.now().isoformat(),
            }
        )

        if (
            self.current_title == "New Chat"
            or not self.current_title
        ):

            self.current_title = (
                self._generate_title(
                    message
                )
            )

    # ==================================================
    # ADD ASSISTANT MESSAGE
    # ==================================================

    def add_assistant(
        self,
        message: str,
    ) -> None:

        message = str(
            message
        ).strip()

        if not message:
            return

        self.current_messages.append(
            {
                "sender": "Atlas",
                "role": "assistant",
                "text": message,
                "content": message,
                "is_user": False,
                "timestamp": datetime.now().isoformat(),
            }
        )

    # ==================================================
    # SAVE
    # ==================================================

    def save(self) -> bool:

        return self._save_current()

    # --------------------------------------------------

    def _save_current(self) -> bool:

        try:

            if not self.current_id:

                return False

            file_path = self._get_path(
                self.current_id
            )

            created = None

            if file_path.exists():

                try:

                    with file_path.open(
                        "r",
                        encoding="utf-8",
                    ) as file:

                        old_data = json.load(
                            file
                        )

                    created = old_data.get(
                        "created"
                    )

                except Exception:

                    created = None

            data = {
                "id": self.current_id,
                "title": (
                    self.current_title
                    or self._get_title()
                ),
                "created": (
                    created
                    or datetime.now().isoformat()
                ),
                "messages": self.current_messages,
            }

            with file_path.open(
                "w",
                encoding="utf-8",
            ) as file:

                json.dump(
                    data,
                    file,
                    indent=2,
                    ensure_ascii=False,
                )

            return True

        except Exception as exc:

            logger.exception(
                "Error saving conversation: %s",
                exc,
            )

            return False

    # ==================================================
    # TITLE
    # ==================================================

    def _generate_title(
        self,
        message: str,
    ) -> str:

        title = " ".join(
            str(message)
            .strip()
            .split()
        )

        if len(title) > 45:

            title = (
                title[:45].rstrip()
                + "..."
            )

        return title or "New Chat"

    # --------------------------------------------------

    def _get_title(self) -> str:

        if self.current_title:

            return self.current_title

        if not self.current_messages:

            return "New Chat"

        first = self.current_messages[0]

        if not isinstance(
            first,
            dict,
        ):

            return "New Chat"

        text = (
            first.get("text")
            or first.get("content")
            or ""
        )

        return self._generate_title(
            text
        )

    # --------------------------------------------------

    def auto_title(
        self,
        message: str,
    ) -> str:

        if (
            not self.current_title
            or self.current_title == "New Chat"
        ):

            self.current_title = (
                self._generate_title(
                    message
                )
            )

            self.save()

        return self.current_title

    # ==================================================

class MemoryService:
    """Manages user memory."""
    
    def __init__(self):
        self.memory_file = MEMORY_FILE
        self.data: Dict[str, Any] = self._load()

    def _load(self) -> Dict[str, Any]:
        """Load memory from disk."""
        try:
            if self.memory_file.exists():
                with self.memory_file.open('r', encoding='utf-8') as f:
                    return json.load(f)
        except Exception as e:
            logger.error(f"Error loading memory: {e}")
        
        return {'user': {}, 'facts': {}}

    def _save(self) -> None:
        """Save memory to disk."""
        try:
            self.memory_file.parent.mkdir(parents=True, exist_ok=True)
            
            with self.memory_file.open('w', encoding='utf-8') as f:
                json.dump(self.data, f, indent=2, ensure_ascii=False)
        
        except Exception as e:
            logger.error(f"Error saving memory: {e}")

    def extract(self, message: str) -> None:
        """Extract facts from message."""
        # Simple pattern matching for common facts
        message_lower = message.lower()
        
        if "my name is" in message_lower or "i'm" in message_lower or "i am" in message_lower:
            # Try to extract name
            words = message.split()
            for i, word in enumerate(words):
                if word.lower() in ["am", "is", "name"]:
                    if i + 1 < len(words):
                        self.data['user']['name'] = words[i + 1]
                        self._save()

        """Try to recall relevant memory."""
        
        if "who am i" in message.lower():
            name = self.data.get('user', {}).get('name')

        if "what is my name" in message.lower():

            for char in text
        )

    @staticmethod
    def contains_hinglish_words(text: str) -> bool:
        """Check if text contains common Hinglish words."""
        text_lower = text.lower()
        # Split into words and check if any are Hinglish indicators
        words = re.findall(r'\b\w+\b', text_lower)
        return any(word in LanguageTranslator.HINGLISH_INDICATORS for word in words)

    @staticmethod
    def is_hinglish(text: str) -> bool:
        """
        Detect if text is Hinglish (mix of English and Hindi).
        Checks for:
        1. Devanagari script
        2. Common Hinglish words
        """
        has_devanagari = LanguageTranslator.contains_devanagari(text)
        has_hinglish_words = LanguageTranslator.contains_hinglish_words(text)
        
        return has_devanagari or has_hinglish_words

    @staticmethod
    def transliterate_devanagari(text: str) -> str:
        """
        Basic Devanagari to Latin transliteration.
        Handles pure Hindi text conversion.
        """
        result = text
        for devanagari, latin in LanguageTranslator.DEVANAGARI_TO_LATIN.items():
            result = result.replace(devanagari, latin)
        return result

    @staticmethod
    def apply_hinglish_rules(text: str) -> str:
        """
        Apply common Hinglish-to-English translation rules.
        Case-insensitive pattern matching.
        """
        result = text
        
        # Apply transliteration patterns (case-insensitive)
        for pattern, replacement in LanguageTranslator.HINGLISH_TO_ENGLISH.items():
            compiled = re.compile(pattern, re.IGNORECASE)
            result = compiled.sub(replacement, result)
        
        return result

    @staticmethod
    def normalize_hinglish(text: str) -> str:
        """
        Normalize Hinglish text by:
        1. Transliterating Devanagari to Latin
        2. Applying Hinglish translation rules
        3. Cleaning up extra spaces
        """
        # Step 1: Transliterate Devanagari if present
        if LanguageTranslator.contains_devanagari(text):
            text = LanguageTranslator.transliterate_devanagari(text)
        
        # Step 2: Apply Hinglish rules
        text = LanguageTranslator.apply_hinglish_rules(text)
        
        # Step 3: Clean up extra spaces
        text = re.sub(r'\s+', ' ', text).strip()
        
        return text

    @staticmethod
    def detect_and_convert(text: str) -> Tuple[str, bool, str]:
        """
        Detect language and convert if needed.
        
        Returns:
            Tuple of (converted_text, was_hinglish, original_language)
        """
        is_hinglish = LanguageTranslator.is_hinglish(text)
        
        if is_hinglish:
            converted = LanguageTranslator.normalize_hinglish(text)
            return converted, True, "hinglish"
        
        return text, False, "english"

    @staticmethod
    def add_hinglish_context(message: str) -> str:
        """
        Add Hinglish to the original message if detected.
        Helps preserve intent in the context.
        """
        if LanguageTranslator.is_hinglish(message):
            return f"[User is speaking in Hinglish/Hindi]\n{message}"
        return message

"""
Atlas V3 Event Bus

The Event Bus allows independent components
to communicate without directly referencing
each other.
"""

from collections import defaultdict
from typing import Callable, Type

from core.events import Event


class EventBus:
    """
    Publish/Subscribe event bus.
    """

    def __init__(self):

        self._listeners = defaultdict(list)

    # ------------------------------------------------------
    # Subscribe
    # ------------------------------------------------------

    def subscribe(
        self,
        event_type: Type[Event],
        callback: Callable,
    ):

        if callback not in self._listeners[event_type]:

            self._listeners[event_type].append(
                callback
            )

    # ------------------------------------------------------
    # Unsubscribe
    # ------------------------------------------------------

    def unsubscribe(
        self,
        event_type: Type[Event],
        callback: Callable,
    ):

        if callback in self._listeners[event_type]:

            self._listeners[event_type].remove(
                callback
            )

    # ------------------------------------------------------
    # Publish
    # ------------------------------------------------------

    def publish(self, event: Event):

        for callback in self._listeners[type(event)]:

            callback(event)

    def _conversation_metadata(
        self,
        conversation_id,
    ):
        """
        Return metadata for a conversation.
        """

        if not conversation_id:
            return None

        conversations = (
            self.conversations.list_all()
        )

        if not isinstance(
            conversations,
            dict,
        ):
            return None

        return conversations.get(
            conversation_id
        )

    # ------------------------------------------------------

    def _get_latest_conversation_id(
        self,
    ):
        """
        Find the newest conversation safely.

        Handles missing/None timestamps without
        allowing max() to compare None and strings.
        """

        conversations = (
            self.conversations.list_all()
        )

        if not conversations:
            return None

        if not isinstance(
            conversations,
            dict,
        ):
            return None

        best_id = None
        best_time = ""

        for conversation_id, metadata in conversations.items():

            if not isinstance(
                metadata,
                dict,
            ):
                metadata = {}

            modified = (
                metadata.get("modified")
                or metadata.get("created")
                or ""
            )

            # Always convert to string so comparison
            # between None and str cannot happen.
            modified = str(
                modified
            )

            if (
                best_id is None
                or modified > best_time
            ):
                best_id = conversation_id
                best_time = modified

        return best_id

    # ENSURE ACTIVE CONVERSATION
    # ======================================================

    def _ensure_active_conversation(
        self,
    ):
        """
        Ensure there is always an active conversation.
        """

        active_id = (
            self.state.active_conversation_id
        )

        # --------------------------------------------------
        # Existing active conversation still exists
        # --------------------------------------------------

        if active_id:

            manager = (
                self.conversations.load(
                    active_id
                )
            )

            if manager is not None:

                logger.info(
                    "Using active conversation: %s",
                    active_id,
                )

                return active_id

        # --------------------------------------------------
        # Try newest existing conversation
        # --------------------------------------------------

        latest_id = (
            self._get_latest_conversation_id()
        )

        if latest_id:

            manager = (
                self.conversations.load(
                    latest_id
                )
            )

            if manager is not None:

                self.state.active_conversation_id = (
                    latest_id
                )

                logger.info(
                    "Restored latest conversation: %s",
                    latest_id,
                )

                return latest_id

        # --------------------------------------------------
        # Nothing exists -> create one
        # --------------------------------------------------

        return self.create_conversation(
            synchronize=False
        )

    # ======================================================


    def create_conversation(
        self,
        synchronize=True,
    ):
        """
        Create a new conversation.

        An untouched active "New Chat" is reused so repeated clicks on
        New Chat cannot create a pile of empty conversations.
        """
        import uuid

        active_id = self.get_active_conversation_id()
        if active_id:
            active = self.get_active_conversation()
            if (
                active is not None
                and not active.get_messages()
                and str(getattr(active, "title", "")) == "New Chat"
            ):
                if synchronize:
                    self._sync_brain()
                return active_id

        conversation_id = str(uuid.uuid4())
        manager = ConversationManager(
            conversation_id=conversation_id,
            title="New Chat",
        )

        if not self.conversations.save(manager):
            logger.error(
                "Failed to create conversation: %s",
                conversation_id,
            )
            return None

        self.state.active_conversation_id = conversation_id
        self.state.current_message = ""
        self.state.current_response = ""
        self.state.status = "Idle"
        self.state.is_generating = False

        if hasattr(self.state, "stop_event"):
            self.state.stop_event.clear()

        if synchronize:
            self._sync_brain()

        logger.info("Created conversation: %s", conversation_id)
        return conversation_id

    def get_conversations(
        self,
    ):
        """
        Return conversations in the format expected
        by the sidebar.

        The sidebar supports a list of dictionaries,
        so return that format.
        """

        conversations = (
            self.conversations.list_all()
        )

        result = []

        if not isinstance(
            conversations,
            dict,
        ):
            return result

        for conversation_id, metadata in conversations.items():

            if not isinstance(
                metadata,
                dict,
            ):
                metadata = {}

            result.append(
                {
                    "id": conversation_id,
                    "title": (
                        metadata.get("title")
                        or "New Chat"
                    ),
                    "created": (
                        metadata.get("created")
                    ),
                    "modified": (
                        metadata.get("modified")
                    ),
                    "message_count": (
                        metadata.get(
                            "message_count",
                            0,
                        )
                    ),
                }
            )

        # --------------------------------------------------
        # Newest conversations first
        # --------------------------------------------------

        result.sort(
            key=lambda item: str(
                item.get("modified")
                or item.get("created")
                or ""
            ),
            reverse=True,
        )

        return result

    def delete_conversation(
        self,
        conversation_id,
    ):
        """Delete a conversation without spawning duplicate empty chats."""
        if not conversation_id:
            return False

        if self.state.is_generating:
            logger.warning(
                "Refusing to delete while Atlas is generating."
            )
            return False

        active_id = self.get_active_conversation_id()
        was_active = str(active_id) == str(conversation_id)

        if not self.conversations.delete(conversation_id):
            return False

        if was_active:
            remaining = self.conversations.list_all()
            candidates = []

            if isinstance(remaining, dict):
                for cid, meta in remaining.items():
                    if not isinstance(meta, dict):
                        meta = {}
                    modified = str(
                        meta.get("modified")
                        or meta.get("created")
                        or ""
                    )
                    candidates.append((modified, str(cid)))

            candidates.sort(reverse=True)

            if candidates:
                self.state.active_conversation_id = candidates[0][1]
                self._sync_brain()
            else:
                self.state.active_conversation_id = None
                self.create_conversation(synchronize=True)
        else:
            self._sync_brain()

        return True

    @staticmethod
    def _profile_setup_tool(message):
        text = str(message or "").lower().strip()

        nutrition_signals = (
            "nutrition", "meal plan", "mealplan", "diet", "food",
            "allerg", "intoleran", "what i eat", "eating",
            "i'm fat", "im fat", "i am fat",
            "i'm skinny", "im skinny", "i am skinny",
        )
        fitness_signals = (
            "gym", "workout", "exercise", "fitness",
            "training", "reps", "sets",
        )

        if any(signal in text for signal in nutrition_signals):
            return "nutrition"
        if any(signal in text for signal in fitness_signals):
            return "fitness"
        return None

    def _profile_needs_setup(self, tool):
        plugin = self.ai.tools.get(tool) if self.ai.tools is not None else None
        if plugin is None:
            return False
        checker = getattr(plugin, "profile_complete", None)
        if callable(checker):
            try:
                return not bool(checker())
            except Exception:
                return True
        return False

    def _handle_profile_setup_request(self, text):
        tool = self._profile_setup_tool(text)
        if tool is None or not self._profile_needs_setup(tool):
            return False

        if tool == "fitness":
            prompt = (
                "Let’s set up your fitness profile first. "
                "I’ll use your experience, goal, equipment and restrictions "
                "to tailor future suggestions."
            )
        else:
            prompt = (
                "Let’s set up a nutrition profile first. "
                "I’ll use age, activity, dietary preferences and allergies "
                "to tailor meal ideas."
            )

        from core.events import ProfileSetupRequested
        self.bus.publish(ProfileSetupRequested(tool=tool, prompt=prompt))
        return True

    def apply_tool_profile(self, tool, profile):
        """Save a GUI profile through the real plugin backend."""
        tool = str(tool or "").strip().lower()
        profile = dict(profile or {})
        plugin = self.ai.tools.get(tool)
        if plugin is None:
            raise ValueError(f"Tool '{tool}' is unavailable.")
        if tool not in ("fitness", "nutrition"):
            raise ValueError(f"Interactive profiles are not supported for '{tool}'.")
        result = plugin.set_profile(**profile)
        from core.events import ProfileSetupCompleted
        self.bus.publish(ProfileSetupCompleted(tool=tool, profile=profile, message=str(result or "")))
        return result

    def run_tool_action(self, tool, action, **kwargs):
        """Execute a concrete fitness/nutrition action from the GUI."""
        tool = str(tool or "").strip().lower()
        plugin = self.ai.tools.get(tool)
        if plugin is None:
            raise ValueError(f"Tool '{tool}' is unavailable.")
        method = getattr(plugin, str(action), None)
        if not callable(method):
            raise ValueError(f"Tool '{tool}' has no action '{action}'.")
        return method(**kwargs)

    # ======================================================
    # AUTO TITLE
    # ======================================================

    @staticmethod
    def _auto_title(text):
        """Create a compact, deterministic title from the first user message."""
        text = " ".join(str(text or "").strip().split())
        if not text:
            return "New Chat"
        text = text.rstrip("?!.,;:")
        replacements = (
            ("please tell me ", ""),
            ("can you ", ""),
            ("could you ", ""),
            ("help me ", ""),
        )
        low = text.lower()
        for prefix, replacement in replacements:
            if low.startswith(prefix):
                text = text[len(prefix):]
                break
        words = text.split()[:7]
        title = " ".join(words).strip()
        if len(title) > 42:
            title = title[:42].rsplit(" ", 1)[0]
        if not title:
            return "New Chat"
        return title[0].upper() + title[1:]

    def _auto_rename_if_needed(self, conversation, first_message):
        if conversation is None:
            return False
        current = str(getattr(conversation, "title", "") or "").strip()
        if current.lower() != "new chat":
            return False
        title = self._auto_title(first_message)
        if title == "New Chat":
            return False
        conversation.title = title
        return bool(self.conversations.save(conversation))

    # ======================================================
    # MESSAGE ROUTING
    # ======================================================

    def _handle_message_received(
        self,
        event,
    ):
        """
        Receive a user message from the GUI,
        save it to the active conversation,
        then send it to AIService.
        """

        # Event classes in Atlas V3 currently carry
        # the message as .text.
        text = getattr(
            event,
            "text",
            None,
        )

        if text is None:

            # Compatibility with possible alternate
            # event attribute names.
            text = getattr(
                event,
                "message",
                None,
            )

        if text is None:
            return

        text = str(
            text
        ).strip()

        if not text:
            return

        profile_tool = self._profile_setup_tool(text)

        # --------------------------------------------------
        # Make sure an active conversation exists
        # --------------------------------------------------

        if not self.get_active_conversation_id():

            self.create_conversation()

        conversation = (
            self.get_active_conversation()
        )

        if conversation is None:

            logger.error(
                "Could not get active conversation."
            )

            self.bus.publish(
                ErrorOccurred(
                    "Could not access the active conversation."
                )
            )

            return

        # --------------------------------------------------
        # Save user message
        # --------------------------------------------------

        conversation.add_message(
            sender="User",
            text=text,
            is_user=True,
        )

        self.conversations.save(
            conversation
        )

        # Give the chat a useful name immediately after its first message.
        self._auto_rename_if_needed(conversation, text)

        if profile_tool is not None:
            self.state.current_message = text
            self.state.current_response = ""
            self.state.is_generating = False
            self.state.status = "Idle"
            self._sync_brain()

            if self._handle_profile_setup_request(text):
                return

        # --------------------------------------------------
        # Update state
        # --------------------------------------------------

        self.state.current_message = text
        self.state.current_response = ""

        self.state.is_generating = True
        self.state.status = "Thinking"

        if hasattr(
            self.state,
            "stop_event",
        ):

            self.state.stop_event.clear()

        # --------------------------------------------------
        # Tell GUI/backend that thinking started
        # --------------------------------------------------

        self.bus.publish(
            ThinkingStarted()
        )

        logger.info(
            "Sending message to AI: %s",
            text,
        )

        # --------------------------------------------------
        # IMPORTANT:
        # Send to AIService, not directly to AIWorker.
        #
        # AIService owns the QThread.
        # --------------------------------------------------

        self.ai.generate(
            text
        )

    # ======================================================
    # STOP GENERATION
    # ======================================================

    def stop_generation(self):
        """Cooperatively stop the current response without killing Atlas."""
        if not getattr(self.state, "is_generating", False):
            return False

        try:
            if hasattr(self.state, "stop_event"):
                self.state.stop_event.set()
            self.ai.cancel_generation()
            self.state.status = "Stopping"
            logger.info("AI generation stop requested.")
            return True
        except Exception as exc:
            logger.warning("Could not stop AI generation: %s", exc)
            return False

    # ======================================================
    # AI CHUNK
    # ======================================================

    def _handle_ai_chunk(
        self,
        text,
    ):
        """
        Receive AI output from AIService.
        """

        if text is None:
            return

        text = str(
            text
        )

        if not text:
            return

        self.state.current_response += (
            text
        )

        self.state.status = "Generating"

        # --------------------------------------------------
        # Send response to GUI
        # --------------------------------------------------

        self.bus.publish(
            ChunkReceived(
                text
            )
        )

    # ======================================================
    # AI FINISHED
    # ======================================================

    def _handle_ai_finished(
        self,
    ):
        """
        Save the completed AI response.
        """

        response = str(
            self.state.current_response
            or ""
        ).strip()

        conversation = (
            self.get_active_conversation()
        )

        if conversation is not None and response:

            conversation.add_message(
                sender="Atlas",
                text=response,
                is_user=False,
            )

            self.conversations.save(
                conversation
            )

        self.state.is_generating = False
        self.state.status = "Idle"

        self.bus.publish(
            ThinkingFinished()
        )

        logger.info(
            "AI response finished."

        )

    def _handle_ai_error(

        """

        """

        error = str(
            error
        )

        self.state.is_generating = False
        self.state.status = "Error"

        logger.error(

            error,
        )

        self.bus.publish(

                error
            )
        )

        self.bus.publish(
            ThinkingFinished()
        )

    # ======================================================
    # BRAIN SYNCHRONIZATION
    # ======================================================

    def _sync_brain(
        self,
    ):
        """
        Synchronize AtlasBrain's conversation history
        with the active conversation.
        """

        conversation = (
            self.get_active_conversation()
        )

        if conversation is None:

            logger.warning(
                "Cannot synchronize brain: "
                "no active conversation."
            )

            return

        messages = (
            conversation.get_messages()
        )

        # --------------------------------------------------
        # AIService -> AIWorker -> AtlasBrain
        # --------------------------------------------------

        brain = getattr(
            self.ai.worker,
            "brain",
            None,
        )

        if brain is None:

            logger.warning(
                "Cannot synchronize brain: "
                "AIWorker brain unavailable."
            )

            return

        # --------------------------------------------------
        # Brain API used by Atlas V3
        # --------------------------------------------------

        if hasattr(
            brain,
            "set_conversation_history",
        ):

            try:

                brain.set_conversation_history(
                    messages
                )

            except Exception:

                logger.exception(
                    "Failed to synchronize "
                    "conversation history with brain."
                )

                return

        elif hasattr(
            brain,
            "conversation_history",
        ):

            brain.conversation_history = (
                list(messages)
            )

        logger.info(
            "Brain synchronized with conversation: %s",
            conversation.id,
        )

    # ======================================================
    # START
    # ======================================================

    def start(

        """

        if self.ai is not None:

                self.ai.stop()

            except Exception as exc:

                    exc,
                )

        logger.info(
            "Controller stopped."
        )

"""
Conversation Management (FIXED)
"""

import json
import logging
from pathlib import Path
from datetime import datetime
from typing import Optional, List, Dict, Any

from config import CONVERSATIONS_DIR

logger = logging.getLogger(__name__)


class ConversationManager:
    """Manages a single conversation."""
    
    def __init__(self, conversation_id: str, title: str = "New Chat"):
        self.id = conversation_id
        self.title = title
        self.created = datetime.now().isoformat()
        self.modified = datetime.now().isoformat()
        self.messages: List[Dict[str, Any]] = []

    def add_message(self, sender: str, text: str, is_user: bool = False) -> None:
        """Add message to conversation."""
        message = {
            'sender': sender,
            'text': text,
            'is_user': is_user,
            'timestamp': datetime.now().isoformat(),
        }
        self.messages.append(message)
        self.modified = datetime.now().isoformat()

    def get_messages(self) -> List[Dict[str, Any]]:
        """Get all messages."""
        return self.messages.copy()

    def clear_messages(self) -> None:
        """Clear all messages."""
        self.messages = []
        self.modified = datetime.now().isoformat()

    def to_dict(self) -> Dict[str, Any]:
        """Convert to dictionary."""
        return {
            'id': self.id,
            'title': self.title,
            'created': self.created,
            'modified': self.modified,
            'messages': self.messages,
        }

    @staticmethod
    def from_dict(data: Dict[str, Any]) -> 'ConversationManager':
        """Create from dictionary."""
        conv = ConversationManager(
            data.get('id', ''),
            data.get('title', 'New Chat')
        )
        conv.created = data.get('created', datetime.now().isoformat())
        conv.modified = data.get('modified', datetime.now().isoformat())
        conv.messages = data.get('messages', [])
        return conv


class ConversationStore:
    """Manages all conversations on disk."""
    
    def __init__(self):
        self.conversations_dir = CONVERSATIONS_DIR
        self.conversations_dir.mkdir(parents=True, exist_ok=True)

    def _get_conversation_file(self, conversation_id: str) -> Path:
        """Get file path for conversation."""
        return self.conversations_dir / f"{conversation_id}.json"

    def save(self, manager: ConversationManager) -> bool:
        """Save conversation to disk."""
        try:
            file_path = self._get_conversation_file(manager.id)
            with file_path.open('w', encoding='utf-8') as f:
                json.dump(manager.to_dict(), f, indent=2, ensure_ascii=False)
            logger.info(f"Saved conversation: {manager.id}")
            return True
        except Exception as e:
            logger.error(f"Error saving conversation: {e}")
            return False

    def load(self, conversation_id: str) -> Optional[ConversationManager]:
        """Load conversation from disk."""
        try:
            file_path = self._get_conversation_file(conversation_id)
            if not file_path.exists():
                logger.warning(f"Conversation file not found: {file_path}")
                return None
            
            with file_path.open('r', encoding='utf-8') as f:
                data = json.load(f)
            
            manager = ConversationManager.from_dict(data)
            logger.info(f"Loaded conversation: {conversation_id}")
            return manager
        except Exception as e:
            logger.error(f"Error loading conversation: {e}")
            return None

    def delete(self, conversation_id: str) -> bool:
        """Delete conversation."""
        try:
            file_path = self._get_conversation_file(conversation_id)
            if file_path.exists():
                file_path.unlink()
                logger.info(f"Deleted conversation: {conversation_id}")
                return True
            return False
        except Exception as e:
            logger.error(f"Error deleting conversation: {e}")
            return False

    def list_all(self) -> Dict[str, Dict]:
        """List all conversations with metadata."""

"""
Atlas V3 Intent Router

Deterministically selects the best plugin for a message.

Responsibilities:
- Select the best enabled plugin
- Score plugin matches
- Extract clean plugin arguments
- Never execute plugins
"""

import re


class IntentRouter:

    def __init__(self, plugins):
        self.plugins = plugins

    # =========================================================
    # MATCH
    # =========================================================

    def match(self, message):

        text = (
            str(message or "")
            .lower()
            .strip()
        )

        if not text:
            return None

        candidates = []

        for plugin in self.plugins.values():

            if not getattr(
                plugin,
                "enabled",
                True,
            ):
                continue

            try:
                if not plugin.can_handle(message):
                    continue

            except Exception:
                continue

            score = self._score(
                plugin,
                text,
            )

            if score <= 0:
                continue

            priority = getattr(
                plugin,
                "priority",
                100,
            )

            try:
                priority = int(priority)

            except (
                TypeError,
                ValueError,
            ):
                priority = 100

            candidates.append(
                (
                    score,
                    priority,
                    str(
                        getattr(
                            plugin,
                            "name",
                            "",
                        )
                    ).lower(),
                    plugin,
                )
            )

        if not candidates:
            return None

        candidates.sort(
            key=lambda item: (
                -item[0],
                item[1],
                item[2],
            )
        )

        return candidates[0][3]

    # =========================================================
    # SCORE
    # =========================================================

    def _score(
        self,
        plugin,
        text,
    ):

        best = 0

        triggers = getattr(
            plugin,
            "triggers",
            [],
        ) or []

        for trigger in triggers:

            trigger = (
                str(trigger)
                .lower()
                .strip()
            )

            if not trigger:
                continue

            # Exact trigger
            if text == trigger:

                best = max(
                    best,
                    1000 + len(trigger),
                )

                continue

            # Trigger followed by arguments
            if text.startswith(
                trigger + " "
            ):

                best = max(
                    best,
                    700 + len(trigger),
                )

                continue

            # Complete phrase inside message
            padded = f" {text} "

            if f" {trigger} " in padded:

                best = max(
                    best,
                    300 + len(trigger),
                )

        return best

    # =========================================================
    # ARGUMENT EXTRACTION
    # =========================================================

    def extract_arguments(
        self,
        plugin,
        message,
    ):
        """
        Extract arguments for a plugin.

        Examples:

            "What time is it?"
                -> ""

            "calculate 25 * 37"
                -> "25 * 37"

            "calc 100 / 4"
                -> "100 / 4"

            "time"
                -> ""

        Plugin-specific extractors are supported, but generic
        AtlasPlugin extraction must not override deterministic
        argument cleanup.
        """

        original = str(
            message or ""
        ).strip()

        if not original:
            return ""

        # -----------------------------------------------------
        # Normalize punctuation for matching.
        #
        # Keep the original casing for returned arguments.
        # -----------------------------------------------------

        text = original.rstrip(
            "?!.,;:"
        ).strip()

        lowered = text.lower()

        # -----------------------------------------------------
        # Plugin name
        # -----------------------------------------------------

        plugin_name = str(
            getattr(
                plugin,
                "name",
                "",
            )
        ).lower()

        # =====================================================
        # TIME / DATE
        # =====================================================

        if plugin_name == "time":

            return ""

        # =====================================================
        # CALCULATOR
        # =====================================================

        if plugin_name == "calculator":

            return self._extract_calculator_expression(
                text
            )

        # =====================================================
        # PLUGIN-SPECIFIC EXTRACTOR
        # =====================================================

        extractor = getattr(
            plugin,
            "extract_arguments",
            None,
        )

        if callable(extractor):

            try:

                result = extractor(
                    original
                )

                # ---------------------------------------------
                # Structured plugin arguments
                # ---------------------------------------------

                if isinstance(
                    result,
                    dict,
                ):

                    # A generic {"message": original_message}
                    # is NOT useful as extracted arguments.
                    #
                    # Only return meaningful structured data.
                    if (
                        "message" in result
                        and len(result) == 1
                    ):

                        extracted_message = str(
                            result.get(
                                "message",
                                "",
                            )
                        ).strip()

                        # If the plugin simply echoed the
                        # original request, continue with
                        # generic extraction instead.
                        if (
                            extracted_message.lower()
                            == lowered
                        ):
                            pass

                        else:
                            return result

                    else:

                        return result

                # ---------------------------------------------
                # Normal string result
                # ---------------------------------------------

                elif result is not None:

                    result = str(
                        result
                    ).strip()

                    if result:
                        return result

            except Exception:
                pass

        # =====================================================
        # GENERIC TRIGGER EXTRACTION
        # =====================================================

        triggers = getattr(
            plugin,
            "triggers",
            [],
        ) or []

        normalized_triggers = []

        for trigger in triggers:

            trigger = (
                str(trigger)
                .strip()
            )

            if trigger:
                normalized_triggers.append(
                    trigger
                )

        # Longest trigger first so that:

        # "what is the time"

        # wins over:

        # "what"

        normalized_triggers.sort(
            key=len,
            reverse=True,
        )

        # -----------------------------------------------------
        # Exact trigger
        # -----------------------------------------------------

        for trigger in normalized_triggers:

            if (
                lowered
                == trigger.lower()
            ):

                return ""

        # -----------------------------------------------------
        # Trigger + arguments
        # -----------------------------------------------------

        for trigger in normalized_triggers:

            trigger_lower = (
                trigger.lower()
            )

            if lowered.startswith(
                trigger_lower + " "
            ):

                return text[
                    len(trigger):
                ].strip()

        # -----------------------------------------------------
        # Default
        # -----------------------------------------------------

        return text

    # =========================================================
    # CALCULATOR EXTRACTION
    # =========================================================

    def _extract_calculator_expression(
        self,
        text,
    ):
        """
        Convert natural-language calculator commands
        into a clean arithmetic expression.
        """

        text = str(
            text or ""
        ).strip()

        if not text:
            return ""

        lowered = text.lower()

        prefixes = (
            "calculate",
            "calc",
            "compute",
            "evaluate",
            "what is",
            "what's",
        )

        # -----------------------------------------------------
        # Explicit calculator command
        # -----------------------------------------------------

        for prefix in prefixes:

            if lowered == prefix:

                return ""

            if lowered.startswith(
                prefix + " "
            ):

                return text[
                    len(prefix):
                ].strip()

        # -----------------------------------------------------
        # Already a raw arithmetic expression
        # -----------------------------------------------------

        if re.fullmatch(
            r"[0-9eEpiPI+\-*/%^().\s]+",
            text,
        ):

            return text

        # -----------------------------------------------------
        # If calculator somehow matched a natural-language
        # expression without one of the known prefixes,
        # return the text rather than destroying information.
        # -----------------------------------------------------

        return text

    "%(levelname)s | "
    "%(name)s | "

)


def get_logger(name="atlas"):

    logger = logging.getLogger(name)

    # Prevent duplicate handlers if this module
    # is imported multiple times.
    if logger.handlers:
        return logger

    logger.setLevel(logging.INFO)

    logger.propagate = False

    formatter = logging.Formatter(
        LOG_FORMAT
    )

    # --------------------------------------------------
    # File Handler
    # --------------------------------------------------

    file_handler = RotatingFileHandler(
        LOG_FILE,
        maxBytes=5 * 1024 * 1024,
        backupCount=3,
        encoding="utf-8",
    )

    file_handler.setFormatter(
        formatter
    )

    logger.addHandler(
        file_handler
    )

    # --------------------------------------------------
    # Console Handler
    # --------------------------------------------------

    console_handler = logging.StreamHandler()

    console_handler.setFormatter(
        formatter
    )

    logger.addHandler(
        console_handler
    )

    return logger

class MemoryManager:

    def __init__(self):

        self.file.parent.mkdir(
            parents=True,
            exist_ok=True,
        )

        if not self.file.exists():
            self._save({})

        else:
            try:
                data = self._load()

                if not isinstance(data, dict):
                    self._save({})

            except (json.JSONDecodeError, OSError):
                self._save({})

    # --------------------------------------------------
    # Internal
    # --------------------------------------------------

    def _load(self):
        try:
            with self.file.open(
                "r",
                encoding="utf-8",
            ) as f:
                data = json.load(f)

            return data if isinstance(data, dict) else {}

        except (
            FileNotFoundError,
            json.JSONDecodeError,
            OSError,
        ):
            return {}

    def _save(self, data):
        temporary = self.file.with_suffix(".tmp")

        with temporary.open(
            "w",
            encoding="utf-8",
        ) as f:
            json.dump(
                data,
                f,
                indent=4,
                ensure_ascii=False,
            )

        temporary.replace(self.file)

    # Public API
    # --------------------------------------------------

    def remember(
        self,
        category,
        key,
        value,
    ):
        data = self._load()

        if category not in data:
            data[category] = {}

        data[category][key] = value

        self._save(data)

"""
Atlas V3 Plugin Manager

Discovers, loads and manages Atlas plugins.

The plugin manager is deliberately independent from
AtlasBrain and ToolManager.
"""

import importlib
import inspect
import logging
import pkgutil

from plugins.base import AtlasPlugin


logger = logging.getLogger("atlas")


class PluginManager:
    """Central plugin discovery and registry."""

    PACKAGES = (
        "plugins.system",
        "plugins.math_engine",
    )

    def __init__(self):
        self.plugins = {}
        self.load_plugins()

    # =========================================================
    # LOAD
    # =========================================================

    def load_plugins(self):
        """Load every plugin from configured plugin packages."""

        self.plugins.clear()

        for package_name in self.PACKAGES:
            self._load_package(package_name)

        logger.info(
            "Plugin system loaded: %s",
            self.available(),
        )

    # =========================================================
    # PACKAGE DISCOVERY
    # =========================================================

    def _load_package(self, package_name):
        try:
            package = importlib.import_module(package_name)

        except Exception as exc:
            logger.exception(
                "Failed to import plugin package '%s': %s",
                package_name,
                exc,
            )
            return

        package_path = getattr(
            package,
            "__path__",
            None,
        )

        if package_path is None:
            return

        for _, module_name, is_package in pkgutil.iter_modules(
            package_path
        ):

            if is_package:
                continue

            module_name = str(module_name)

            # Never load Python cache/helper modules.
            if module_name.startswith("_"):
                continue

            full_name = (
                f"{package_name}.{module_name}"
            )

            try:
                module = importlib.import_module(
                    full_name
                )

            except Exception as exc:
                # One broken plugin must not prevent every
                # other plugin from loading.
                logger.exception(
                    "Failed to load plugin module '%s': %s",
                    full_name,
                    exc,
                )
                continue

            self._load_module(module)

    # =========================================================
    # MODULE
    # =========================================================

    def _load_module(self, module):
        """Find AtlasPlugin subclasses defined in a module."""

        for _, obj in inspect.getmembers(
            module,
            inspect.isclass,
        ):

            try:
                is_plugin = issubclass(
                    obj,
                    AtlasPlugin,
                )

            except TypeError:
                continue

            if not is_plugin:
                continue

            if obj is AtlasPlugin:
                continue

            # Only classes actually defined inside this module.
            if obj.__module__ != module.__name__:
                continue

            try:
                plugin = obj()

            except Exception as exc:
                logger.exception(
                    "Failed to instantiate plugin '%s': %s",
                    obj.__name__,
                    exc,
                )
                continue

            name = getattr(
                plugin,
                "name",
                None,
            )

            if not name:
                logger.warning(
                    "Skipping plugin '%s': missing name",
                    obj.__name__,
                )
                continue

            # Canonical plugin IDs are lowercase.
            name = str(name).strip().lower()

            if not name:
                continue

            # Keep plugin object's name canonical too.
            try:
                plugin.name = name
            except Exception:
                pass

            if name in self.plugins:
                logger.warning(
                    "Skipping duplicate plugin '%s' from %s",
                    name,
                    module.__name__,
                )
                continue

            self.plugins[name] = plugin

            logger.info(
                "Loaded plugin: %s",
                name,
            )

    # =========================================================
    # EXECUTION
    # =========================================================

    def execute(self, name, **kwargs):
        name = str(name).strip().lower()

        plugin = self.get(name)

        if plugin is None:
            raise ValueError(
                f"Unknown tool: {name}"
            )

        if not getattr(
            plugin,
            "enabled",
            True,
        ):
            raise RuntimeError(
                f"Tool is disabled: {name}"
            )

        return plugin.run(
            **kwargs
        )

    # =========================================================
    # GET
    # =========================================================

    def get(self, name):
        if not name:
            return None

        return self.plugins.get(
            str(name).strip().lower()
        )

    # =========================================================
    # AVAILABLE
    # =========================================================

    def available(self):
        return list(
            self.plugins.keys()
        )

    # =========================================================
    # METADATA
    # =========================================================

    def metadata(self):
        return [
            plugin.info()
            for plugin in self.plugins.values()
        ]

    # =========================================================
    # STATUS
    # =========================================================

        enabled = 0

            else:

            "enabled": enabled,
            "disabled": disabled,
            "tools": self.available(),

class ToolExecutor:
    def __init__(self, controller):
        self.controller = controller

    def manager(self):
        manager = getattr(
            getattr(self.controller, "ai", None),
            "tools",
            None,
        )
        if manager is None:
            raise RuntimeError("Atlas tool manager is unavailable.")
        return manager

    def execute(self, name, prompt=""):
        name = str(name or "").strip().lower()
        plugin = self.manager().get(name)

        if plugin is None:
            raise ValueError(f"Tool '{name}' is not available.")

        if not getattr(plugin, "enabled", True):
            raise RuntimeError(f"Tool '{name}' is disabled.")

        if name == "calculator":
            return plugin.run(expression=prompt)

        if name == "math":
            operation = plugin.detect_operation(prompt)
            if operation is None:
                raise ValueError(
                    "Please specify an operation such as integrate or differentiate."
                )
            return plugin.run(
                expression=prompt,
                operation=operation,
                variable=plugin.detect_variable(prompt, prompt),
            )

        if name == "web_search":
            return plugin.run(expression=prompt)

        if name == "time":
            return plugin.run()

        if name == "calendar":

"""
Atlas V3 Tool Manager

Central interface for Atlas tools/plugins.

Responsibilities:
- Load and expose plugins
- Find the best tool for a message
- Execute tools safely
- Provide tool metadata
- Provide diagnostic status
- Keep routing logic out of AtlasBrain
"""

from typing import Any, Optional

from core.plugin_manager import PluginManager
from core.logger import logger


class ToolManager:
    """Central manager for Atlas tools."""

    def __init__(self):
        self.plugins = PluginManager()

        logger.info(
            "Available Atlas tools: %s",
            self.available(),
        )

    # ==================================================
    # EXECUTION
    # ==================================================

    def execute(
        self,
        name: str,
        **kwargs,
    ) -> Any:
        """
        Execute a registered tool by name.
        """

        if not name:
            raise ValueError(
                "Tool name is required."
            )

        plugin = self.get(name)

        if plugin is None:
            raise ValueError(
                f"Unknown tool: {name}"
            )

        if not getattr(
            plugin,
            "enabled",
            True,
        ):
            raise RuntimeError(
                f"Tool is disabled: {name}"
            )

        logger.info(
            "Executing tool: %s",
            name,
        )

        try:

            return self.plugins.execute(
                name,
                **kwargs,
            )

        except Exception as exc:

            logger.exception(
                "Tool '%s' execution failed: %s",
                name,
                exc,
            )

            raise

    # ==================================================
    # GET
    # ==================================================

    def get(
        self,
        name: str,
    ) -> Optional[Any]:
        """
        Return a plugin instance by name.
        """

        if not name:
            return None

        return self.plugins.get(
            name
        )

    # ==================================================
    # AVAILABLE
    # ==================================================

    def available(self):
        """
        Return all registered tool names.
        """

        return self.plugins.available()

    # ==================================================
    # MATCH
    # ==================================================

    def match(
        self,
        message: str,
    ) -> Optional[Any]:
        """
        Find the best tool capable of handling a message.

        Each plugin decides whether it can handle the message
        through its can_handle() method.

        Selection order:

        1. Enabled tools only
        2. Tools that can_handle() the message
        3. Lowest priority number wins
        4. Tool name breaks priority ties
        """

        if message is None:
            return None

        message = str(
            message
        ).strip()

        if not message:
            return None

        candidates = []

        for name in self.available():

            plugin = self.get(
                name
            )

            if plugin is None:
                continue

            # ------------------------------------------
            # Disabled plugin
            # ------------------------------------------

            if not getattr(
                plugin,
                "enabled",
                True,
            ):
                continue

            # ------------------------------------------
            # Check capability
            # ------------------------------------------

            try:

                can_handle = plugin.can_handle(
                    message
                )

            except Exception as exc:

                logger.warning(
                    "Tool '%s' can_handle() failed: %s",
                    name,
                    exc,
                )

                continue

            if not can_handle:
                continue

            # ------------------------------------------
            # Priority
            # ------------------------------------------

            priority = getattr(
                plugin,
                "priority",
                100,
            )

            try:
                priority = int(
                    priority
                )

            except (
                TypeError,
                ValueError,
            ):
                priority = 100

            candidates.append(
                (
                    priority,
                    name,
                    plugin,
                )
            )

        # ----------------------------------------------
        # No matching tool
        # ----------------------------------------------

        if not candidates:

            logger.debug(
                "No tool matched message: %s",
                message,
            )

            return None

        # ----------------------------------------------
        # Sort candidates
        # ----------------------------------------------

        candidates.sort(
            key=lambda item: (
                item[0],
                item[1].lower(),
            )
        )

        priority, name, plugin = candidates[0]

        logger.info(
            "Tool match: %s",
            name,
        )

        # ----------------------------------------------
        # Debug competing tools
        # ----------------------------------------------

        if len(candidates) > 1:

            logger.debug(
                "Tool candidates for '%s': %s",
                message,
                [
                    {
                        "name": item[1],
                        "priority": item[0],
                    }
                    for item in candidates
                ],
            )

        return plugin

    # ==================================================
    # TOOL INFO
    # ==================================================

    def metadata(self):
        """
        Return metadata for all loaded tools.
        """

        result = []

        for plugin in self.plugins.plugins.values():

            result.append(
                {
                    "name": getattr(
                        plugin,
                        "name",
                        "",
                    ),

                    "display_name": getattr(
                        plugin,
                        "display_name",
                        getattr(
                            plugin,
                            "name",
                            "",
                        ),
                    ),

                    "description": getattr(
                        plugin,
                        "description",
                        "",
                    ),

                    "icon": getattr(
                        plugin,
                        "icon",
                        "🧩",
                    ),

                    "triggers": list(
                        getattr(
                            plugin,
                            "triggers",
                            [],
                        )
                        or []
                    ),

                    "capabilities": list(
                        getattr(
                            plugin,
                            "capabilities",
                            [],
                        )
                        or []
                    ),

                    "priority": getattr(
                        plugin,
                        "priority",
                        100,
                    ),

                    "enabled": getattr(
                        plugin,
                        "enabled",
                        True,
                    ),

                    "version": getattr(
                        plugin,
                        "version",
                        "1.0",
                    ),

                    "author": getattr(
                        plugin,
                        "author",
                        "Atlas",
                    ),

                    "has_gui": getattr(
                        plugin,
                        "has_gui",
                        False,
                    ),

                    "gui_id": getattr(
                        plugin,
                        "gui_id",
                        None,
                    ),
                }
            )

        return result

    # ==================================================
    # STATUS
    # ==================================================

    def status(self):
        """
        Return diagnostic information about the tool system.
        """

        plugins = self.plugins.plugins

        enabled = 0
        disabled = 0

        for plugin in plugins.values():

            if getattr(
                plugin,
                "enabled",
                True,
            ):
                enabled += 1

            else:
                disabled += 1

        return {
            "total": len(
                plugins
            ),

            "enabled": enabled,

            "disabled": disabled,

            "tools": self.available(),
        }

"""Atlas V3 AI worker running outside the GUI thread."""

from threading import Event

from PySide6.QtCore import QObject, Signal, Slot

from ai.brain import AtlasBrain
from core.logger import logger


class AIWorker(QObject):
    chunk = Signal(str)
    error = Signal(str)
    finished = Signal()

    def __init__(self, tools=None):
        super().__init__()
        self.tools = tools
        self.brain = AtlasBrain(tools=tools)
        self.stop_event = Event()
        logger.info("AIWorker initialized with tool system")

    def request_stop(self):
        """Thread-safe cancellation request, including an in-flight HTTP call."""
        self.stop_event.set()
        try:
            self.brain.model.cancel_current()
        except Exception:
            pass

    @Slot(str)
    def process(self, message):
        self.stop_event.clear()
        try:
            for chunk in self.brain.stream_message(
                message,
                stop_event=self.stop_event,
            ):
                if self.stop_event.is_set():
                    break
                if chunk:
                    self.chunk.emit(str(chunk))
        except Exception as exc:
            logger.exception("AIWorker processing failed")
            self.error.emit(str(exc))
        finally:
            self.finished.emit()

    def _conversation_messages(self, conversation):
        if conversation is None:
            return []
        if isinstance(conversation, dict):

        for attr in ("get_messages", "history", "last"):
            fn = getattr(conversation, attr, None)
            if callable(fn):
                try:
                    result = fn()
                    if isinstance(result, list):
                        return list(result)
                except Exception:
                    pass

        for attr in ("messages", "current_messages"):
            result = getattr(conversation, attr, None)
            if isinstance(result, list):
                return list(result)
        return []

            self.thinking_frame = 0
            self.thinking_timer.start()

    def stop_thinking(self):
        if self.thinking_timer.isActive():
            self.thinking_timer.stop()

    def animate_thinking(self):
        if self.current_ai_message is None:
            return
        frames = ("•", "••", "•••", "••")
        frame = frames[self.thinking_frame % len(frames)]
        self.thinking_frame += 1
        self.current_ai_message.set_thinking(frame)
        if self.auto_follow:
            self.scroll_to_bottom(force=True)

    def show_profile_setup(self, event):
        if event.prompt:
            self.add_message("Atlas", event.prompt)
        card = ProfileSetupCard(event.tool, controller=self.controller)

        QTimer.singleShot(0, self._message_geometry_changed)

    def profile_completed(self, event):
        self.add_message(
            "Atlas",
            event.message or "Profile saved. I’ll use it to tailor future suggestions.",
        )
        QTimer.singleShot(0, self._message_geometry_changed)

    def _scroll_changed(self, value):
        if self._programmatic_scroll:
            return
        bar = self.scroll.verticalScrollBar()
        distance = bar.maximum() - value
        if distance <= 24:
            self.auto_follow = True
        elif self.is_generating:
            self.auto_follow = False

    def scroll_to_bottom(self, force=False):
        if not force and not self.auto_follow:
            return
        bar = self.scroll.verticalScrollBar()
        self._programmatic_scroll = True
        try:
            bar.setValue(bar.maximum())
        finally:
            self._programmatic_scroll = False

def markdown_to_html(text):
    raw = html.unescape(str(text or ""))
    raw = re.sub(r"<br\s*/?>", "\n", raw, flags=re.IGNORECASE)
    raw = re.sub(r"</?(?:p|div)>", "\n", raw, flags=re.IGNORECASE)
    text = html.escape(raw)
    text = re.sub(
        r"```(?:[^\n]*)\n(.*?)```",
        lambda m: (
            '<pre style="white-space:pre-wrap; margin:8px 0; '
            'font-family:monospace;">' + m.group(1) + "</pre>"
        ),
        text,
        flags=re.DOTALL,
    )
    text = re.sub(r"^### (.+)$", r"<h3>\1</h3>", text, flags=re.MULTILINE)
    text = re.sub(r"^## (.+)$", r"<h2>\1</h2>", text, flags=re.MULTILINE)
    text = re.sub(r"^# (.+)$", r"<h1>\1</h1>", text, flags=re.MULTILINE)
    text = re.sub(r"\*\*(.+?)\*\*", r"<b>\1</b>", text)
    text = re.sub(r"`([^`]+)`", r"<code>\1</code>", text)
    text = re.sub(
        r"\[([^\]]+)\]\((https?://[^)]+)\)",
        r'<a href="\2">\1</a>',
        text,
    )
    return text.replace("\n", "<br>")

    def clean_value(key, value):
        value = str(value or "").strip()
        if not value:
            return None

        if key in (
            "equipment",
            "restrictions",
            "allergies",
            "intolerances",
            "preferences",
        ):
            return [
                item.strip()
                for item in value.split(",")
                if item.strip()
            ]

        if key == "age":
            try:
                return int(value)
            except ValueError:
                return None

        return value

                result = self.controller.apply_tool_profile(self.tool, profile)

                result = "Profile submitted."

        except Exception as exc:
            self.status.setText(f"Could not save profile: {exc}")

        self.history_list.setContextMenuPolicy(Qt.CustomContextMenu)
        self.history_list.customContextMenuRequested.connect(self.show_history_menu)

        # Controller auto-titles the conversation before the event bus
        # reaches this subscriber, so refresh immediately.
        QTimer.singleShot(0, self.refresh_history)

    def refresh_history(self):
        self.history_list.clear()
        if self.controller is None:
            return

        try:
            conversations = self.controller.get_conversations()
            active_id = self.controller.get_active_conversation_id()
        except Exception:
            return

        if isinstance(conversations, list):
            items = conversations
        elif isinstance(conversations, dict):
            items = []
            for cid, data in conversations.items():
                data = dict(data or {})
                data["id"] = data.get("id", cid)
                items.append(data)
        else:
            items = []

        active_item = None

        for data in items:
            if not isinstance(data, dict) or not data.get("id"):
                continue

            item = QListWidgetItem(
                str(data.get("title") or "New Chat")
            )
            item.setData(Qt.UserRole, str(data["id"]))
            self.history_list.addItem(item)

            if active_id is not None and str(active_id) == str(data["id"]):
                active_item = item

        if active_item is not None:
            self.history_list.setCurrentItem(active_item)

    def handle_new_chat(self):
        if self.controller is None:
            return

        try:
            self.controller.create_conversation()
            conversation = self.controller.get_active_conversation()
            self.refresh_history()
            self.conversation_loaded.emit(conversation)
        except Exception as exc:
            QMessageBox.warning(
                self,
                "Atlas",
                f"Could not create new chat:\n\n{exc}",
            )

    def handle_history_clicked(self, item):
        cid = item.data(Qt.UserRole)
        if not cid or self.controller is None:
            return

        try:
            if not self.controller.load_conversation(cid):
                QMessageBox.warning(
                    self,
                    "Atlas",
                    "Could not load conversation.",
                )
                return

            conversation = self.controller.get_active_conversation()
            self.conversation_loaded.emit(conversation)
            self.refresh_history()

        except Exception as exc:
            QMessageBox.warning(
                self,
                "Atlas",
                f"Could not load conversation:\n\n{exc}",
            )

    def show_history_menu(self, position):
        item = self.history_list.itemAt(position)
        if item is None:
            return

        cid = item.data(Qt.UserRole)
        menu = QMenu(self)
        rename_action = menu.addAction("Rename")
        delete_action = menu.addAction("Delete")
        action = menu.exec(
            self.history_list.mapToGlobal(position)
        )

        if action == rename_action:
            self.rename_conversation(cid)
        elif action == delete_action:
            self.delete_conversation(cid)

    def rename_conversation(self, cid):
        current = self.history_list.currentItem()
        current_title = current.text() if current else "New Chat"

        title, ok = QInputDialog.getText(
            self,
            "Rename Conversation",
            "Conversation name:",
            text=current_title,
        )
        if not ok or not title.strip():
            return

        if self.controller.rename_conversation(cid, title.strip()):
            self.refresh_history()

    def delete_conversation(self, cid):
        if self.controller is None:
            return

        if self.controller.state.is_generating:
            QMessageBox.information(
                self,
                "Atlas",
                "Please wait for Atlas to finish before deleting a chat.",
            )
            return

        if (
            QMessageBox.question(
                self,
                "Delete Conversation",
                "Delete this conversation?",
                QMessageBox.Yes | QMessageBox.No,
                QMessageBox.No,
            )
            != QMessageBox.Yes
        ):
            return

        if not self.controller.delete_conversation(cid):
            QMessageBox.warning(
                self,
                "Atlas",
                "Could not delete conversation.",
            )
            return

        conversation = self.controller.get_active_conversation()
        self.refresh_history()
        self.conversation_loaded.emit(conversation)

class ToolTask(QObject):
    """Execute a tool away from the GUI thread."""
    result = Signal(str)
    error = Signal(str)
    finished = Signal()

    def __init__(self, executor, name, prompt):
        super().__init__()
        self.executor = executor
        self.name = name
        self.prompt = prompt

    @Slot()
    def run(self):
        try:
            result = self.executor.execute(self.name, self.prompt)
            self.result.emit(str(result or ""))
        except Exception as exc:
            self.error.emit(str(exc))
        finally:
            self.finished.emit()


class AnalogClock(QWidget):
    def __init__(self, parent=None):
        super().__init__(parent)
        self.setMinimumSize(250, 250)
        self.timer = QTimer(self)
        self.timer.setInterval(1000)
        self.timer.timeout.connect(self.update)
        self.timer.start()

    def paintEvent(self, event):
        now = datetime.now()
        side = min(self.width(), self.height()) - 18
        center = QPointF(self.width() / 2, self.height() / 2)
        radius = side / 2

        painter = QPainter(self)
        painter.setRenderHint(QPainter.Antialiasing)
        painter.setPen(QPen(self.palette().text(), 2))
        painter.setBrush(Qt.NoBrush)
        painter.drawEllipse(center, radius, radius)

        for hour in range(12):
            rad = math.radians(hour * 30 - 90)
            outer = QPointF(
                center.x() + radius * 0.88 * math.cos(rad),
                center.y() + radius * 0.88 * math.sin(rad),
            )
            inner = QPointF(
                center.x() + radius * 0.79 * math.cos(rad),
                center.y() + radius * 0.79 * math.sin(rad),
            )
            painter.drawLine(inner, outer)

        second = now.second + now.microsecond / 1_000_000
        minute = now.minute + second / 60
        hour = (now.hour % 12) + minute / 60

        def draw_hand(angle, length, width):
            rad = math.radians(angle - 90)
            end = QPointF(
                center.x() + radius * length * math.cos(rad),
                center.y() + radius * length * math.sin(rad),
            )
            painter.setPen(QPen(self.palette().text(), width))
            painter.drawLine(center, end)

        draw_hand(hour * 30, 0.50, 6)
        draw_hand(minute * 6, 0.69, 4)
        draw_hand(second * 6, 0.78, 2)

        painter.setBrush(QBrush(self.palette().text()))
        painter.drawEllipse(center, 5, 5)
        painter.end()

        self.expression = str(result)

"""
Atlas V3 Browser Runtime

Headless runtime for the LAN/browser interface.
Uses the same Brain, Ollama model, ToolManager and persistent
ConversationService as Atlas without constructing the Qt GUI.
"""

from __future__ import annotations

from dataclasses import dataclass
from threading import RLock
from typing import Any, Dict, Optional

from ai.brain import AtlasBrain
from ai.model import AtlasModel
from core.tool_manager import ToolManager
from ai.services.conversation_service import ConversationService
from config import MODEL_NAME, OLLAMA_HOST


@dataclass
class WebSession:
    conversation: ConversationService
    brain: AtlasBrain
    lock: RLock


class WebAtlasRuntime:
    def __init__(self):
        self._sessions: Dict[str, WebSession] = {}
        self._sessions_lock = RLock()

        self.tools = ToolManager()
        self.model = AtlasModel()
        self.catalog = ConversationService()

    def session(self, session_id: str) -> WebSession:
        session_id = str(session_id).strip()
        if not session_id:
            raise ValueError("session_id is required")

        with self._sessions_lock:
            existing = self._sessions.get(session_id)
            if existing is not None:
                return existing

            conversation = ConversationService()
            conversation_id = conversation.new()

            # The current Brain accepts memory=None and does not require
            # the old memory.manager.MemoryManager object.
            brain = AtlasBrain(
                model=self.model,
                memory=None,
                tools=self.tools,
            )

            session = WebSession(
                conversation=conversation,
                brain=brain,
                lock=RLock(),
            )

            session.conversation.load(conversation_id)
            brain.set_conversation_history(
                session.conversation.history()
            )

            self._sessions[session_id] = session
            return session

    def ensure_conversation(
        self,
        session_id: str,
        conversation_id: Optional[str] = None,
    ) -> WebSession:
        session = self.session(session_id)

        if conversation_id:
            conversation_id = str(conversation_id)

            if session.conversation.current_id != conversation_id:
                if not session.conversation.load(conversation_id):
                    raise ValueError("Conversation not found.")

                session.brain.set_conversation_history(
                    session.conversation.history()
                )

        return session

    def conversations(self):
        return self.catalog.list()

    def create_conversation(self, session_id: str):
        session = self.session(session_id)

        with session.lock:
            conversation_id = session.conversation.new()
            session.brain.set_conversation_history([])
            return conversation_id

    def load_conversation(
        self,
        session_id: str,
        conversation_id: str,
    ):
        session = self.session(session_id)

        with session.lock:
            if not session.conversation.load(conversation_id):
                raise ValueError("Conversation not found.")

            session.brain.set_conversation_history(
                session.conversation.history()
            )

            return session.conversation.history()

    def delete_conversation(
        self,
        session_id: str,
        conversation_id: str,
    ):
        session = self.session(session_id)

        with session.lock:
            ok = session.conversation.delete(conversation_id)

            if (
                ok
                and session.conversation.current_id == conversation_id
            ):
                new_id = session.conversation.new()
                session.brain.set_conversation_history([])
                return new_id

            return None

    def execute_tool(
        self,
        session_id: str,
        tool_name: str,
        action: str = "",
        expression: str = "",
        arguments: Optional[Dict[str, Any]] = None,
    ):
        """Execute a tool directly from the browser Tool Workspace."""

        tool_name = str(tool_name or "").strip().lower()
        if not tool_name:
            raise ValueError("Tool name is required.")

        arguments = dict(arguments or {})

        if action:
            arguments["action"] = action

        if expression:
            arguments["expression"] = expression

        # Some plugins expect expression while others accept named kwargs.
        # The ToolManager/PluginManager layer handles the actual dispatch.
        result = self.tools.execute(
            tool_name,
            **arguments,
        )

        return {
            "tool": tool_name,
            "result": str(result),
        }

    def status(self, session_id: str):
        session = self.session(session_id)

        try:
            tool_status = self.tools.status()
        except Exception:
            tool_status = {
                "total": 0,
                "enabled": 0,
                "disabled": 0,
                "tools": [],
            }

        return {
            "model": MODEL_NAME,
            "ollama": OLLAMA_HOST,
            "conversation_id": session.conversation.current_id,
            "conversation_title": session.conversation.current_title,
            "history_messages": len(session.conversation.history()),
            "tools": tool_status,
            "web_search": True,
        }

    def chat_stream(
        self,
        session_id: str,
        message: str,
        conversation_id: Optional[str] = None,
    ):
        session = self.ensure_conversation(
            session_id,
            conversation_id,
        )

        with session.lock:
            session.conversation.add_user(message)
            session.conversation.save()

            history_before = session.conversation.history()[:-1]
            session.brain.set_conversation_history(history_before)

            full_response = []

            try:
                for chunk in session.brain.stream_message(
                    message,
                    conversation_history=history_before,
                ):
                    text = str(chunk)
                    full_response.append(text)

                    yield {
                        "type": "chunk",
                        "text": text,
                    }

                response = "".join(full_response).strip()

                if response:
                    session.conversation.add_assistant(response)
                    session.conversation.save()

                session.brain.set_conversation_history(
                    session.conversation.history()
                )

                yield {
                    "type": "done",
                    "conversation_id": session.conversation.current_id,
                    "title": session.conversation.current_title,
                    "web": {
                        "query": getattr(
                            session.brain,
                            "last_web_query",
                            None,
                        ),
                        "sources": getattr(
                            session.brain,
                            "last_web_sources",
                            [],
                        ),
                    },
                }

            except Exception as exc:
                yield {
                    "type": "error",
                    "message": str(exc),
                }

async function refreshHistory() {
  try {
    const res = await fetch(
      "/api/conversations",
      {
        headers: {
          "X-Atlas-Session": sessionId
        }
      }
    );

    if (!res.ok) {
      throw new Error("History unavailable");
    }

    const data = await res.json();
    historyEl.innerHTML = "";

    for (const item of data.conversations || []) {
      const row = document.createElement("div");

      row.className =
        "history-item" +
        (
          item.id === conversationId
            ? " active"
            : ""
        );

      const title = document.createElement("span");
      title.className = "history-title";
      title.textContent =
        item.title || "New Chat";

      const del = document.createElement("button");
      del.className = "delete-chat";
      del.title = "Delete";
      del.textContent = "×";

      del.addEventListener("click", async event => {
        event.stopPropagation();

        if (!confirm("Delete this conversation?")) {
          return;
        }

        const deleteRes = await fetch(
          `/api/conversations/${encodeURIComponent(item.id)}`,
          {
            method: "DELETE",
            headers: {
              "X-Atlas-Session": sessionId
            }
          }
        );

        if (!deleteRes.ok) return;

        const deleted = await deleteRes.json();

        if (item.id === conversationId) {
          if (deleted.replacement) {
            await loadConversation(
              deleted.replacement
            );
          } else {
            conversationId = null;
            renderConversation([]);
            conversationTitleEl.textContent = "New Chat";
          }
        }

        await refreshHistory();
      });

      row.appendChild(title);
      row.appendChild(del);

      row.addEventListener("click", async () => {
        await loadConversation(item.id);
      });

      historyEl.appendChild(row);
    }
  } catch (error) {
    console.error(error);
  }
}

      }
    );

    const data = await res.json();

    if (!res.ok) {
      throw new Error(
        data.detail || "Tool execution failed."
      );
    }

    activeToolResult =
      String(data.result ?? "");

    toolResult.textContent =
      activeToolResult;

    toolResultWrap.classList.remove("hidden");
    toolInsert.classList.remove("hidden");

    webStatusEl.textContent =
      activeTool.name === "web_search"
        ? "Web Search executed"
        : "Tool executed";

  } catch (error) {
    toolResult.textContent =
      `Error: ${error.message}`;

    toolResultWrap.classList.remove(
      "hidden"
    );

  } finally {
    toolRun.disabled = false;
    toolRun.textContent = "Run Tool";
  }
}

async function sendMessage() {
  if (generating) return;

  const text =
    input.value.trim();

  if (!text) return;

  if (!conversationId) {
    await createNewChat();
  }

  addMessage(
    "You",
    text,
    "user"
  );

  input.value = "";
  input.style.height = "auto";

  const atlas =
    addMessage(
      "Atlas",
      "",
      "atlas"
    );

  const body = atlas.body;

  generating = true;
  abortController =
    new AbortController();

  sendButton.classList.add("hidden");
  stopButton.classList.remove("hidden");
  input.disabled = true;

  setState("Thinking");
  setActivity(
    true,
    "Atlas is thinking…"
  );

  try {
    const response =
      await fetch(
        "/api/chat",
        {
          method: "POST",
          headers: headers(),
          body: JSON.stringify({
            message: text,
            conversation_id: conversationId,
            session_id: sessionId
          }),
          signal: abortController.signal
        }
      );

    if (!response.ok) {
      throw new Error(
        `HTTP ${response.status}`
      );
    }

    if (!response.body) {
      throw new Error(
        "Streaming is unavailable."
      );
    }

    const reader =
      response.body.getReader();

    const decoder =
      new TextDecoder("utf-8");

    let buffer = "";

    while (true) {
      const {
        value,
        done
      } = await reader.read();

      if (done) break;

      buffer += decoder.decode(
        value,
        { stream: true }
      );

      const lines =
        buffer.split("\n");

      buffer =
        lines.pop() || "";

      for (const line of lines) {
        if (!line.trim()) continue;

        let event;

        try {
          event = JSON.parse(line);
        } catch {
          continue;
        }

        if (event.type === "chunk") {
          setActivity(
            true,
            "Atlas is writing…"
          );

          body.textContent +=
            event.text || "";

          scrollIfPinned();
        }

        if (event.type === "done") {
          conversationId =
            event.conversation_id ||
            conversationId;

          conversationTitleEl.textContent =
            event.title ||
            conversationTitleEl.textContent;

          const sources =
            event.web?.sources || [];

          if (sources.length) {
            renderSources(
              atlas.bubble,
              sources
            );

            webStatusEl.textContent =
              `Searched: ${
                event.web.query || "web"
              }`;
          } else {
            webStatusEl.textContent =
              "Ready for live search";
          }

          await refreshHistory();
        }

        if (event.type === "error") {
          throw new Error(
            event.message ||
            "Atlas error"
          );
        }
      }
    }

  } catch (error) {
    if (error.name === "AbortError") {
      body.textContent +=
        "\n\n[Generation stopped]";
    } else {
      body.textContent =
        `Error: ${error.message}`;
      setState("Error");
    }

  } finally {
    generating = false;
    abortController = null;
    input.disabled = false;

    sendButton.classList.remove(
      "hidden"
    );

    stopButton.classList.add(
      "hidden"
    );

    setActivity(false);

    if (
      stateEl.textContent !==
      "Error"
    ) {
      setState("Ready");
    }

    input.focus();

    await refreshStatus();

    scrollIfPinned();
  }
}

async function init() {
  updateClock();
  setInterval(
    updateClock,
    1000
  );

  await loadTools();
  await refreshHistory();

  try {
    const statusRes =
      await fetch(
        "/api/status",
        {
          headers: {
            "X-Atlas-Session":
              sessionId
          }
        }
      );

    const status =
      await statusRes.json();

    if (status.conversation_id) {
      conversationId =
        status.conversation_id;

      await loadConversation(
        conversationId
      );

    } else {
      await createNewChat();
    }

  } catch {
    await createNewChat();
  }

  input.focus();
}

"""
Atlas V3 Plugin Base

Base interface for all Atlas plugins/tools.

Responsibilities:
- Define the standard plugin contract
- Provide deterministic message matching
- Provide optional argument extraction
- Provide standardized metadata
- Keep plugins independent from AtlasBrain

This class is intentionally LLM-free.

The plugin system should remain deterministic at this layer.
Higher-level planning/reasoning can be added above it later.
"""

from typing import Any, Dict, Optional


class AtlasPlugin:
    """
    Base class for every Atlas plugin/tool.

    Required:
        name
        description
        can_handle()
        run()

    Optional:
        extract_arguments()

    Metadata fields provide a consistent interface for
    ToolManager, future planners, Skills, and GUI systems.
    """

    # ==========================================================
    # IDENTITY
    # ==========================================================

    name = "unknown"

    display_name = "Atlas Plugin"

    description = "Atlas plugin"

    version = "1.0"

    author = "Atlas"

    # ==========================================================
    # UI
    # ==========================================================

    icon = "🧩"

    has_gui = False

    gui_id = None

    # ==========================================================
    # ROUTING
    # ==========================================================

    triggers = []

    capabilities = []

    # Lower number = higher priority.
    priority = 100

    # Plugin can be disabled without being unloaded.
    enabled = True

    # ==========================================================
    # MATCHING
    # ==========================================================

    def can_handle(
        self,
        message: str,
    ) -> bool:
        """
        Determine whether this plugin can handle a message.

        Plugins should override this method.

        Requirements:
        - Must be deterministic.
        - Must not call the LLM.
        - Must not execute the tool.
        - Should return True/False only.
        """

        return False

    # ==========================================================
    # ARGUMENT EXTRACTION
    # ==========================================================

    def extract_arguments(
        self,
        message: str,
    ) -> Dict[str, Any]:
        """
        Extract structured arguments from a user message.

        Example:

            User:
                "Calculate 25 * 4"

            Returns:

                {
                    "expression": "25 * 4"
                }

        Plugins may override this when they need structured
        arguments.

        The default implementation preserves the raw message.
        """

        return {
            "message": message
        }

    # ==========================================================
    # EXECUTION
    # ==========================================================

    def run(
        self,
        message: Optional[str] = None,
        **kwargs,
    ) -> Any:
        """
        Execute the plugin.

        Every executable plugin MUST override this method.
        """

        raise NotImplementedError(
            f"Plugin '{self.name}' "
            "does not implement run()."
        )

    # ==========================================================
    # METADATA
    # ==========================================================

    def info(self) -> Dict[str, Any]:
        """
        Return standardized plugin metadata.

        ToolManager and future planner/skills systems can use
        this without knowing the concrete plugin class.
        """

        return {
            "name": self.name,
            "display_name": self.display_name,
            "description": self.description,
            "version": self.version,
            "author": self.author,
            "icon": self.icon,
            "triggers": list(self.triggers or []),
            "capabilities": list(
                self.capabilities or []
            ),
            "priority": self.priority,
            "enabled": self.enabled,
            "has_gui": self.has_gui,
            "gui_id": self.gui_id,
        }

    # ==========================================================
    # STATUS
    # ==========================================================

    def is_enabled(self) -> bool:
        """
        Return whether this plugin is currently enabled.
        """

        return bool(
            self.enabled
        )

    # ==========================================================
    # REPRESENTATION
    # ==========================================================

    def __repr__(self) -> str:
        return (
            f"<{self.__class__.__name__} "
            f"name='{self.name}'>"
        )

"""
Atlas V3 deterministic math engine.

Rejects arbitrary prose before SymPy parsing, supports implicit
multiplication, differentiation, integration, simplification, expansion,
factorization and equation solving.
"""

import re
import sympy as sp
from sympy.parsing.sympy_parser import (
    convert_xor,
    implicit_multiplication_application,
    parse_expr,
    standard_transformations,
)


class MathEngine:
    DEFAULT_VARIABLE = "x"
    SYMBOL_NAMES = ("x", "y", "z", "t", "a", "b", "c", "n")

    TRANSFORMATIONS = (
        standard_transformations
        + (convert_xor, implicit_multiplication_application)
    )

    FUNCTIONS = {
        "sin": sp.sin,
        "cos": sp.cos,
        "tan": sp.tan,
        "cot": sp.cot,
        "sec": sp.sec,
        "csc": sp.csc,
        "asin": sp.asin,
        "acos": sp.acos,
        "atan": sp.atan,
        "sqrt": sp.sqrt,
        "log": sp.log,
        "ln": sp.log,
        "exp": sp.exp,
        "abs": sp.Abs,
        "pi": sp.pi,
        "E": sp.E,
    }

    def __init__(self):
        self.symbols = {
            name: sp.Symbol(name)
            for name in self.SYMBOL_NAMES
        }

    def detect_operation(self, text):
        low = str(text or "").lower()

        if re.search(r"\b(?:differentiat(?:e|ion)|diffrentiat(?:e|ion))\b", low):
            return "differentiate"
        if re.search(r"\bderivative\b", low):
            return "differentiate"
        if re.search(r"\bintegr(?:ate|al)\b", low):
            return "integrate"
        if re.search(r"\bsimplify\b", low):
            return "simplify"
        if re.search(r"\bexpand\b", low):
            return "expand"
        if re.search(r"\bfactor(?:ize|ise|ization)?\b", low):
            return "factor"
        if re.search(r"\bsolve\b|\bequation\b|\broots?\b", low):
            return "solve"
        if re.search(
            r"\bcalculate\b|\bcalc\b|\bcompute\b|\bevaluate\b",
            low,
        ):
            return "calculate"

        # Plain numeric arithmetic belongs to CalculatorPlugin.
        # MathPlugin should only claim symbolic/named operations.
        return None

    def extract_expression(
        self,
        text,
        operation=None,
        fallback="",
    ):
        text = str(text or "").strip().rstrip("?!.,;")
        if not text:
            return fallback

        op = operation or self.detect_operation(text)
        if op is None:
            return ""

        patterns = [
            r"^\s*what\s+is\s+the\s+derivative\s+of\s+",
            r"^\s*find\s+the\s+derivative\s+of\s+",
            r"^\s*what\s+is\s+the\s+integral\s+of\s+",
            r"^\s*find\s+the\s+integral\s+of\s+",
            r"^\s*derivative\s+of\s+",
            r"^\s*integral\s+of\s+",
            r"^\s*diffrentiate\s+",
            r"^\s*diffrentiation\s+of\s+",
            r"^\s*differentiate\s+",
            r"^\s*integrate\s+",
            r"^\s*calculate\s+",
            r"^\s*calc\s+",
            r"^\s*compute\s+",
            r"^\s*evaluate\s+",
            r"^\s*simplify\s+",
            r"^\s*expand\s+",
            r"^\s*factor(?:ize|ise)?\s+",
            r"^\s*solve\s+",
        ]

        result = text
        for pattern in patterns:
            cleaned = re.sub(
                pattern,
                "",
                result,
                flags=re.IGNORECASE,
            )
            if cleaned != result:
                result = cleaned.strip()
                break

        if op == "calculate":
            result = re.sub(
                r"^\s*what(?:'s|\s+is)\s+",
                "",
                result,
                flags=re.IGNORECASE,
            ).strip()

        result = re.sub(
            r"\s+(?:with\s+respect\s+to|respect\s+to|wrt)\s+[A-Za-z_]\w*$",
            "",
            result,
            flags=re.IGNORECASE,
        ).strip()

        return result

    def _validate_expression(self, expression):
        expr = str(expression or "").strip()

        if not expr:
            raise ValueError(
                "No mathematical expression was provided."
            )

        if len(expr) > 500:
            raise ValueError("Expression is too long.")

        if not re.fullmatch(
            r"[0-9A-Za-z_+\-*/%^().,\s]+",
            expr,
        ):
            raise ValueError(
                "Invalid mathematical characters."
            )

        identifiers = re.findall(
            r"[A-Za-z_]\w*",
            expr,
        )
        allowed = set(self.SYMBOL_NAMES) | set(self.FUNCTIONS)

        for name in identifiers:
            if name not in allowed:
                raise ValueError(
                    f"'{name}' is not a recognized mathematical "
                    "symbol or function."
                )

        if not re.search(
            r"\d|[xyztabcn]\b",
            expr,
        ):
            raise ValueError(
                "Please provide a mathematical expression."
            )

        return (
            expr
            .replace("−", "-")
            .replace("×", "*")
            .replace("÷", "/")
        )

    def parse_expression(self, expression):
        expr = self._validate_expression(expression)

        local_dict = dict(self.symbols)
        local_dict.update(self.FUNCTIONS)

        return parse_expr(
            expr,
            local_dict=local_dict,
            transformations=self.TRANSFORMATIONS,
            evaluate=True,
        )

    def calculate(self, expression):
        return sp.simplify(
            self.parse_expression(expression)
        )

    def differentiate(self, expression, variable="x"):
        parsed = self.parse_expression(expression)
        variable = str(variable or "x").strip()
        symbol = self.symbols.get(
            variable,
            sp.Symbol(variable),
        )
        return sp.simplify(
            sp.diff(parsed, symbol)
        )

    def integrate(self, expression, variable="x"):
        parsed = self.parse_expression(expression)
        variable = str(variable or "x").strip()
        symbol = self.symbols.get(
            variable,
            sp.Symbol(variable),
        )
        return sp.integrate(parsed, symbol)

    def simplify(self, expression):
        return sp.simplify(
            self.parse_expression(expression)
        )

    def expand(self, expression):
        return sp.expand(
            self.parse_expression(expression)
        )

    def factor(self, expression):
        return sp.factor(
            self.parse_expression(expression)
        )

    def solve(self, equation, variable="x"):
        equation = str(equation or "").strip()
        if not equation:
            raise ValueError("Equation is empty.")

        variable = str(variable or "x").strip()
        symbol = self.symbols.get(
            variable,
            sp.Symbol(variable),
        )

        if "=" in equation:
            left, right = equation.split("=", 1)
            return sp.solve(
                sp.Eq(
                    self.parse_expression(left),
                    self.parse_expression(right),
                ),
                symbol,
            )

        return sp.solve(
            self.parse_expression(equation),
            symbol,
        )

    def execute(self, operation, expression, variable="x"):
        operation = str(operation or "").lower().strip()

        aliases = {
            "calc": "calculate",
            "evaluate": "calculate",
            "derivative": "differentiate",
            "diff": "differentiate",
            "integral": "integrate",
            "factorize": "factor",
            "factorise": "factor",
            "equation": "solve",
            "equations": "solve",
        }
        operation = aliases.get(operation, operation)

        if operation == "calculate":
            return self.calculate(expression)
        if operation == "differentiate":
            return self.differentiate(expression, variable)
        if operation == "integrate":
            return self.integrate(expression, variable)
        if operation == "simplify":
            return self.simplify(expression)
        if operation == "expand":
            return self.expand(expression)
        if operation == "factor":
            return self.factor(expression)
        if operation == "solve":
            return self.solve(expression, variable)

        raise ValueError(
            "Unsupported mathematical operation."
        )

    def format(self, expression):
        return self._pretty(expression)

    @staticmethod
    def _pretty(expression):
        if isinstance(expression, (list, tuple)):
            return ", ".join(
                MathEngine._pretty(item)
                for item in expression
            )

        if isinstance(expression, sp.Equality):
            return (
                MathEngine._pretty(expression.lhs)
                + " = "
                + MathEngine._pretty(expression.rhs)
            )

        text = sp.sstr(expression)

        superscripts = str.maketrans(
            "0123456789+-=()",
            "⁰¹²³⁴⁵⁶⁷⁸⁹⁺⁻⁼⁽⁾",
        )

        def power_replace(match):
            base = match.group(1)
            power = match.group(2)

            if len(power) <= 3 and re.fullmatch(
                r"[0-9+\-()]+",
                power,
            ):
                return base + power.translate(superscripts)

            return f"{base}^{power}"

        text = re.sub(
            r"([A-Za-z0-9.)]+)\*\*([0-9+\-()]+)",
            power_replace,
            text,
        )
        text = text.replace("*", "·")
        text = text.replace("/", "⁄")

        return text

    def explain(
        self,
        question,
        expression="",
        result="",
        variable="x",
    ):
        question = str(question or "").strip()
        operation = self.detect_operation(question)

        if operation is None:
            return (
                "### Math Error\n\n"
                "I need a specific mathematical operation, such as "
                "differentiate, integrate, simplify, solve or calculate."
            )

        expression = str(expression or "").strip()
        if not expression:
            expression = self.extract_expression(
                question,
                operation,
            )

        if not expression:
            return (
                "### Math Error\n\n"
                "Please provide the mathematical expression to work on."
            )

        try:
            if not result:
                result = self.execute(
                    operation,
                    expression,
                    variable,
                )

            formatted = self.format(result)

        except Exception as exc:
            return f"### Math Error\n\n{exc}"

        lines = [
            "### Atlas Math",
            "",
            "**Expression**",
            "",
            f"`{expression}`",
            "",
            "**Answer**",
            "",
            formatted,
        ]

        if operation == "differentiate":
            lines += [
                "",
                "**Steps**",
                "",
                f"1. Treat `{variable}` as the differentiation variable.",
                "2. Differentiate each term.",
                f"3. Simplify the result: **{formatted}**.",
            ]
        elif operation == "integrate":
            lines += [
                "",
                "**Steps**",
                "",
                f"1. Treat `{variable}` as the integration variable.",
                "2. Integrate each term.",
                f"3. Add the constant of integration: **{formatted} + C**.",
            ]
        else:
            lines += [
                "",
                "**Steps**",
                "",
                "1. Parse the expression symbolically.",
                f"2. Apply **{operation}**.",
                f"3. Simplify the result: **{formatted}**.",
            ]

        return "\n".join(lines)

    def detect_variable(self, text, expression=""):
        patterns = (
            r"\bwith\s+respect\s+to\s+([a-zA-Z_]\w*)",
            r"\brespect\s+to\s+([a-zA-Z_]\w*)",
            r"\bwrt\s+([a-zA-Z_]\w*)",

        for pattern in patterns:
            match = re.search(
                pattern,

        expression = str(expression or "")
        for symbol in self.engine.SYMBOL_NAMES:
            if re.search(
                rf"\b{re.escape(symbol)}\b",

    def can_handle(self, message):

        if not text:

        if operation is None or operation == "calculate":

            text,

"""Atlas calculator: exact arithmetic with fractions."""

import ast
import math
import operator
import re
from fractions import Fraction
from plugins.base import AtlasPlugin

_OPERATORS = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.Mod: operator.mod,
    ast.Pow: operator.pow,
    ast.FloorDiv: operator.floordiv,
    ast.UAdd: operator.pos,
    ast.USub: operator.neg,
}

class CalculatorPlugin(AtlasPlugin):
    name = "calculator"
    display_name = "Calculator"
    description = "Exact numeric arithmetic, including fractions."
    icon = "🧮"
    has_gui = True
    gui_id = "calculator"
    priority = 10
    triggers = ["calculate", "calc", "compute", "evaluate", "what is", "what's"]

    def can_handle(self, message):
        text = str(message or "").strip()
        if not text:
            return False
        low = text.lower()
        if any(low == t or low.startswith(t + " ") for t in self.triggers):
            expr = self._extract_expression(text)
            return bool(expr) and self._is_numeric_expression(expr)
        return self._is_numeric_expression(text)

    def _extract_expression(self, message):
        text = str(message or "").strip().rstrip("?!.,")
        low = text.lower()
        for prefix in ("what is", "what's", "calculate", "calc", "compute", "evaluate"):
            if low == prefix:
                return ""
            if low.startswith(prefix + " "):
                return text[len(prefix):].strip()
        return text

    @staticmethod
    def _normalize(text):
        return (str(text or "")
                .replace("÷", "/")
                .replace("×", "*")
                .replace("−", "-")
                .replace("–", "-")
                .replace("—", "-")
                .replace("π", "pi"))

    def _is_numeric_expression(self, text):
        text = self._normalize(text).strip()
        if not text or not re.search(r"\d", text):
            return False
        if not re.fullmatch(r"[0-9eEpiPI+\-*/%^().\s]+", text):
            return False
        return bool(re.search(r"[+\-*/%^]", text))

    def _evaluate(self, node):
        if isinstance(node, ast.Constant):
            if isinstance(node.value, int):
                return Fraction(node.value, 1)
            if isinstance(node.value, float):
                return Fraction(str(node.value))
            raise ValueError("Invalid constant.")
        if isinstance(node, ast.BinOp):
            op = _OPERATORS.get(type(node.op))
            if op is None:
                raise ValueError("Unsupported operator.")
            return op(self._evaluate(node.left), self._evaluate(node.right))
        if isinstance(node, ast.UnaryOp):
            op = _OPERATORS.get(type(node.op))
            if op is None:
                raise ValueError("Unsupported operator.")
            return op(self._evaluate(node.operand))
        if isinstance(node, ast.Name) and node.id.lower() in ("pi", "e"):
            return math.pi if node.id.lower() == "pi" else math.e
        raise ValueError("Unsupported expression.")

    def run(self, expression="", **kwargs):
        try:
            expression = self._normalize(self._extract_expression(expression))
            if not self._is_numeric_expression(expression):
                return "Invalid expression."
            expression = expression.replace("^", "**")
            tree = ast.parse(expression, mode="eval")
            return self._format_result(self._evaluate(tree.body))
        except ZeroDivisionError:
            return "Cannot divide by zero."
        except (ValueError, SyntaxError, TypeError, OverflowError):
            return "Invalid expression."
        except Exception:
            return "Invalid expression."

    @staticmethod
    def _format_result(result):
        if isinstance(result, Fraction):
            return str(result.numerator) if result.denominator == 1 else f"{result.numerator}/{result.denominator}"
        if isinstance(result, float):
            if not math.isfinite(result):
                return "Result is not finite."
            return str(int(result)) if result.is_integer() else f"{result:.12g}"
        return str(result)

    TIMEOUT = 10
    MAX_RESULTS = 8

    USER_AGENT = (
        "Mozilla/5.0 (X11; Linux x86_64) "
        "AppleWebKit/537.36 (KHTML, like Gecko) "
        "Chrome/151.0 Safari/537.36"
    )

    def can_handle(self, message):
        text = str(message or "").strip().lower()
        if not text:
            return False

        explicit = (
            "search the web",
            "web search",
            "search online",
            "search the internet",
            "look it up online",
            "look online",
        )
        if any(text == p or text.startswith(p + " ") for p in explicit):
            return True

        # Current/fresh-information questions.
        patterns = (
            r"\bwhen\s+(?:is|was|will|does|did)\b",
            r"\bwhat(?:'s| is)\s+(?:the\s+)?latest\b",
            r"\bwho(?:'s| is)\s+(?:the\s+)?current\b",
            r"\bwhere\s+is\s+(?:the\s+)?current\b",
            r"\bhow\s+many\s+.*\b(?:today|currently|now)\b",
            r"\bcurrent\b",
            r"\blatest\b",
            r"\btoday\b",
            r"\bright\s+now\b",
            r"\brecent(?:ly)?\b",
            r"\bnews\b",
            r"\bthis\s+(?:year|month|week)\b",
        )
        return any(re.search(pattern, text) for pattern in patterns)

    def run(self, expression="", **kwargs):
        query = self._clean_query(expression)
        if not query:
            return "No search query provided."

        results = self.search_results(query)
        if not results:
            return f"No useful web results found for: {query}"

        return self._format_results(query, results)

    def search_results(self, query):
        query = self._clean_query(query)
        if not query:
            return []

        try:
            encoded = urllib.parse.quote_plus(query)
            url = self.SEARCH_URL.format(encoded)

            request = urllib.request.Request(
                url,
                headers={
                    "User-Agent": self.USER_AGENT,
                    "Accept": (
                        "text/html,application/xhtml+xml,"
                        "application/xml;q=0.9,*/*;q=0.8"
                    ),
                    "Accept-Language": "en-US,en;q=0.9",
                    "Connection": "close",
                },
            )

            with urllib.request.urlopen(request, timeout=self.TIMEOUT) as response:
                page = response.read().decode("utf-8", errors="ignore")

            return self._parse_results(page)

        except (TimeoutError, socket.timeout):
            return []
        except Exception:
            try:
                from core.logger import logger
                logger.exception("Web search failed")
            except Exception:
                pass
            return []

    @staticmethod
    def _clean_query(query):
        text = str(query or "").strip()
        prefixes = (
            "search the web for ",
            "search the web ",
            "web search for ",
            "web search ",
            "search online for ",
            "search online ",
            "search the internet for ",
            "search the internet ",
            "look it up online for ",
            "look it up online ",
            "look online for ",
            "look online ",
        )
        lowered = text.lower()
        for prefix in prefixes:
            if lowered.startswith(prefix):
                return text[len(prefix):].strip()
        return text

    def _parse_results(self, page):
        results = []

        blocks = re.findall(
            r'<div[^>]*class=["\']result\b[^>]*>(.*?)</div>\s*</div>',
            page,
            re.DOTALL | re.IGNORECASE,
        )

        if not blocks:
            blocks = re.findall(
                r'(<a[^>]*class=["\'][^"\']*result__a[^"\']*["\'][^>]*>.*?</a>)',
                page,
                re.DOTALL | re.IGNORECASE,
            )

        for block in blocks:
            if len(results) >= self.MAX_RESULTS:
                break
            parsed = self._parse_result_block(block)
            if parsed:
                results.append(parsed)

        return results

    def _parse_result_block(self, block):
        title_match = re.search(
            r'<a[^>]*class=["\'][^"\']*result__a[^"\']*["\'][^>]*'
            r'href=["\']([^"\']+)["\'][^>]*>(.*?)</a>',
            block,
            re.DOTALL | re.IGNORECASE,
        )
        if not title_match:
            return None

        title = self._clean_html(title_match.group(2))
        url = self._decode_url(title_match.group(1))
        if not title or not url:
            return None

        snippet_match = re.search(
            r'class=["\'][^"\']*result__snippet[^"\']*["\'][^>]*>(.*?)</',
            block,
            re.DOTALL | re.IGNORECASE,
        )
        snippet = (
            self._clean_html(snippet_match.group(1))
            if snippet_match
            else ""
        )

        return {
            "title": title,
            "url": url,
            "snippet": snippet,
        }

    def _format_results(self, query, results):
        lines = [f"Web results for: {query}", ""]
        for index, result in enumerate(results, 1):
            lines.append(f"{index}. {result['title']}")
            if result.get("snippet"):
                lines.append(f"   {result['snippet']}")
            lines.append(f"   {result['url']}")
            lines.append("")
        return "\n".join(lines).strip()

    @staticmethod
    def _decode_url(url):
        url = html.unescape(url or "")
        if "uddg=" in url:
            try:
                parsed = urllib.parse.urlparse(
                    "https:" + url if url.startswith("//") else url
                )
                params = urllib.parse.parse_qs(parsed.query)
                if params.get("uddg"):
                    return params["uddg"][0]
            except Exception:
                pass
        return url

    @staticmethod
    def _clean_html(text):
        text = html.unescape(text or "")
        text = re.sub(r"<[^>]+>", " ", text)
        text = re.sub(r"\s+", " ", text)
        return text.strip()

"""
Time Plugin
"""

from datetime import datetime

from plugins.base import AtlasPlugin


class TimePlugin(AtlasPlugin):

    name = "time"

    display_name = "Time"

    description = (
        "Returns the current time."
    )

    icon = "🕐"

    has_gui = False

    gui_id = "generic"

    triggers = [
        "what time is it",
        "what's the time",
        "what is the time",
        "tell me the time",
        "current time",
        "current time please",
        "time now",
        "what time",
        "time please",
        "time",
        "clock",
    ]

    def can_handle(
        self,
        message,
    ):

        text = (
            message
            .lower()
            .strip()
            .rstrip("?!.")
        )

        for trigger in self.triggers:

            if text == trigger:
                return True

            if text.startswith(
                trigger + " "
            ):
                return True

        return False

    def run(
        self,
        **kwargs,
    ):

        return datetime.now().strftime(
            "%I:%M %p"
        )
