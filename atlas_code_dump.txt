const chat = document.getElementById("chat");
const input = document.getElementById("message");
const sendButton = document.getElementById("send");


function addMessage(
    sender,
    text,
    type
) {

    const div =
        document.createElement("div");

    div.className =
        `message ${type}`;

    const senderElement =
        document.createElement("b");

    senderElement.textContent =
        `${sender}: `;

    const textElement =
        document.createElement("span");

    textElement.textContent =
        text;

    div.appendChild(
        senderElement
    );

    div.appendChild(
        textElement
    );

    chat.appendChild(
        div
    );

    chat.scrollTop =
        chat.scrollHeight;

    return textElement;
}


async function sendMessage() {

    const text =
        input.value.trim();

    if (!text)
        return;


    addMessage(
        "You",
        text,
        "user"
    );


    input.value = "";

    sendButton.disabled = true;
    input.disabled = true;


    const responseElement =
        addMessage(
            "Atlas",
            "",
            "atlas"
        );


    try {

        const response =
            await fetch(
                "/stream",
                {
                    method: "POST",

                    headers: {
                        "Content-Type":
                            "application/json",

                        "Accept":
                            "application/x-ndjson"
                    },

                    body: JSON.stringify({
                        message: text
                    })
                }
            );


        if (!response.ok) {

            throw new Error(
                `HTTP ${response.status}`
            );
        }


        if (!response.body) {

            throw new Error(
                "Streaming is unavailable."
            );
        }


        const reader =
            response.body.getReader();

        const decoder =
            new TextDecoder("utf-8");


        let buffer = "";


        while (true) {

            const {
                value,
                done
            } = await reader.read();


            if (done)
                break;


            buffer +=
                decoder.decode(
                    value,
                    {
                        stream: true
                    }
                );


            const lines =
                buffer.split("\n");


            buffer =
                lines.pop() || "";


            for (
                const line of lines
            ) {

                const trimmed =
                    line.trim();


                if (!trimmed)
                    continue;


                let data;


                try {

                    data =
                        JSON.parse(
                            trimmed
                        );

                } catch (error) {

                    console.error(
                        "Invalid stream data:",
                        trimmed
                    );

                    continue;
                }


                // ----------------------------------
                // Backend error
                // ----------------------------------

                if (data.error) {

                    throw new Error(
                        data.error
                    );
                }


                // ----------------------------------
                // Normal chunk
                // ----------------------------------

                if (
                    typeof data.chunk
                    === "string"
                ) {

                    responseElement
                        .textContent +=
                            data.chunk;

                    chat.scrollTop =
                        chat.scrollHeight;
                }
            }
        }


        // Process any final buffered line.

        const finalLine =
            buffer.trim();


        if (finalLine) {

            try {

                const data =
                    JSON.parse(
                        finalLine
                    );


                if (data.error) {

                    throw new Error(
                        data.error
                    );
                }


                if (
                    typeof data.chunk
                    === "string"
                ) {

                    responseElement
                        .textContent +=
                            data.chunk;
                }

            } catch (error) {

                if (
                    error instanceof Error
                    &&
                    error.message !==
                    "Unexpected end of JSON input"
                ) {

                    throw error;
                }
            }
        }


    } catch (error) {

        console.error(
            "Atlas request failed:",
            error
        );


        responseElement.textContent =
            `Error: ${error.message}`;

    } finally {

        sendButton.disabled = false;
        input.disabled = false;

        input.focus();
    }
}


sendButton.addEventListener(
    "click",
    sendMessage
);


input.addEventListener(
    "keydown",
    (event) => {

        if (
            event.key === "Enter"
            && !event.shiftKey
        ) {

            event.preventDefault();

            sendMessage();
        }
    }
);

from fastapi.staticfiles import StaticFiles
from fastapi.responses import FileResponse, StreamingResponse

logger = get_logger("web")


app = FastAPI(
    title="Atlas Web Interface",
    version="1.0",

    "/static",
    StaticFiles(
        directory="interfaces/web/static"
    ),
    name="static",

@app.on_event("startup")
def startup():

    logger.info(
        "Atlas Web Server started."
    )


@app.on_event("shutdown")
def shutdown():

    logger.info(
        "Atlas Web Server shutting down."
    )


@app.get("/")
def home():

    logger.info(
        "Web interface requested."
    )

    return FileResponse(
        "interfaces/web/static/index.html"
    )


class ChatRequest(BaseModel):
    message: str


@app.post("/chat")
def chat(request: ChatRequest):

    logger.info(
        "Chat request received."
    )

    try:

        response = controller.chat(
            request.message
        )

        logger.info(
            "Chat request completed."
        )

        return {
            "response": response
        }

    except Exception:

        logger.exception(
            "Chat request failed."
        )

        raise HTTPException(
            status_code=500,
            detail="Atlas encountered an internal error.",
        )


@app.post("/stream")
def stream(request: ChatRequest):

    logger.info(
        "Streaming request received."
    )

    def generate():

        try:

            for chunk in controller.stream_chat(
                request.message
            ):

                yield (
                    json.dumps(
                        {
                            "chunk": chunk
                        }
                    )
                    + "\n"
                )

            logger.info(
                "Streaming request completed."
            )

        except Exception:

            logger.exception(
                "Streaming request failed."
            )

            # Send an error event to the client
            yield (
                json.dumps(
                    {
                        "error":
                        "Atlas encountered an internal error."
                    }
                )
                + "\n"
            )

    return StreamingResponse(
        generate(),
        media_type="application/x-ndjson",
    )

# 🤖 Atlas AI V3

A sophisticated local conversational AI assistant with dual interfaces (Qt GUI & Web API), built with event-driven architecture, persistent memory, and extensible plugin system.

![Python](https://img.shields.io/badge/Python-3.8+-blue) ![License](https://img.shields.io/badge/License-MIT-green) ![Status](https://img.shields.io/badge/Status-Active-brightgreen)

## 🌟 Features

### Core Capabilities
- **Local AI Processing** - Runs completely offline using Ollama + Qwen2.5:1.5b model
- **Dual Interfaces** - Desktop GUI (Qt/PySide6) + Web API (FastAPI)
- **Hinglish Support** - Native support for Hindi + English conversations 🇮🇳
- **Persistent Memory** - Remembers user identity, preferences, and conversation history
- **Smart Intent Detection** - Routes messages to appropriate plugins or LLM
- **Conversation Management** - Auto-titles, history trimming, persistence
- **Streaming Responses** - Real-time response generation with chunk-by-chunk delivery
- **Plugin System** - Extensible architecture for custom tools (Calculator, Time, etc.)

### AI Processing Pipeline
1. **Memory Extraction** - Learns facts from natural language
2. **Intent Detection** - Identifies if message is a query, tool request, or memory recall
3. **Tool Execution** - Runs plugins (calculator, time, etc.) when applicable
4. **Context Building** - Combines user memory + conversation history + current message
5. **LLM Generation** - Streams response from local Ollama model
6. **Persistence** - Saves conversations and memory to disk

## 🚀 Quick Start

## 🧠 Architecture Patterns

### Event-Driven (Pub/Sub)
- Components publish events without knowing subscribers
- `EventBus` enables loose coupling
- Example: `ChunkReceived` → `AIService` → GUI

### Plugin/Strategy Pattern
- `AtlasPlugin` base class for extensibility
- Auto-discovery from `plugins/system/` directory
- Easy to add new tools without core changes

### Threading
- `AIWorker` runs on separate `QThread`
- Prevents GUI freezing during LLM inference
- Signal/slot communication (thread-safe)

### Dependency Injection
- Components receive dependencies via constructor
- `Controller` injects into widgets
- `EventBus` injected everywhere

## 🐛 Known Issues & Limitations

### Current Limitations
- **Single-user** - Not designed for multi-user deployment
- **JSON storage** - Doesn't scale beyond 10K conversations (use DB for production)
- **Memory search** - Pattern-based only (no semantic similarity)
- **Model quality** - Depends on Ollama's underlying training
- **Web API** - Lacks authentication and rate limiting

### Workarounds
- Use local network only (no remote access)
- Archive old conversations periodically
- Run multiple instances for different users
- Implement reverse proxy with auth for production

"""
Time Plugin
"""

from datetime import datetime

from plugins.base import AtlasPlugin


class TimePlugin(AtlasPlugin):

    name = "time"

    display_name = "Time"

    description = (
        "Returns the current time."
    )

    icon = "🕐"

    has_gui = False

    gui_id = "generic"

    triggers = [
        "what time is it",
        "what's the time",
        "what is the time",
        "tell me the time",
        "current time",
        "current time please",
        "time now",
        "what time",
        "time please",
        "time",
        "clock",
    ]

    def can_handle(
        self,
        message,
    ):

        text = (
            message
            .lower()
            .strip()
            .rstrip("?!.")
        )

        for trigger in self.triggers:

            if text == trigger:
                return True

            if text.startswith(
                trigger + " "
            ):
                return True

        return False

    def run(
        self,
        **kwargs,
    ):

        return datetime.now().strftime(
            "%I:%M %p"
        )

gui/window.py

Main Atlas Window.
"""

from PySide6.QtWidgets import (
    QHBoxLayout,
    QMainWindow,
    QWidget,
    QStackedWidget,
)

from gui.widgets.sidebar import AtlasSidebar
from gui.widgets.chat import AtlasChat
from gui.widgets.statusbar import AtlasStatusBar
from gui.widgets.tool_workspace import ToolWorkspace


class AtlasWindow(QMainWindow):

    def __init__(
        self,
        bus,
        state,
        controller,
    ):

        super().__init__()

        self.bus = bus
        self.state = state
        self.controller = controller

        self.setup_window()
        self.build_ui()

    # =====================================================
    # Window
    # =====================================================

    def setup_window(self):

        self.setWindowTitle(
            "Atlas AI"
        )

        self.resize(
            1400,
            900,
        )

        self.setMinimumSize(
            1100,
            700,
        )

    # =====================================================
    # UI
    # =====================================================

    def build_ui(self):

        root = QWidget()

        root.setObjectName(
            "AtlasWindow"
        )

        self.setCentralWidget(
            root
        )

        layout = QHBoxLayout(
            root
        )

        layout.setContentsMargins(

from PySide6.QtCore import Qt, QTimer

from core.events import (

from gui.widgets.inputbar import AtlasInputBar

class AtlasChat(QWidget):

    def __init__(
        self,
        bus,
        parent=None,
    ):

        super().__init__(parent)

        self.bus = bus

        # =================================================
        # Current AI Response
        # =================================================

        self.current_ai_message = None
        self.current_ai_response = ""

        # =================================================
        # Thinking Animation
        # =================================================

        self.thinking_frame = 0

        self.thinking_timer = QTimer(self)

        self.thinking_timer.setInterval(

        self.thinking_timer.timeout.connect(

        )

        self.scroll.setFrameShape(

        )

        )

        )

        layout.addWidget(

        )

    # =====================================================
    # Messages
    # =====================================================

    def add_message(
        self,
        sender,
        text,
    ):

        message = AtlasMessage(
            sender,
            text,
        )

        self.messages_layout.insertWidget(
            self.messages_layout.count() - 1,
            message,
        )

        return message

    # =====================================================
    # Clear Messages
    # =====================================================

    def clear_messages(self):

        self.stop_thinking_animation()

        while self.messages_layout.count() > 1:

            item = self.messages_layout.takeAt(
                0
            )

            widget = item.widget()

            if widget is not None:

                widget.deleteLater()

        self.current_ai_message = None
        self.current_ai_response = ""

    # =====================================================
    # Load Conversation
    # =====================================================

    def load_conversation(
        self,
        conversation,
    ):

        self.stop_thinking_animation()

        self.clear_messages()

        for message in conversation.history():

            role = message.get(
                "role"
            )

            text = message.get(
                "content",
                "",
            )

            if role == "user":

                sender = "User"

            else:

                sender = "Atlas"

            self.add_message(
                sender,
                text,
            )

        self.scroll_to_bottom()

    # =====================================================
    # Input Handling
    # =====================================================

    def handle_message(
        self,
        text,
    ):

        self.stop_thinking_animation()

        self.current_ai_message = None
        self.current_ai_response = ""

        # ---------------------------------
        # User message
        # ---------------------------------

        self.add_message(
            "User",
            text,
        )

        # ---------------------------------
        # Atlas thinking message
        # ---------------------------------

        self.current_ai_message = self.add_message(
            "Atlas",
            "· · ·",
        )

        self.current_ai_message.set_thinking(
            True
        )

        self.thinking_frame = 0

        self.thinking_timer.start()

        self.scroll_to_bottom()

        # ---------------------------------
        # Backend
        # ---------------------------------

        self.bus.publish(
            MessageReceived(
                text
            )
        )

    # =====================================================
    # Thinking Animation
    # =====================================================

    def animate_thinking(self):

        if self.current_ai_message is None:

            self.stop_thinking_animation()

            return

        frames = (
            "● · ·",
            "● ● ·",
            "● ● ●",
            "· ● ●",
            "· · ●",
            "· ● ●",
            "● ● ●",
            "● ● ·",
            "● · ·",
            "· · ·",
        )

        frame = frames[
            self.thinking_frame
            % len(frames)
        ]

        self.current_ai_message.set_text(
            frame
        )

        self.thinking_frame += 1

        self.scroll_to_bottom()

    # =====================================================
    # Stop Thinking Animation
    # =====================================================

    def stop_thinking_animation(self):

        if self.thinking_timer.isActive():

            self.thinking_timer.stop()

        self.thinking_frame = 0

    # =====================================================
    # AI Response Handling
    # =====================================================

    def handle_response(
        self,
        event,
    ):

        if self.current_ai_message is None:

            self.current_ai_message = self.add_message(
                "Atlas",
                "",
            )

            self.current_ai_response = ""

        # ---------------------------------
        # Stop thinking animation
        # ---------------------------------

        if self.thinking_timer.isActive():

            self.stop_thinking_animation()

            self.current_ai_message.set_thinking(
                False
            )

        # ---------------------------------
        # Accumulate response
        # ---------------------------------

        self.current_ai_response += event.text

        # ---------------------------------
        # Display response
        # ---------------------------------

        self.current_ai_message.set_text(
            self.current_ai_response
        )

        self.scroll_to_bottom()

    # =====================================================
    # Scrolling
    # =====================================================

    def scroll_to_bottom(self):

        scrollbar = (
            self.scroll.verticalScrollBar()
        )

        scrollbar.setValue(
            scrollbar.maximum()
        )

    # ======================================================
    # Button Handling
    # ======================================================

    def button_pressed(
        self,
        value,
    ):

        # --------------------------------------------------
        # Clear
        # --------------------------------------------------

        if value == "C":

            self.display.clear()

            self.current_expression = ""
            self.current_result = ""

            self.explanation.clear()

            return

        # --------------------------------------------------
        # Equals
        # --------------------------------------------------

        if value == "=":

            self.calculate()

            return

        # --------------------------------------------------
        # Power
        # --------------------------------------------------

        if value == "^":

            value = "^"

        self.display.insert(
            value
        )

    # ======================================================
    # Calculate
    # ======================================================

    def calculate(self):

        expression = (
            self.display.text()
            .strip()
        )

        if not expression:

            return

        try:

            result = self.math.calculate(
                expression
            )

            result_text = self.math.format(
                result
            )

            # ==================================================
            # Replace active context
            # ==================================================

            self.current_expression = (
                expression
            )

            self.current_result = (
                result_text
            )

            # ==================================================
            # Keep expression visible
            # ==================================================

            self.display.setText(
                expression
            )

            # ==================================================
            # Display result
            # ==================================================

            self.explanation.setMarkdown(
                "### Result\n\n"
                f"**Expression**\n\n"
                f"`{expression}`\n\n"
                "**Answer**\n\n"
                f"**{result_text}**\n\n"
                "Ask Atlas Math to explain it."
            )

        except Exception as error:

            self.current_expression = ""
            self.current_result = ""

            self.explanation.setMarkdown(
                "### Calculation Error\n\n"
                f"`{error}`"
            )

    # ======================================================
    # Ask Atlas Math
    # ======================================================

    def ask_atlas(self):

        question = (
            self.question.text()
            .strip()
        )

        if not question:

            return

        expression = (
            self.current_expression
            .strip()
        )

        result = (
            self.current_result
            .strip()
        )

        # ==================================================
        # No existing calculation
        # ==================================================

        if not expression:

            # --------------------------------------------------
            # Try to extract an expression directly from
            # the user's question.
            # --------------------------------------------------

            extracted = (
                self.math.extract_expression(
                    question
                )
            )

            if extracted:

                expression = extracted

            else:

                self.explanation.setMarkdown(
                    "### No calculation\n\n"
                    "Enter a calculation first, "
                    "or ask a mathematical question "
                    "containing an expression."
                )

                return

        # ==================================================
        # Disable UI while calculating
        #
        # This is extremely fast because it is SymPy,
        # not Qwen.
        # ==================================================

        self.ask_button.setEnabled(
            False
        )

        self.question.setEnabled(
            False
        )

        self.explanation.setMarkdown(
            "### Atlas Math\n\n"
            "Calculating..."
        )

        try:

            response = self.math.explain(
                question,
                expression,
                result,
            )

            # --------------------------------------------------
            # If the question contained a NEW expression,
            # update the active state.
            # --------------------------------------------------

            new_expression = (
                self.math.extract_expression(
                    question,
                    fallback="",
                )
            )

            if new_expression:

                self.current_expression = (
                    new_expression
                )

                try:

                    new_result = (
                        self.math.calculate(
                            new_expression
                        )
                    )

                    self.current_result = (
                        self.math.format(
                            new_result
                        )
                    )

                except Exception:

                    pass

            self.explanation.setMarkdown(
                response
            )

        except Exception as error:

            self.explanation.setMarkdown(
                "### Math Error\n\n"
                f"{error}"
            )

        finally:

            self.ask_button.setEnabled(
                True
            )

            self.question.setEnabled(
                True
            )

            self.question.clear()

            self.question.setFocus()

    # =====================================================
    # Markdown / Link Rendering
    # =====================================================

    @staticmethod
    def markdown_to_html(
        text,
    ):

        if not text:
            return ""

        # ---------------------------------
        # Escape HTML
        # ---------------------------------

        text = html.escape(
            text
        )

        # ---------------------------------
        # Markdown links
        #
        # [Ubuntu](https://ubuntu.com)
        # ---------------------------------

        text = re.sub(
            r'\[([^\]]+)\]\((https?://[^\s\)]+)\)',
            r'<a href="\2">\1</a>',
            text,
            flags=re.IGNORECASE,
        )

        # ---------------------------------
        # Plain URLs
        # ---------------------------------

        parts = re.split(
            r'(<a\s+href="[^"]+">.*?</a>)',
            text,
            flags=re.IGNORECASE | re.DOTALL,
        )

        for index in range(
            0,
            len(parts),
            2,
        ):

            parts[index] = re.sub(
                r'(?<!["=])(https?://[^\s<]+)',
                r'<a href="\1">\1</a>',
                parts[index],
                flags=re.IGNORECASE,
            )

        text = "".join(
            parts
        )

        # ---------------------------------
        # Bold
        # ---------------------------------

        text = re.sub(
            r'\*\*(.+?)\*\*',
            r'<strong>\1</strong>',
            text,
            flags=re.DOTALL,
        )

        # ---------------------------------
        # Inline code
        # ---------------------------------

        text = re.sub(
            r'`([^`]+)`',
            r'<code>\1</code>',
            text,
        )

        # ---------------------------------
        # New lines
        # ---------------------------------

        text = text.replace(
            "\n",
            "<br>",
        )

        return text

"""
Atlas V3 Conversation Service

Handles conversation creation, loading,
saving, renaming, deletion, and state.

from core.conversation import ConversationManager
from core.conversation_store import ConversationStore


class ConversationService:

    def __init__(self):

        self.conversation = ConversationManager()
        self.store = ConversationStore()

    # ==================================================
    # Active Conversation
    # ==================================================

    def get(self):

        return self.conversation

    def get_id(self):

        return self.conversation.get_id()

    def get_title(self):

        return self.conversation.get_title()

    def history(self):

        return self.conversation.history()

    def last(self):

        return self.conversation.last()

    # ==================================================
    # Conversation Lifecycle
    # ==================================================

    def clear(self):

        self.conversation = ConversationManager()

    def new(self):

        self.save()

        self.conversation = ConversationManager()

        return self.get_id()

    def load(
        self,
        conversation_id,
    ):

        self.save()

        data = self.store.load(
            conversation_id
        )

        if data is None:
            return False

        if isinstance(
            data,
            ConversationManager,
        ):

            self.conversation = data

        elif isinstance(
            data,
            dict,
        ):

            conversation = ConversationManager()

            conversation.load(
                data
            )

            self.conversation = conversation

        else:

            return False

        return True

    # ==================================================
    # Conversation List
    # ==================================================

    def list(self):

        return self.store.list_conversations()

    def delete(
        self,
        conversation_id,
    ):

        deleted = self.store.delete(
            conversation_id
        )

        if not deleted:
            return False

        if (
            conversation_id
            == self.conversation.get_id()
        ):

            self.conversation = (
                ConversationManager()
            )

        return True

    def rename(
        self,
        conversation_id,
        title,
    ):

        title = title.strip()

        if not title:
            return False

        data = self.store.load(
            conversation_id
        )

        if data is None:
            return False

        if isinstance(
            data,
            ConversationManager,
        ):

            conversation = data

        elif isinstance(
            data,
            dict,
        ):

            conversation = ConversationManager()

            conversation.load(
                data
            )

        else:

            return False

        conversation.set_title(
            title
        )

        self.store.save(
            conversation
        )

        if (
            conversation_id
            == self.conversation.get_id()
        ):

            self.conversation.set_title(
                title
            )

        return True

    # ==================================================
    # Messages
    # ==================================================

    def add_user(
        self,
        message,
    ):

        self.conversation.add_user(
            message
        )

    def add_assistant(
        self,
        message,
    ):

        self.conversation.add_assistant(
            message
        )

    # ==================================================
    # Title
    # ==================================================

    def auto_title(
        self,
        message,
    ):

        if (
            self.conversation.get_title()
            != "New Chat"
        ):

            return

        text = " ".join(
            message.strip().split()
        )

        if not text:
            return

        words = text.split()

        if len(words) > 7:

            title = (
                " ".join(words[:7])
                + "..."
            )

        else:

            title = text

        self.conversation.set_title(
            title
        )

    # ==================================================
    # Persistence
    # ==================================================

    def save(self):

        if not self.conversation.history():

            return

        self.store.save(
            self.conversation
        )

"""
Atlas V3 Memory Service

Handles memory extraction and recall.
"""

import re

from ai.memory_patterns import PATTERNS
from core.memory import MemoryManager


class MemoryService:

    def __init__(self, memory=None):

        self.memory = memory or MemoryManager()

    # ==================================================
    # Memory Extraction
    # ==================================================

    def extract(self, message):

        for pattern, category, key in PATTERNS:

            match = re.search(
                pattern,
                message,
                re.IGNORECASE,
            )

            if not match:
                continue

            value = match.group(1).strip()

            if not value:
                continue

            self.memory.remember(
                category,
                key,
                value,
            )

            return True

        return False

    # ==================================================
    # Memory Recall
    # ==================================================

    def recall(
        self,
        category,
        key,
    ):

        return self.memory.recall(
            category,
            key,
        )

    # ==================================================
    # Natural Language Recall
    # ==================================================

    def try_recall(self, message):

        text = message.lower().strip()

        recall = {

            "what's my name": (
                "identity",
                "name",
            ),

            "what is my name": (
                "identity",
                "name",
            ),

            "who am i": (
                "identity",
                "name",
            ),

            "where do i live": (
                "identity",
                "city",
            ),

            "what city do i live in": (
                "identity",
                "city",
            ),
        }

        for trigger, memory in recall.items():

            if trigger not in text:
                continue

            value = self.recall(
                memory[0],
                memory[1],
            )

            if value:

                return value

            return "I don't know."

        return None

"""
Atlas V3 Brain

Coordinates memory, context,
conversation, tools, knowledge routing,
and the local model.
"""

from ai.model import AtlasModel
from ai.context_builder import ContextBuilder
from ai.planner import Planner

from ai.knowledge_decider import (
    KnowledgeDecider,
    Decision,
)

from ai.services.conversation_service import (
    ConversationService,
)

from ai.services.memory_service import (
    MemoryService,
)

from core.tool_manager import ToolManager
from core.intent import IntentRouter


class AtlasBrain:

    def __init__(self):

        self.model = AtlasModel()

        self.memory = MemoryService()

        self.tools = ToolManager()

        self.context = ContextBuilder()

        self.conversations = ConversationService()

        self.router = IntentRouter(
            self.tools.plugins.plugins
        )

        self.planner = Planner(
            self.router
        )

        # =================================================
        # Knowledge Routing
        # =================================================

        self.knowledge_decider = KnowledgeDecider()

    # ==================================================
    # Conversation
    # ==================================================

    def clear_conversation(self):

        self.conversations.clear()

    def new_conversation(self):

        return self.conversations.new()

    def load_conversation(
        self,
        conversation_id,
    ):

        return self.conversations.load(
            conversation_id
        )

    def list_conversations(self):

        return self.conversations.list()

    def delete_conversation(
        self,
        conversation_id,
    ):

        return self.conversations.delete(
            conversation_id
        )

    def rename_conversation(
        self,
        conversation_id,
        title,
    ):

        return self.conversations.rename(
            conversation_id,
            title,
        )

    def get_active_conversation_id(self):

        return self.conversations.get_id()

    # ==================================================
    # Processing
    # ==================================================

    def process(
        self,
        message,
    ):

        message = message.strip()

        if not message:
            return

        # =================================================
        # Conversation
        # =================================================

        self.conversations.add_user(
            message
        )

        # =================================================
        # Automatic Conversation Title
        # =================================================

        self.conversations.auto_title(
            message
        )

        # =================================================
        # Memory Extraction
        # =================================================

        self.memory.extract(
            message
        )

        # =================================================
        # Memory Recall
        # =================================================

        memory_response = (
            self.memory.try_recall(
                message
            )
        )

        if memory_response is not None:

            self.conversations.add_assistant(
                memory_response
            )

            self.conversations.save()

            yield memory_response

            return

        # =================================================
        # Knowledge Decision
        # =================================================

        decision = (
            self.knowledge_decider.decide(
                message
            )
        )

        print(
            "[ATLAS] Knowledge decision: "
            f"{decision.decision.value} | "
            f"{decision.reason} | "
            f"{decision.confidence:.2f}"
        )

        # =================================================
        # WEB
        # =================================================

        if decision.decision == Decision.WEB:

            print(
                "[ATLAS] Running WEB search..."
            )

            web_result = self._web_search(
                message
            )

            print(
                "[ATLAS] Web result length: "
                f"{len(web_result)}"
            )

            context = self._build_web_prompt(
                message,
                web_result,
            )

            yield from self._generate_and_save(
                context
            )

            return

        # =================================================
        # BOTH
        # =================================================

        if decision.decision == Decision.BOTH:

            print(
                "[ATLAS] Running BOTH search..."
            )

            web_result = self._web_search(
                message
            )

            print(
                "[ATLAS] Web result length: "
                f"{len(web_result)}"
            )

            context = self._build_web_prompt(
                message,
                web_result,
                both=True,
            )

            yield from self._generate_and_save(
                context
            )

            return

        # =================================================
        # LOCAL
        # =================================================

        print(
            "[ATLAS] Using LOCAL model..."
        )

        # =================================================
        # Existing Tool / Planner System
        # =================================================

        plan = self.planner.plan(
            message
        )

        if plan.action == Planner.TOOL:

            plugin = plan.plugin

            result = self.tools.execute(
                plugin.name,
                expression=plan.arguments,
            )

            self.conversations.add_assistant(
                result
            )

            self.conversations.save()

            yield result

            return

        # =================================================
        # Local Model
        # =================================================

        context = self.context.build(
            self.conversations.last(),
            message,
        )

        yield from self._generate_and_save(
            context
        )

    # ==================================================
    # Web Search
    # ==================================================

    def _web_search(
        self,
        question,
    ):

        try:

            result = self.tools.execute(
                "web_search",
                expression=question,
            )

            if not result:

                return (
                    "[WEB_SEARCH_FAILED]\n"
                    "The web search returned no data."
                )

            return result

        except Exception as error:

            print(
                "[ATLAS] Web search error: "
                f"{error}"
            )

            return (
                "[WEB_SEARCH_FAILED]\n"
                f"{error}"
            )

    # ==================================================
    # Web Prompt
    # ==================================================

    def _build_web_prompt(
        self,
        question,
        web_result,
        both=False,
    ):

        search_failed = (
            web_result.startswith(
                "[WEB_SEARCH_FAILED]"
            )
        )

        if search_failed:

            instruction = """
The web search failed.

DO NOT pretend that you received search results.

DO NOT invent current information.

Answer only if you can safely answer from
your existing knowledge. Otherwise clearly
tell the user that the web search failed.
"""

        else:

            instruction = """
The information below was retrieved from
an external web search.

IMPORTANT:

The web results are the PRIMARY SOURCE for
current or changing information.

Use the retrieved evidence instead of relying
on your pretrained memory for current facts.

For every factual claim about the current
state of something, prefer information that
appears in the search results.

DO NOT invent facts that are not supported
by the retrieved information.

DO NOT combine an old fact from your training
data with a current fact unless the results
support that combination.

If the results conflict with your memory,
trust the retrieved evidence.

If the results do not contain enough information,
say:

"The search results did not provide enough
reliable information to answer that."

Do not guess.

If a source provides a specific version,
date, number, name, or other factual value,
preserve it accurately.

If a source provides a URL, preserve the URL
when useful.

Do not claim that a web search was performed
unless web results are actually present.
"""

        if both:

            instruction += """

After using the web evidence, you may apply
your reasoning to explain, compare, or analyze
the information.

Keep a clear distinction between:

1. Facts retrieved from the web.
2. Your reasoning based on those facts.
"""

        return f"""
You are Atlas.

==================================================
USER QUESTION
==================================================

{question}

==================================================
WEB SEARCH EVIDENCE
==================================================

{web_result}

==================================================
INSTRUCTIONS
==================================================

{instruction}

==================================================
ANSWER
==================================================

Answer the user's original question directly.

Be concise.

Do not talk about:

- KnowledgeDecider
- internal routing
- plugins
- prompt construction
- hidden reasoning
- internal architecture

If the web evidence is insufficient,
be honest rather than guessing.
""".strip()

    # ==================================================
    # Generate + Save
    # ==================================================

    def _generate_and_save(
        self,
        context,
    ):

        reply = ""

        for chunk in self.model.generate(
            context
        ):

            reply += chunk

            yield chunk

        if reply.strip():

            self.conversations.add_assistant(
                reply
            )

            self.conversations.save()

    # ============================================================
    # Public API
    # ============================================================

    def decide(
        self,
        question,
    ):

        question = question.strip()

        if not question:

            return DecisionResult(
                Decision.LOCAL,
                "Empty question.",
                1.0,
            )

        text = question.lower()

        # ====================================================
        # 1. Capability questions
        # ====================================================
        #
        # These must happen BEFORE web detection.
        #
        # "Can you search the web?"
        # should NOT actually perform a search.
        #
        # It is asking about Atlas's capability.
        #

        if self._matches(
            text,
            self.capability_patterns,
        ):

            return DecisionResult(
                Decision.LOCAL,
                "User is asking about Atlas capabilities.",
                0.99,
            )

        # ====================================================
        # 2. Explicit web request
        # ====================================================

        if self._matches(
            text,
            self.explicit_web_patterns,
        ):

            return DecisionResult(
                Decision.WEB,
                "User explicitly requested external information.",
                0.99,
            )

        # ====================================================
        # 3. Current information
        # ====================================================

        is_current = self._matches(
            text,
            self.web_patterns,
        )

        has_reasoning = self._matches(
            text,
            self.reasoning_patterns,
        )

        # Current + reasoning = BOTH

        if is_current and has_reasoning:

            return DecisionResult(
                Decision.BOTH,
                "Question requires current information plus reasoning.",
                0.95,
            )

        # Pure current fact = WEB

        if is_current:

            return DecisionResult(
                Decision.WEB,
                "Question depends on current or changing information.",
                0.98,
            )

        # ====================================================
        # 4. New / recent technical knowledge
        # ====================================================

        if self._matches(
            text,
            self.both_patterns,
        ):

            return DecisionResult(
                Decision.BOTH,
                "Question may require current external knowledge plus local reasoning.",
                0.90,
            )

        # ====================================================
        # 5. Current-event entities
        # ====================================================

        if self._matches(
            text,
            self.current_entity_patterns,
        ):

            if has_reasoning:

                return DecisionResult(
                    Decision.BOTH,
                    "Question involves a potentially changing topic and requires reasoning.",
                    0.88,
                )

            return DecisionResult(
                Decision.WEB,
                "Question involves a topic whose information may change over time.",
                0.90,
            )

        # ====================================================
        # 6. Mathematical / computational questions
        # ====================================================

        if self._matches(
            text,
            self.calculation_patterns,
        ):

            return DecisionResult(
                Decision.LOCAL,
                "Question is computational and does not require current information.",
                0.98,
            )

        # ====================================================
        # 7. Established knowledge
        # ====================================================

        if self._matches(
            text,
            self.local_patterns,
        ):

            return DecisionResult(
                Decision.LOCAL,
                "Question concerns established knowledge suitable for local reasoning.",
                0.90,
            )

        # ====================================================
        # 8. Default
        # ====================================================

        return DecisionResult(
            Decision.LOCAL,
            "No indication that current external information is required.",
            0.70,
        )

    # ============================================================
    # Helper
    # ============================================================

    @staticmethod
    def _matches(
        text,
        patterns,
    ):

        return any(
            re.search(
                pattern,
                text,
            )
            for pattern in patterns
        )

    @staticmethod
    def contains_devanagari(text: str) -> bool:
        """Check if text contains Devanagari script."""
        return any(
            ord(char) in LanguageTranslator.DEVANAGARI_RANGE
            for char in text
        )

    @staticmethod
    def contains_hinglish_words(text: str) -> bool:
        """Check if text contains common Hinglish words."""
        text_lower = text.lower()
        # Split into words and check if any are Hinglish indicators
        words = re.findall(r'\b\w+\b', text_lower)
        return any(word in LanguageTranslator.HINGLISH_INDICATORS for word in words)

    @staticmethod
    def is_hinglish(text: str) -> bool:
        """
        Detect if text is Hinglish (mix of English and Hindi).
        Checks for:
        1. Devanagari script
        2. Common Hinglish words
        """
        has_devanagari = LanguageTranslator.contains_devanagari(text)
        has_hinglish_words = LanguageTranslator.contains_hinglish_words(text)
        
        return has_devanagari or has_hinglish_words

    @staticmethod
    def transliterate_devanagari(text: str) -> str:
        """
        Basic Devanagari to Latin transliteration.
        Handles pure Hindi text conversion.
        """
        result = text
        for devanagari, latin in LanguageTranslator.DEVANAGARI_TO_LATIN.items():
            result = result.replace(devanagari, latin)
        return result

    @staticmethod
    def apply_hinglish_rules(text: str) -> str:
        """
        Apply common Hinglish-to-English translation rules.
        Case-insensitive pattern matching.
        """
        result = text
        
        # Apply transliteration patterns (case-insensitive)
        for pattern, replacement in LanguageTranslator.HINGLISH_TO_ENGLISH.items():
            compiled = re.compile(pattern, re.IGNORECASE)
            result = compiled.sub(replacement, result)
        
        return result

    @staticmethod
    def normalize_hinglish(text: str) -> str:
        """
        Normalize Hinglish text by:
        1. Transliterating Devanagari to Latin
        2. Applying Hinglish translation rules
        3. Cleaning up extra spaces
        """
        # Step 1: Transliterate Devanagari if present
        if LanguageTranslator.contains_devanagari(text):
            text = LanguageTranslator.transliterate_devanagari(text)
        
        # Step 2: Apply Hinglish rules
        text = LanguageTranslator.apply_hinglish_rules(text)
        
        # Step 3: Clean up extra spaces
        text = re.sub(r'\s+', ' ', text).strip()
        
        return text

    @staticmethod
    def detect_and_convert(text: str) -> Tuple[str, bool, str]:
        """
        Detect language and convert if needed.
        
        Returns:
            Tuple of (converted_text, was_hinglish, original_language)
        """
        is_hinglish = LanguageTranslator.is_hinglish(text)
        
        if is_hinglish:
            converted = LanguageTranslator.normalize_hinglish(text)
            return converted, True, "hinglish"
        
        return text, False, "english"

    @staticmethod
    def add_hinglish_context(message: str) -> str:
        """
        Add Hinglish to the original message if detected.
        Helps preserve intent in the context.
        """
        if LanguageTranslator.is_hinglish(message):
            return f"[User is speaking in Hinglish/Hindi]\n{message}"
        return message


# Convenience functions
def is_hinglish(text: str) -> bool:
    """Quick check if text is Hinglish."""
    return LanguageTranslator.is_hinglish(text)


def normalize_hinglish(text: str) -> str:
    """Normalize Hinglish to English."""
    return LanguageTranslator.normalize_hinglish(text)


def detect_language(text: str) -> str:
    """Detect if text is Hinglish or English."""
    _, is_hin, lang = LanguageTranslator.detect_and_convert(text)
    return lang

"""
Atlas V3 Context Builder

Builds the minimum useful context for Atlas.
"""

from core.memory import MemoryManager


class ContextBuilder:

    def __init__(self):

        self.memory = MemoryManager()

    # ==================================================
    # Build Context
    # ==================================================

    def build(
        self,
        history,
        message,
        is_hinglish=False,
    ):

        message = message.strip()

        # --------------------------------------------------
        # Simple messages
        # --------------------------------------------------

        simple_messages = {
            "hi",
            "hello",
            "hey",
            "hii",
            "hiii",
            "yo",
            "sup",
            "hello atlas",
            "hey atlas",
            "hi atlas",
        }

        if message.lower() in simple_messages:

            return message

        parts = []

        # --------------------------------------------------
        # Language
        # --------------------------------------------------

        if is_hinglish:

            parts.append(
                "The user may be speaking Hinglish or Hindi. "
                "Respond naturally in the same style."
            )

        # --------------------------------------------------
        # Memory
        # --------------------------------------------------

        memory = self.memory.all()

        if memory:

            parts.append(
                "=== RELEVANT USER MEMORY ==="
            )

            for category, values in memory.items():

                parts.append(
                    f"[{category}]"
                )

                if isinstance(
                    values,
                    dict,
                ):

                    for key, value in values.items():

                        parts.append(
                            f"{key}: {value}"
                        )

                else:

                    parts.append(
                        str(values)
                    )

        # --------------------------------------------------
        # Recent conversation
        # --------------------------------------------------

        if history:

            # Only keep the most recent messages.
            recent_history = history[-6:]

            parts.append(
                "\n=== RECENT CONVERSATION ==="
            )

            for item in recent_history:

                role = item.get(
                    "role",
                    "user",
                ).capitalize()

                content = item.get(
                    "content",
                    "",
                )

                parts.append(
                    f"{role}: {content}"
                )

        # --------------------------------------------------
        # Current message
        # --------------------------------------------------

        parts.append(
            "\n=== USER MESSAGE ==="
        )

        parts.append(
            message
        )

        return "\n".join(parts)

"""
Atlas Planner

Decides what Atlas should do with a user request.
"""

from dataclasses import dataclass


@dataclass
class Plan:

    action: str

    plugin: object = None

    arguments: str = ""


class Planner:

    DIRECT = "direct"

    TOOL = "tool"

    def __init__(
        self,
        router,
    ):

        self.router = router

    def plan(
        self,
        message,
    ):

        plugin = self.router.match(
            message
        )

        if plugin:

            arguments = (
                self.router.extract_arguments(
                    plugin,
                    message,
                )
            )

            return Plan(
                action=self.TOOL,
                plugin=plugin,
                arguments=arguments,
            )

        return Plan(
            action=self.DIRECT
        )

    def add_user(self, text):
        self.messages.append(
            {
                "role": "user",
                "content": text,
            }
        )

        self._trim()

    def add_assistant(self, text):
        self.messages.append(
            {
                "role": "assistant",
                "content": text,
            }
        )

        self._trim()

    # --------------------------------------------------
    # Retrieval
    # --------------------------------------------------

    def history(self):
        return self.messages.copy()

    def last(self, amount=None):
        if amount is None:
            amount = MAX_HISTORY

        return self.messages[-amount:]

    # --------------------------------------------------
    # Maintenance
    # --------------------------------------------------

    def clear(self):
        self.messages.clear()

    # --------------------------------------------------
    # Internal
    # --------------------------------------------------

    def _trim(self):
        if len(self.messages) > MAX_HISTORY:
            self.messages = self.messages[-MAX_HISTORY:]

"""

Lightweight web search for Atlas.
"""

import html
import re
import urllib.parse
import urllib.request
import socket

from plugins.base import AtlasPlugin


class WebSearchPlugin(AtlasPlugin):

    name = "web_search"

    description = (
        "Searches the web for current information."
    )

    triggers = [
        "search the web",
        "web search",
        "search online",
    ]

    # ==================================================
    # Plugin Matching
    # ==================================================

    def can_handle(
        self,
        message,
    ):

        text = message.lower().strip()

        for trigger in self.triggers:

            if text == trigger:
                return True

            if text.startswith(
                trigger + " "
            ):
                return True

        return False

    # ==================================================
    # Search
    # ==================================================

    def run(
        self,
        expression="",
        **kwargs,
    ):

        query = expression.strip()

        if not query:
            return "No search query provided."

        try:

            encoded_query = (
                urllib.parse.quote_plus(
                    query
                )
            )

            url = (
                "https://html.duckduckgo.com/html/?q="
                + encoded_query
            )

            request = urllib.request.Request(
                url,
                headers={
                    "User-Agent":
                        "Mozilla/5.0 "
                        "(X11; Linux x86_64) "
                        "AppleWebKit/537.36 "
                        "(KHTML, like Gecko) "
                        "Chrome/151.0 Safari/537.36"
                },
            )

            # ------------------------------------------
            # Network timeout
            # ------------------------------------------

            with urllib.request.urlopen(
                request,
                timeout=5,
            ) as response:

                page = response.read().decode(
                    "utf-8",
                    errors="ignore",
                )

            results = self._parse_results(
                page
            )

            if not results:

                return (
                    "No useful web results found."
                )

            output = []

            for index, result in enumerate(
                results[:5],
                start=1,
            ):

                output.append(
                    f"{index}. {result['title']}\n"
                    f"   {result['snippet']}\n"
                    f"   {result['url']}"
                )

            return "\n\n".join(output)

        except (
            TimeoutError,
            socket.timeout,
        ):

            return (
                "Web search timed out."
            )

        except Exception as error:

            return (
                "Web search failed: "
                + str(error)
            )

    # ==================================================
    # Parse Results
    # ==================================================

    def _parse_results(
        self,
        page,
    ):

        results = []

        blocks = re.findall(
            r'<div[^>]+class="result[^"]*"'
            r'[^>]*>(.*?)</div>\s*</div>',
            page,
            re.DOTALL | re.IGNORECASE,
        )

        if not blocks:

            blocks = re.findall(
                r'(<a[^>]+class="result__a".*?</a>)',
                page,
                re.DOTALL | re.IGNORECASE,
            )

        for block in blocks:

            # ------------------------------------------
            # Title
            # ------------------------------------------

            title_match = re.search(
                r'class="result__a"[^>]*>'
                r'(.*?)'
                r'</a>',
                block,
                re.DOTALL | re.IGNORECASE,
            )

            if not title_match:
                continue

            title = self._clean_html(
                title_match.group(1)
            )

            # ------------------------------------------
            # URL
            # ------------------------------------------

            url_match = re.search(
                r'class="result__a"'
                r'[^>]+href="([^"]+)"',
                block,
                re.IGNORECASE,
            )

            if not url_match:
                continue

            raw_url = html.unescape(
                url_match.group(1)
            )

            real_url = self._decode_url(
                raw_url
            )

            # ------------------------------------------
            # Snippet
            # ------------------------------------------

            snippet_match = re.search(
                r'class="result__snippet"'
                r'[^>]*>'
                r'(.*?)'
                r'</',
                block,
                re.DOTALL | re.IGNORECASE,
            )

            if snippet_match:

                snippet = self._clean_html(
                    snippet_match.group(1)
                )

            else:

                snippet = (
                    "No description available."
                )

            results.append(
                {
                    "title": title,
                    "snippet": snippet,
                    "url": real_url,
                }
            )

        return results

    # ==================================================
    # Decode DuckDuckGo Redirect
    # ==================================================

    @staticmethod
    def _decode_url(
        url,
    ):

        if "uddg=" in url:

            parsed = urllib.parse.urlparse(
                "https:" + url
                if url.startswith("//")
                else url
            )

            query = urllib.parse.parse_qs(
                parsed.query
            )

            if "uddg" in query:

                return query["uddg"][0]

        return url

    # ==================================================
    # Clean HTML
    # ==================================================

    @staticmethod
    def _clean_html(
        text,
    ):

        text = html.unescape(
            text
        )

        text = re.sub(
            r"<[^>]+>",
            "",
            text,
        )

        text = re.sub(
            r"\s+",
            " ",
            text,
        )

        return text.strip()

"""
Atlas Math Engine

Fast local mathematics engine used by Atlas tools.

This module performs mathematical operations locally
instead of sending simple mathematical questions to
the LLM.

Supported:
- arithmetic
- differentiation
- integration
- simplification
- expansion
- factorization
"""

import sympy as sp


class MathEngine:

    def __init__(self):
        self.symbols = {}

    # ==================================================
    # Expression Parser
    # ==================================================

    def parse_expression(self, expression):
        """
        Convert a text expression into a SymPy expression.
        """

        expression = expression.strip()

        if not expression:
            raise ValueError("Expression is empty.")

        # Common mathematical notation
        expression = expression.replace("^", "**")

        # Create common symbols automatically
        symbols = {
            name: sp.Symbol(name)
            for name in [
                "x",
                "y",
                "z",
                "t",
                "a",
                "b",
                "c",
                "n",
            ]
        }

        return sp.sympify(
            expression,
            locals=symbols,
        )

    # ==================================================
    # Arithmetic
    # ==================================================

    def calculate(self, expression):
        """
        Evaluate/simplify an arithmetic expression.
        """

        parsed = self.parse_expression(
            expression
        )

        return sp.simplify(parsed)

    # ==================================================
    # Differentiation
    # ==================================================

    def differentiate(
        self,
        expression,
        variable="x",
    ):
        """
        Differentiate an expression with respect
        to the specified variable.
        """

        parsed = self.parse_expression(
            expression
        )

        symbol = sp.Symbol(variable)

        return sp.diff(
            parsed,
            symbol,
        )

    # ==================================================
    # Integration
    # ==================================================

    def integrate(
        self,
        expression,
        variable="x",
    ):
        """
        Integrate an expression with respect
        to the specified variable.
        """

        parsed = self.parse_expression(
            expression
        )

        symbol = sp.Symbol(variable)

        return sp.integrate(
            parsed,
            symbol,
        )

    # ==================================================
    # Simplification
    # ==================================================

    def simplify(self, expression):
        """
        Simplify a mathematical expression.
        """

        parsed = self.parse_expression(
            expression
        )

        return sp.simplify(
            parsed
        )

    # ==================================================
    # Expansion
    # ==================================================

    def expand(self, expression):
        """
        Expand an expression.
        """

        parsed = self.parse_expression(
            expression
        )

        return sp.expand(
            parsed
        )

    # ==================================================
    # Factorization
    # ==================================================

    def factor(self, expression):
        """
        Factor an expression.
        """

        parsed = self.parse_expression(
            expression
        )

        return sp.factor(
            parsed
        )

    # ==================================================
    # Generic Operation
    # ==================================================

    def execute(
        self,
        operation,
        expression,
        variable="x",
    ):
        """
        Execute a mathematical operation.
        """

        operation = operation.lower().strip()

        if operation in (
            "calculate",
            "calc",
            "evaluate",
            "evaluate_expression",
        ):
            return self.calculate(
                expression
            )

        if operation in (
            "differentiate",
            "derivative",
            "diff",
        ):
            return self.differentiate(
                expression,
                variable,
            )

        if operation in (
            "integrate",
            "integral",
        ):
            return self.integrate(
                expression,
                variable,
            )

        if operation in (
            "simplify",
        ):
            return self.simplify(
                expression
            )

        if operation in (
            "expand",
        ):
            return self.expand(
                expression
            )

        if operation in (
            "factor",
            "factorize",
            "factorise",
        ):
            return self.factor(
                expression
            )

        raise ValueError(
            f"Unsupported mathematical operation: "
            f"{operation}"
        )

"""

SymPy performs the calculation.

from plugins.base import AtlasPlugin

from plugins.math_engine.engine import MathEngine


class MathPlugin(AtlasPlugin):

    name = "math"

    display_name = "Mathematics"

    description = (
        "Symbolic mathematics including "
        "differentiation, integration, algebra, "
        "simplification, factoring and equations."
    )

    icon = "∑"

    has_gui = False

    gui_id = "math"

    triggers = [
        "differentiate",
        "derivative",
        "differentiate",
        "integrate",
        "integral",
        "simplify",
        "factor",
        "expand",
        "solve",
    ]

    # ==================================================
    # Can Handle
    # ==================================================

    def can_handle(
        self,
        message,
    ):

        text = (
            message
            .lower()
            .strip()
        )

        for trigger in self.triggers:

            if text == trigger:
                return True

            if text.startswith(
                trigger + " "
            ):

                return True

        return False

    # ==================================================
    # Run
    # ==================================================

    def run(
        self,
        expression="",
        operation="",
        variable="t",
        **kwargs,
    ):

        try:

            operation = (
                operation
                .lower()
                .strip()
            )

            # --------------------------------------------------
            # Differentiation
            # --------------------------------------------------

            if operation in (
                "differentiate",
                "derivative",
                "diff",
            ):

                return MathEngine.differentiate(
                    expression,
                    variable,
                )

            # --------------------------------------------------
            # Integration
            # --------------------------------------------------

            if operation in (
                "integrate",
                "integral",
            ):

                return MathEngine.integrate(
                    expression,
                    variable,
                )

            # --------------------------------------------------
            # Simplify
            # --------------------------------------------------

            if operation == "simplify":

                return MathEngine.simplify(
                    expression
                )

            # --------------------------------------------------
            # Expand
            # --------------------------------------------------

            if operation == "expand":

                return MathEngine.expand(
                    expression
                )

            # --------------------------------------------------
            # Factor
            # --------------------------------------------------

            if operation == "factor":

                return MathEngine.factor(
                    expression
                )

            # --------------------------------------------------
            # Solve
            # --------------------------------------------------

            if operation == "solve":

                return MathEngine.solve(
                    expression,
                    variable,
                )

            raise ValueError(
                f"Unsupported mathematical operation: "
                f"{operation}"
            )

        except Exception as error:

            return {
                "operation": operation,
                "error": str(error),
            }

import sympy as sp


class MathFormatter:

    # ======================================================
    # Basic
    # ======================================================

    @staticmethod
    def text(
        expression,
    ):

        if isinstance(
            expression,
            sp.Basic,
        ):

            return sp.sstr(
                expression
            )

        return str(
            expression
        )

    # ======================================================
    # Equation
    # ======================================================

    @staticmethod
    def equation(
        left,
        right,
    ):

        return (
            f"`{MathFormatter.text(left)}`"
            " = "
            f"**{MathFormatter.text(right)}**"
        )

    # ======================================================
    # Result Card
    # ======================================================

    @staticmethod
    def result(
        expression,
        result,
    ):

        return (
            "### Result\n\n"
            f"`{expression}` = **{result}**"
        )

    # ======================================================
    # Error
    # ======================================================

    @staticmethod
    def error(
        message,
    ):

        return (
            "### Math Error\n\n"
            f"{message}"
        )

"""
Atlas V3 Conversation Manager

Manages the active conversation independently
from the LLM implementation.
"""

from datetime import datetime
from uuid import uuid4

from config import MAX_HISTORY


class ConversationManager:

    def __init__(self):

        self.conversation_id = self._new_id()
        self.title = "New Chat"
        self.created_at = self._timestamp()
        self.updated_at = self.created_at

        self.messages = []

    # ==================================================
    # Conversation Identity
    # ==================================================

    @staticmethod
    def _new_id():

        return str(
            uuid4()
        )

    @staticmethod
    def _timestamp():

        return datetime.now().isoformat(
            timespec="seconds"
        )

    # ==================================================
    # Add Messages
    # ==================================================

    def add_user(
        self,
        text,
    ):

        self.messages.append(
            {
                "role": "user",
                "content": text,
            }
        )

        self._update()

        self._trim()

    def add_assistant(
        self,
        text,
    ):

        self.messages.append(
            {
                "role": "assistant",
                "content": text,
            }
        )

        self._update()

        self._trim()

    # ==================================================
    # Retrieval
    # ==================================================

    def history(self):

        return self.messages.copy()

    def last(
        self,
        amount=None,
    ):

        if amount is None:
            amount = MAX_HISTORY

        return self.messages[-amount:]

    # ==================================================
    # Conversation Information
    # ==================================================

    def get_id(self):

        return self.conversation_id

    def get_title(self):

        return self.title

    def get_created_at(self):

        return self.created_at

    def get_updated_at(self):

        return self.updated_at

    # ==================================================
    # Title
    # ==================================================

    def set_title(
        self,
        title,
    ):

        title = title.strip()

        if not title:
            return

        self.title = title

        self._update()

    # ==================================================
    # Clear
    # ==================================================

    def clear(self):

        self.messages.clear()

        self.title = "New Chat"

        self._update()

    # ==================================================
    # New Conversation
    # ==================================================

    def new_conversation(self):

        self.conversation_id = self._new_id()

        self.title = "New Chat"

        self.created_at = self._timestamp()

        self.updated_at = self.created_at

        self.messages.clear()

    # ==================================================
    # Serialization
    # ==================================================

    def to_dict(self):

        return {
            "id": self.conversation_id,
            "title": self.title,
            "created_at": self.created_at,
            "updated_at": self.updated_at,
            "messages": self.messages.copy(),
        }

    # ==================================================
    # Loading
    # ==================================================

    def load(
        self,
        data,
    ):

        self.conversation_id = data.get(
            "id",
            self._new_id(),
        )

        self.title = data.get(
            "title",
            "New Chat",
        )

        self.created_at = data.get(
            "created_at",
            self._timestamp(),
        )

        self.updated_at = data.get(
            "updated_at",
            self.created_at,
        )

        self.messages = data.get(
            "messages",
            [],
        ).copy()

        self._trim()

    # ==================================================
    # Internal
    # ==================================================

    def _update(self):

        self.updated_at = self._timestamp()

    def _trim(self):

        if len(self.messages) > MAX_HISTORY:

            self.messages = self.messages[
                -MAX_HISTORY:
            ]

    # Tool AI Signals
    # ==================================================

    # Used by the GUI/tool workspace.

    tool_chunk_ready = Signal(str)

        self.bus = bus
        self.state = state

        # ==================================================

        # ==================================================

        self.thread = QThread()

        self.worker = AIWorker()

        self.worker.moveToThread(
            self.thread
        )

        # ==================================================

            self.worker.process
        )

            self.chunk_ready.emit
        )

        self.worker.finished.connect(
            self.finished.emit
        )

        # ==================================================
        # Tool AI
        # ==================================================

        self.tool_requested.connect(
            self.worker.process_tool
        )

        self.worker.tool_chunk.connect(
            self.tool_chunk_ready.emit
        )

        self.worker.tool_finished.connect(

        )

        self.worker.tool_error.connect(
            self.tool_error.emit
        )

        logger.info(

    # ==================================================
    # Tool AI
    # ==================================================

    def generate_tool(

"""
Atlas V3 Workers

Background workers for long-running AI tasks.
"""

from PySide6.QtCore import (
    QObject,
    Signal,
    Slot,
)

from ai.brain import AtlasBrain


class AIWorker(QObject):

    # ==================================================
    # Signals
    # ==================================================

    chunk = Signal(str)

    error = Signal(str)

    finished = Signal()

    tool_chunk = Signal(str)

    tool_finished = Signal()

    tool_error = Signal(str)

    # ==================================================
    # Initialization
    # ==================================================

    def __init__(self):

        super().__init__()

        self.brain = AtlasBrain()

    # ==================================================
    # Normal Chat
    # ==================================================

    @Slot(str)
    def process(
        self,
        message,
    ):

        try:

            for piece in self.brain.process(
                message
            ):

                self.chunk.emit(
                    piece
                )

        except Exception as exc:

            self.error.emit(
                str(exc)
            )

        finally:

            self.finished.emit()

    # ==================================================
    # Tool AI
    # ==================================================

    @Slot(str)
    def process_tool(
        self,
        message,
    ):

        """
        Process a lightweight AI request for a tool.

        IMPORTANT:

        This intentionally does NOT call:

            brain.process()

        because brain.process() invokes the entire
        Atlas conversational pipeline.

        Tool explanations should go directly through
        AtlasModel.
        """

        try:

            for piece in self.brain.model.generate(
                message
            ):

                self.tool_chunk.emit(
                    piece
                )

        except Exception as exc:

            self.tool_error.emit(
                str(exc)
            )

        finally:

            self.tool_finished.emit()

"""
Atlas V3 Logging System

Centralized logging for the Atlas runtime,
web interface, and GUI.
"""

import logging
from logging.handlers import RotatingFileHandler

from config import LOG_FILE


LOG_DIR = LOG_FILE.parent

LOG_DIR.mkdir(
    parents=True,
    exist_ok=True,
)


LOG_FORMAT = (
    "%(asctime)s | "
    "%(levelname)s | "
    "%(name)s | "
    "%(message)s"
)


def get_logger(name="atlas"):

    logger = logging.getLogger(name)

    # Prevent duplicate handlers if this module
    # is imported multiple times.
    if logger.handlers:
        return logger

    logger.setLevel(logging.INFO)

    logger.propagate = False

    formatter = logging.Formatter(
        LOG_FORMAT
    )

    # --------------------------------------------------
    # File Handler
    # --------------------------------------------------

    file_handler = RotatingFileHandler(
        LOG_FILE,
        maxBytes=5 * 1024 * 1024,
        backupCount=3,
        encoding="utf-8",
    )

    file_handler.setFormatter(
        formatter
    )

    logger.addHandler(
        file_handler
    )

    # --------------------------------------------------
    # Console Handler
    # --------------------------------------------------

    console_handler = logging.StreamHandler()

    console_handler.setFormatter(
        formatter
    )

    logger.addHandler(
        console_handler
    )

    return logger


logger = get_logger("atlas")

"""
Atlas V3 Event Bus

The Event Bus allows independent components
to communicate without directly referencing
each other.
"""

from collections import defaultdict
from typing import Callable, Type

from core.events import Event


class EventBus:
    """
    Publish/Subscribe event bus.
    """

    def __init__(self):

        self._listeners = defaultdict(list)

    # ------------------------------------------------------
    # Subscribe
    # ------------------------------------------------------

    def subscribe(
        self,
        event_type: Type[Event],
        callback: Callable,
    ):

        if callback not in self._listeners[event_type]:

            self._listeners[event_type].append(
                callback
            )

    # ------------------------------------------------------
    # Unsubscribe
    # ------------------------------------------------------

    def unsubscribe(
        self,
        event_type: Type[Event],
        callback: Callable,
    ):

        if callback in self._listeners[event_type]:

            self._listeners[event_type].remove(
                callback
            )

    # ------------------------------------------------------
    # Publish
    # ------------------------------------------------------

    def publish(self, event: Event):

        for callback in self._listeners[type(event)]:

            callback(event)

"""
Atlas V3 Controller

Coordinates communication between
the GUI, Web API and backend services.
"""

from core.tool_manager import ToolManager
from core.ai_service import AIService

from core.events import (
    ChunkReceived,
    ConversationCleared,
    MessageReceived,
    ThinkingFinished,
    ThinkingStarted,
)

from core.logger import logger


class Controller:

    def __init__(
        self,
        bus,
        state,
    ):

        self.bus = bus

        self.state = state

        self.tools = ToolManager()

        self.ai = AIService(
            bus,
            state,
        )

        # ==================================================
        # Normal AI
        # ==================================================

        self.ai.chunk_ready.connect(
            self.on_chunk
        )

        self.ai.finished.connect(
            self.on_finished
        )

        # ==================================================
        # Tool AI
        # ==================================================

        self.ai.tool_chunk_ready.connect(
            self.on_tool_chunk
        )

        self.ai.tool_finished.connect(
            self.on_tool_finished
        )

        self.ai.tool_error.connect(
            self.on_tool_error
        )

        # Tool response currently being generated
        self.tool_response_text = ""

    # ==================================================
    # Lifecycle
    # ==================================================

    def start(self):

        logger.info(
            "Starting Controller..."
        )

        self.bus.subscribe(
            MessageReceived,
            self.on_message_received,
        )

        self.bus.subscribe(
            ConversationCleared,
            self.on_conversation_cleared,
        )

        self.ai.start()

        self._sync_active_conversation()

    def stop(self):

        logger.info(
            "Stopping Controller..."
        )

        self.bus.unsubscribe(
            MessageReceived,
            self.on_message_received,
        )

        self.bus.unsubscribe(
            ConversationCleared,
            self.on_conversation_cleared,
        )

        self.ai.stop()

    # ==================================================
    # Synchronous API
    # ==================================================

    def chat(
        self,
        message,
    ):

        response = ""

        for chunk in self.ai.worker.brain.process(
            message
        ):

            response += chunk

        self._sync_active_conversation()

        return response

    # ==================================================
    # Streaming API
    # ==================================================

    def stream_chat(
        self,
        message,
    ):

        for chunk in self.ai.worker.brain.process(
            message
        ):

            yield chunk

        self._sync_active_conversation()

    # ==================================================
    # Tool AI API
    # ==================================================

    def ask_tool_ai(
        self,
        message,
    ):

        """
        Send a lightweight AI request to a tool.

        This does NOT enter the normal AtlasBrain
        conversation pipeline.
        """

        self.tool_response_text = ""

        self.ai.generate_tool(
            message
        )

    # ==================================================
    # Tool API
    # ==================================================

    def available_tools(
        self,
    ):

        return self.tools.metadata()

    def execute_tool(
        self,
        name,
        **kwargs,
    ):

        return self.tools.execute(
            name,
            **kwargs,
        )

    # ==================================================
    # Tool AI Events
    # ==================================================

    def on_tool_chunk(
        self,
        text,
    ):

        self.tool_response_text += text

    def on_tool_finished(
        self,
    ):

        logger.info(
            "Tool AI response finished."
        )

        # Emit through the existing AI service signal
        # so tool GUIs can receive the complete response.
        self.ai.tool_response.emit(
            self.tool_response_text
        )

        self.tool_response_text = ""

    def on_tool_error(
        self,
        error,
    ):

        logger.error(
            f"Tool AI error: {error}"
        )

        self.ai.tool_response.emit(
            f"Atlas error: {error}"
        )

        self.tool_response_text = ""

    # ==================================================
    # Conversation API
    # ==================================================

    def new_conversation(
        self,
    ):

        conversation_id = (
            self.ai.worker.brain.new_conversation()
        )

        self.state.active_conversation_id = (
            conversation_id
        )

        self.state.current_message = ""

        self.state.current_response = ""

        self.state.status = "Ready"

        self.state.is_generating = False

        logger.info(
            f"New conversation created: "
            f"{conversation_id}"
        )

        return conversation_id

    def load_conversation(
        self,
        conversation_id,
    ):

        success = (
            self.ai.worker.brain.load_conversation(
                conversation_id
            )
        )

        if not success:

            logger.warning(
                f"Failed to load conversation: "
                f"{conversation_id}"
            )

            return False

        self.state.active_conversation_id = (
            self.ai.worker.brain
            .get_active_conversation_id()
        )

        self.state.current_message = ""

        self.state.current_response = ""

        self.state.status = "Ready"

        self.state.is_generating = False

        logger.info(
            f"Conversation loaded: "
            f"{conversation_id}"
        )

        return True

    def get_active_conversation(
        self,
    ):

        return (
            self.ai.worker.brain
            .conversations
            .get()
        )

    def delete_conversation(
        self,
        conversation_id,
    ):

        success = (
            self.ai.worker.brain
            .delete_conversation(
                conversation_id
            )
        )

        if success:

            self._sync_active_conversation()

            logger.info(
                f"Conversation deleted: "
                f"{conversation_id}"
            )

        return success

    def rename_conversation(
        self,
        conversation_id,
        title,
    ):

        success = (
            self.ai.worker.brain
            .rename_conversation(
                conversation_id,
                title,
            )
        )

        if success:

            logger.info(
                f"Conversation renamed: "
                f"{conversation_id}"
            )

        return success

    def list_conversations(
        self,
    ):

        return (
            self.ai.worker.brain
            .list_conversations()
        )

    # ==================================================
    # Event Handlers
    # ==================================================

    def on_message_received(
        self,
        event,
    ):

        self.state.current_message = (
            event.text
        )

        self.state.current_response = ""

        self.state.status = "Thinking"

        self.state.is_generating = True

        self.bus.publish(
            ThinkingStarted()
        )

        self.ai.generate(
            event.text
        )

    def on_chunk(
        self,
        text,
    ):

        self.state.current_response += text

        self.bus.publish(
            ChunkReceived(
                text
            )
        )

    def on_finished(
        self,
    ):

        self.state.status = "Ready"

        self.state.is_generating = False

        self._sync_active_conversation()

        self.bus.publish(
            ThinkingFinished()
        )

    def on_conversation_cleared(
        self,
        event,
    ):

        conversation_id = (
            self.new_conversation()
        )

        logger.info(
            f"Conversation cleared. "
            f"Active conversation: "
            f"{conversation_id}"
        )

    # ==================================================
    # Internal
    # ==================================================

    def _sync_active_conversation(
        self,
    ):

        brain = self.ai.worker.brain

        self.state.active_conversation_id = (
            brain.get_active_conversation_id()
        )

"""
Atlas V3 Intent Router
"""


class IntentRouter:

    def __init__(self, plugins):
        self.plugins = plugins

    def match(self, message):

        text = message.lower().strip()

        best_plugin = None
        best_score = 0

        for plugin in self.plugins.values():

            if not plugin.enabled:
                continue

            if not plugin.can_handle(message):
                continue

            trigger_score = 0

            for trigger in plugin.triggers:

                trigger = trigger.lower().strip()

                if not trigger:
                    continue

                # Exact command
                if text == trigger:
                    trigger_score = max(
                        trigger_score,
                        1000,
                    )

                # Command followed by arguments
                elif text.startswith(
                    trigger + " "
                ):
                    trigger_score = max(
                        trigger_score,
                        100,
                    )

                # Trigger appears as a complete word/phrase
                elif (
                    f" {trigger} "
                    in f" {text} "
                ):
                    trigger_score = max(
                        trigger_score,
                        len(trigger),
                    )

            # IMPORTANT:
            # No trigger match = plugin does not match.
            if trigger_score == 0:
                continue

            score = (
                trigger_score
                + plugin.priority
            )

            if score > best_score:

                best_score = score
                best_plugin = plugin

        return best_plugin

    def extract_arguments(
        self,
        plugin,
        message,
    ):

        expression = message.strip()

        for trigger in sorted(
            plugin.triggers,
            key=len,
            reverse=True,
        ):

            trigger_lower = trigger.lower()

            if expression.lower().startswith(
                trigger_lower
            ):

                expression = expression[
                    len(trigger):
                ].strip()

                break

        return expression.rstrip(
            "?!.,;:"
        ).strip()

"""
Atlas V3 Plugin Manager
"""

import importlib
import inspect
import pkgutil

from plugins.base import AtlasPlugin


class PluginManager:

    def __init__(self):

        self.plugins = {}

        self.load_plugins()

    # ==================================================
    # Loading
    # ==================================================

    def load_plugins(self):

        import plugins.system

        for _, module_name, _ in pkgutil.iter_modules(
            plugins.system.__path__
        ):

            module = importlib.import_module(
                f"plugins.system.{module_name}"
            )

            for _, obj in inspect.getmembers(
                module,
                inspect.isclass,
            ):

                if not issubclass(
                    obj,
                    AtlasPlugin,
                ):
                    continue

                if obj is AtlasPlugin:
                    continue

                plugin = obj()

                if not plugin.enabled:
                    continue

                if not plugin.name:
                    continue

                self.plugins[
                    plugin.name
                ] = plugin

    # ==================================================
    # Execution
    # ==================================================

    def execute(
        self,
        name,
        **kwargs,
    ):

        plugin = self.plugins.get(name)

        if plugin is None:
            raise ValueError(
                f"Unknown plugin: {name}"
            )

        return plugin.run(
            **kwargs
        )

    # ==================================================
    # Information
    # ==================================================

    def available(self):

        return list(
            self.plugins.keys()
        )

"""
Atlas V3 Memory Manager

Persistent user memory stored in the configured data directory.
"""

import json

from config import MEMORY_FILE


class MemoryManager:

    def __init__(self):
        self.file = MEMORY_FILE

        self.file.parent.mkdir(
            parents=True,
            exist_ok=True,
        )

        if not self.file.exists():
            self._save({})

        else:
            try:
                data = self._load()

                if not isinstance(data, dict):
                    self._save({})

            except (json.JSONDecodeError, OSError):
                self._save({})

    # --------------------------------------------------
    # Internal
    # --------------------------------------------------

    def _load(self):
        try:
            with self.file.open(
                "r",
                encoding="utf-8",
            ) as f:
                data = json.load(f)

            return data if isinstance(data, dict) else {}

        except (
            FileNotFoundError,
            json.JSONDecodeError,
            OSError,
        ):
            return {}

    def _save(self, data):
        temporary = self.file.with_suffix(".tmp")

        with temporary.open(
            "w",
            encoding="utf-8",
        ) as f:
            json.dump(
                data,
                f,
                indent=4,
                ensure_ascii=False,
            )

        temporary.replace(self.file)

    # --------------------------------------------------
    # Public API
    # --------------------------------------------------

    def remember(
        self,
        category,
        key,
        value,
    ):
        data = self._load()

        if category not in data:
            data[category] = {}

        data[category][key] = value

        self._save(data)

    def recall(
        self,
        category,
        key,
    ):
        data = self._load()

        return data.get(
            category,
            {},
        ).get(key)

    def forget(
        self,
        category,
        key,
    ):
        data = self._load()

        if category not in data:
            return

        data[category].pop(
            key,
            None,
        )

        if not data[category]:
            del data[category]

        self._save(data)

    def all(self):
        return self._load()

    def clear(self):
        self._save({})
